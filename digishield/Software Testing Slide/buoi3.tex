\section{Kiểm thử hộp trắng}
%=====================================================================

%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}[shrink=8]{Kiểm thử hộp trắng là gì?}
  \begin{block}{Định nghĩa}
    Kiểm thử hộp trắng (white-box testing) thiết kế test case
    \textbf{dựa trên cấu trúc bên trong} của phần mềm: mã nguồn,
    luồng điều khiển, luồng dữ liệu.\sref{1,2}
  \end{block}

  \vspace{0.3em}
  \begin{columns}
    \column{0.48\textwidth}
    \textbf{Tên gọi khác:}
    \begin{itemize}
      \item Structural testing (kiểm thử cấu trúc)
      \item Glass-box / clear-box testing
      \item Code-based testing
    \end{itemize}

    \column{0.52\textwidth}
    \textbf{Mục tiêu:}
    \begin{itemize}
      \item Mọi câu lệnh được thực thi ít nhất một lần
      \item Mọi nhánh true/false đều được đi qua
      \item Phát hiện dead code, nhánh cài đặt sai
      \item \textbf{Đo được} chất lượng bộ test bằng độ phủ (coverage)
    \end{itemize}
  \end{columns}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}{Hộp trắng vs Hộp đen}
  \renewcommand{\arraystretch}{1.2}
  \footnotesize
  \begin{tabularx}{\textwidth}{l>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X}
    \toprule
     & \textbf{Hộp trắng} & \textbf{Hộp đen} \\
    \midrule
    Cơ sở thiết kế test & Mã nguồn, CFG, luồng dữ liệu & Đặc tả, yêu cầu, giao diện \\
    Cần biết code? & Có (đọc và hiểu code) & Không \\
    Ai thực hiện & Developer, tester biết lập trình & Tester, BA, người dùng \\
    Mức áp dụng chủ yếu & Unit test, integration test & System test, acceptance test \\
    Phát hiện tốt & Nhánh sai, dead code, lỗi logic cài đặt & Thiếu chức năng, sai so với yêu cầu \\
    Thước đo & Độ phủ câu lệnh / nhánh / đường đi & Độ phủ yêu cầu, lớp tương đương \\
    \bottomrule
  \end{tabularx}
  \renewcommand{\arraystretch}{1}

  \vspace{0.3em}
  {\small \textbf{Hai kỹ thuật bổ sung cho nhau}: hộp trắng trả lời ``code
  \textbf{đã viết} có được test đủ chưa?''; hộp đen trả lời ``phần mềm có làm
  \textbf{đúng yêu cầu} không?''.}
\end{frame}

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}[shrink=8]{Khi nào dùng kiểm thử hộp trắng?}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Tình huống điển hình:}
    \begin{itemize}
      \item Viết \textbf{unit test} cho service, hàm tiện ích
        (buổi 5--6 sẽ cài đặt bằng JUnit/Mockito)
      \item Code review: chỉ ra nhánh chưa được test
      \item Nâng độ phủ theo yêu cầu dự án
        (DigiShield đang ép line coverage $\geq$ 10\%, mục tiêu 50\%)
      \item Kiểm thử logic nghiệp vụ nhiều rẽ nhánh:
        chấm điểm rủi ro, quyết định can thiệp giao dịch
    \end{itemize}

    \column{0.46\textwidth}
    \begin{exampleblock}{Trên DigiShield hôm nay}
      Ta sẽ phân tích 2 method \textbf{thật}:
      \begin{itemize}
        \item \texttt{ReportingServiceImpl\allowbreak.ageLabel()} --- nhãn tuổi báo cáo phishing
        \item \texttt{InterceptionServiceImpl\allowbreak.evaluate()} --- quyết định
          chặn giao dịch đáng ngờ
      \end{itemize}
      và đối chiếu với \textbf{unit test thật} trong repo.
    \end{exampleblock}
  \end{columns}
\end{frame}

\section{Đồ thị luồng điều khiển (CFG)}

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}[shrink=8]{Control Flow Graph --- Định nghĩa}
  \begin{block}{Định nghĩa}
    CFG là đồ thị có hướng $G = (N, E)$:
    \begin{itemize}
      \item $N$: tập \textbf{node} --- mỗi node là một khối câu lệnh
            tuần tự (basic block)
      \item $E$: tập \textbf{cạnh có hướng} --- mỗi cạnh là một bước chuyển
            điều khiển có thể xảy ra
    \end{itemize}
  \end{block}

  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Các loại node:}
    \begin{itemize}
      \item \textbf{Entry node}: điểm vào duy nhất của method
      \item \textbf{Exit node}: điểm kết thúc
      \item \textbf{Node quyết định} (decision): có $\geq 2$ cạnh ra
        --- sinh từ \texttt{if}, \texttt{while}, \texttt{for}, \texttt{switch}
      \item \textbf{Node hợp} (junction): có $\geq 2$ cạnh vào
    \end{itemize}

    \column{0.4\textwidth}
    \centering

      nd/.style={circle,draw=darkgray,fill=lightgray,minimum size=0.6cm,font=\scriptsize},

\end{frame}

\begin{frame}[shrink=8]{Quy ước vẽ CFG trong môn học}
  \begin{enumerate}
    \item Gộp các câu lệnh tuần tự liên tiếp thành \textbf{một node}
      (basic block) --- đồ thị gọn, không mất thông tin.
    \item Mỗi điều kiện rẽ nhánh vẽ bằng \textbf{hình thoi}, ghi rõ cạnh
      \textbf{T} (true) và \textbf{F} (false).
    \item Method có nhiều \texttt{return}: nối tất cả điểm \texttt{return}
      về \textbf{một exit node chung} --- để công thức $V(G) = E - N + 2$
      cho kết quả đúng.
    \item Đánh \textbf{số thứ tự} node để mô tả đường đi, ví dụ:
      $1 \to 2(T) \to 3 \to 7$.
  \end{enumerate}

  \vspace{0.3em}
  \begin{alertblock}{Lỗi hay gặp của sinh viên}
    Quên nhánh F của \texttt{if} không có \texttt{else};
    quên exit node chung khi method có nhiều \texttt{return};
    vẽ mỗi dòng code một node làm đồ thị rối.
  \end{alertblock}
\end{frame}

    if (req.onCall())   signals.add("ON_CALL");     // S1

        .findByTenantIdAndValue(tenantId, req.destAccount())

\begin{frame}{Độ phủ câu lệnh (Statement Coverage --- SC)}
  \begin{block}{Định nghĩa\sref{1,3}}
    \[
      SC = \frac{\text{số câu lệnh đã được thực thi bởi bộ test}}
                {\text{tổng số câu lệnh thực thi được}} \times 100\%
    \]
    Đạt 100\% khi \textbf{mỗi câu lệnh} (mỗi node của CFG) được thực thi
    ít nhất một lần.
  \end{block}

  \vspace{0.4em}
  \begin{itemize}
    \item Là tiêu chí \textbf{yếu nhất} --- mức tối thiểu mà mọi bộ unit test
      nên đạt cho code nghiệp vụ.
    \item Phát hiện được: dead code, câu lệnh ném exception chưa từng chạy.
    \item Công cụ đo tự động: JaCoCo (buổi 4) --- hôm nay ta \textbf{tính tay}
      trên CFG để hiểu bản chất.
  \end{itemize}
\end{frame}

\begin{frame}{Độ phủ nhánh (Branch/Decision Coverage --- BC)}
  \begin{block}{Định nghĩa\sref{1,3}}
    \[
      BC = \frac{\text{số nhánh đã được thực thi}}{\text{tổng số nhánh}} \times 100\%
    \]
    Mỗi node quyết định (\texttt{if}, \texttt{while}, \texttt{for},
    \texttt{switch}) phải có \textbf{mọi cạnh ra} (cả T lẫn F) được đi qua
    ít nhất một lần.
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Mạnh hơn SC vì:}
    \begin{itemize}
      \item 100\% BC $\Rightarrow$ 100\% SC (đi hết cạnh thì hết node)
      \item Chiều ngược lại \textbf{không đúng}: \texttt{if} không có
        \texttt{else} --- SC bỏ sót nhánh F
    \end{itemize}

    \begin{exampleblock}{Ngưỡng thực tế}
      \small
      \begin{tabular}{ll}

\end{frame}

\section{Basis path testing và $V(G)$}
%=====================================================================

%--- SLIDE 28 ---------------------------------------------------------
\begin{frame}{Độ phức tạp cyclomatic $V(G)$ --- McCabe, 1976 (1/2)}
  \begin{block}{Ba cách tính tương đương (đồ thị liên thông, 1 exit chung)\sref{5}}
    \[
      V(G) = E - N + 2 \quad
      V(G) = \text{số node quyết định} + 1 \quad
      V(G) = \text{số miền kín} + 1
    \]
    $E$: số cạnh, $N$: số node của CFG; miền kín = region khi vẽ đồ thị phẳng.

    \smallskip
    {\small \textit{Điều kiện áp dụng:} ``số node quyết định $+\,1$'' chỉ đúng
    khi mỗi node quyết định có đúng \textbf{2 cạnh ra}; với \texttt{switch}
    có $k$ cạnh ra (slide 11) dùng
    $V(G) = \sum(\text{bậc ra (out-degree)} - 1) + 1$.}
  \end{block}
\end{frame}

%--- SLIDE 28b --------------------------------------------------------
\begin{frame}{Độ phức tạp cyclomatic $V(G)$ --- McCabe, 1976 (2/2)}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Ý nghĩa:}
    \begin{itemize}
      \item = số đường đi \textbf{độc lập tuyến tính} tối đa
      \item = số test case \textbf{tối thiểu} cho basis path testing
      \item Thước đo độ phức tạp: $V(G) > 10$ $\Rightarrow$ method nên tách nhỏ
    \end{itemize}

    \column{0.44\textwidth}
    \begin{exampleblock}{Lưu ý khi tính}
      \small
      Method nhiều \texttt{return}: phải nối các \texttt{return}
      về \textbf{exit node chung}, nếu không $E - N + 2$ cho kết quả sai
      (nhỏ hơn thực tế).
    \end{exampleblock}
  \end{columns}

  \vspace{0.3em}
  {\small Buổi 4 sẽ đào sâu: dùng $V(G)$ chọn basis path có hệ thống,
  liên hệ với MC/DC và JaCoCo.}
\end{frame}

\section{Kiểm thử vòng lặp}

%--- SLIDE 36 ---------------------------------------------------------
\begin{frame}{Vòng lặp trong luồng điều khiển --- loop testing}
  \begin{block}{Vấn đề}
    Cạnh quay lui của vòng lặp tạo \textbf{vô số} đường đi
    (0 lần, 1 lần, 2 lần, ... lặp). Không thể phủ hết mọi đường
    $\Rightarrow$ chọn các trường hợp \textbf{đại diện}.\sref{3}
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Bộ test chuẩn cho vòng lặp đơn (số lần lặp tối đa $n$):}
    \begin{enumerate}
      \item \textbf{0 lần} --- nhảy qua vòng lặp
      \item \textbf{1 lần} lặp
      \item \textbf{2 lần} lặp
      \item Số lần \textbf{điển hình} $m$ ($2 < m < n$)
      \item $n-1$, $n$, và thử $n+1$ lần (nếu ép được)
    \end{enumerate}

    \column{0.44\textwidth}
    \begin{exampleblock}{Lỗi vòng lặp hay bắt được}
      \begin{itemize}
        \item Sai điều kiện dừng (off-by-one)
        \item Biến tích lũy khởi tạo sai
        \item Xử lý sai khi tập dữ liệu \textbf{rỗng}
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

    \begin{alertblock}{Hạn chế}
      \begin{itemize}
        \item \textbf{Code thiếu}: yêu cầu chưa được cài đặt thì

        \item Đường đi \textbf{bất khả thi} (infeasible path) làm mục tiêu

      \begin{itemize}
        \item Kết hợp \textbf{hộp đen} (EP, BVA, bảng quyết định --- buổi 8)
          để soi từ phía yêu cầu
        \item Traceability: mỗi yêu cầu $\to$ ít nhất 1 test case
        \item Review chéo đặc tả và code
      \end{itemize}
    \end{block}

    \begin{exampleblock}{Ghi nhớ}
      Độ phủ đo ``bộ test đã chạm bao nhiêu \textbf{code}'',
      không đo ``phần mềm đúng bao nhiêu \textbf{yêu cầu}''.
    \end{exampleblock}
  \end{columns}
\end{frame}

public ModerationView moderate(String content) {
    String text = content == null
        ? "" : content.toLowerCase(Locale.ROOT);
    List<String> reasons = new ArrayList<>();
    for (String word : UNSAFE_WORDS) {  // 10 tu khoa
        if (text.contains(word)) {
            reasons.add("Noi dung chua tu khong an toan: "
                + word);
        }
    }
    String verdict;
    if (reasons.isEmpty()) {
        verdict = "pass";
    } else if (reasons.size() >= 2) {
        verdict = "block";
    } else {
        verdict = "flag";
    }
    return new ModerationView(verdict, List.copyOf(reasons));
}

public int benchmark(RiskScope scope) {
    UUID tenantId = TenantContext.requireUuid();
    List<RiskScore> scores = riskScoreRepository
        .findByTenantIdAndScope(tenantId, scope);
    if (scores.isEmpty()) {
        return 0;               // danh sach rong
    }
    long sum = scores.stream()
        .mapToInt(RiskScore::getValue).sum();
    return (int) (sum / scores.size());
}
\end{lstlisting}
  \end{block}

    \item Thiết kế bộ test 100\% BC --- chú ý trường hợp \textbf{list rỗng
      $\to$ 0} (nhánh chống chia cho 0) và áp dụng loop testing cho phần
      tính tổng: 0 / 1 / nhiều phần tử.
