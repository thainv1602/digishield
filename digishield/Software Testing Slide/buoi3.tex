\section{Kiểm thử hộp trắng}

%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}[shrink=8]{Kiểm thử hộp trắng là gì?}
  \begin{block}{Định nghĩa}
    Kiểm thử hộp trắng (white-box testing) thiết kế test case
    \textbf{dựa trên cấu trúc bên trong} của phần mềm: mã nguồn,
    luồng điều khiển, luồng dữ liệu.\sref{1,2}
  \end{block}

  \vspace{0.3em}
  \begin{columns}
    \column{0.48\textwidth}
    \textbf{Tên gọi khác:}
    \begin{itemize}
      \item Structural testing (kiểm thử cấu trúc)
      \item Glass-box / clear-box testing
      \item Code-based testing
    \end{itemize}

    \column{0.52\textwidth}
    \textbf{Mục tiêu:}
    \begin{itemize}
      \item Mọi câu lệnh được thực thi ít nhất một lần
      \item Mọi nhánh true/false đều được đi qua
      \item Phát hiện dead code, nhánh cài đặt sai
      \item \textbf{Đo được} chất lượng bộ test bằng độ phủ (coverage)
    \end{itemize}
  \end{columns}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}{Hộp trắng vs Hộp đen}
  \renewcommand{\arraystretch}{1.2}
  \footnotesize
  \begin{tabularx}{\textwidth}{l>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X}
    \toprule
     & \textbf{Hộp trắng} & \textbf{Hộp đen} \\
    \midrule
    Cơ sở thiết kế test & Mã nguồn, CFG, luồng dữ liệu & Đặc tả, yêu cầu, giao diện \\
    Cần biết code? & Có (đọc và hiểu code) & Không \\
    Ai thực hiện & Developer, tester biết lập trình & Tester, BA, người dùng \\
    Mức áp dụng chủ yếu & Unit test, integration test & System test, acceptance test \\
    Phát hiện tốt & Nhánh sai, dead code, lỗi logic cài đặt & Thiếu chức năng, sai so với yêu cầu \\
    Thước đo & Độ phủ Câu lệnh (SC) / Nhánh (BC) / Đường đi (PC) & Độ phủ yêu cầu, lớp tương đương \\
    \bottomrule
  \end{tabularx}
  \renewcommand{\arraystretch}{1}

  \vspace{0.3em}
  {\small \textbf{Hai kỹ thuật bổ sung cho nhau}: hộp trắng trả lời ``code
  \textbf{đã viết} có được test đủ chưa?''; hộp đen trả lời ``phần mềm có làm
  \textbf{đúng yêu cầu} không?''.}
\end{frame}

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}[shrink=8]{Khi nào dùng kiểm thử hộp trắng?}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Tình huống điển hình:}
    \begin{itemize}
      \item Viết \textbf{unit test} cho service, hàm tiện ích
        (buổi 5--6 sẽ cài đặt bằng JUnit/Mockito)
      \item Code review: chỉ ra nhánh chưa được test
      \item Nâng độ phủ theo yêu cầu dự án
        (DigiShield đang ép line coverage $\geq$ 50\%)
      \item Kiểm thử logic nghiệp vụ nhiều rẽ nhánh:
        đánh giá rủi ro, quyết định \textbf{tạm dừng / cảnh báo / cho qua}
        một giao dịch
    \end{itemize}

    \column{0.46\textwidth}
    \begin{exampleblock}{Trên DigiShield hôm nay}
      Ta sẽ phân tích 2 method \textbf{thật}:
      \begin{itemize}
        \item \texttt{ReportingServiceImpl\allowbreak.ageLabel()} --- nhãn tuổi báo cáo phishing
        \item \texttt{InterceptionServiceImpl\allowbreak.evaluate()} --- quyết định
          tạm dừng, cảnh báo hoặc cho qua giao dịch
      \end{itemize}
      và đối chiếu với \textbf{unit test thật} trong repo.
    \end{exampleblock}
  \end{columns}
\end{frame}

\section{Control Flow Graph (CFG --- Đồ thị luồng điều khiển)}
%=====================================================================

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}[shrink=8]{Control Flow Graph (CFG --- Đồ thị luồng điều khiển)}
  \begin{block}{Định nghĩa}
    CFG là đồ thị có hướng $G = (N, E)$:
    \begin{itemize}
      \item $N$: tập \textbf{node} --- mỗi node là một khối câu lệnh
            tuần tự (basic block)
      \item $E$: tập \textbf{cạnh có hướng} --- mỗi cạnh là một bước chuyển
            điều khiển có thể xảy ra
    \end{itemize}
  \end{block}

  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Các loại node:}
    \begin{itemize}
      \item \textbf{Entry node}: điểm vào duy nhất của method
      \item \textbf{Exit node}: điểm kết thúc
      \item \textbf{Node quyết định} (decision): có $\geq 2$ cạnh ra
        --- sinh từ \texttt{if}, \texttt{while}, \texttt{for}, \texttt{switch}
      \item \textbf{Node hợp} (junction): có $\geq 2$ cạnh vào
    \end{itemize}

    \column{0.4\textwidth}
    \centering

      nd/.style={circle,draw=darkgray,fill=lightgray,minimum size=0.6cm,font=\scriptsize},
      dn/.style={diamond,draw=darkgray,fill=white,minimum size=0.7cm,font=\scriptsize,aspect=1.4},
      arr/.style={-{Stealth[length=3.5pt]},darkgray},
      lbl/.style={font=\tiny,midgray}

\end{frame}

\begin{frame}[shrink=8]{Quy ước vẽ CFG trong môn học}
  \begin{enumerate}
    \item Gộp các câu lệnh tuần tự liên tiếp thành \textbf{một node}
      (basic block) --- đồ thị gọn, không mất thông tin.
    \item Mỗi điều kiện rẽ nhánh vẽ bằng \textbf{hình thoi}, ghi rõ cạnh
      \textbf{T} (true) và \textbf{F} (false).
    \item Method có nhiều \texttt{return}: nối tất cả điểm \texttt{return}
      về \textbf{một exit node chung} --- để đường đi dễ nhìn hơn.
    \item Đánh \textbf{số thứ tự} node để mô tả đường đi, ví dụ:
      $1 \to 2(T) \to 3 \to 7$.
  \end{enumerate}

  \vspace{0.3em}
  \begin{alertblock}{Lỗi hay gặp của sinh viên}
    Quên nhánh F của \texttt{if} không có \texttt{else};
    quên exit node chung khi method có nhiều \texttt{return};
    vẽ mỗi dòng code một node làm đồ thị rối.
  \end{alertblock}
\end{frame}

    if (reportedAt == null) {              // node 1
        return null;                       // node 2
    }
    // node 3: gop 2 cau lenh tuan tu (tinh d, mins thanh 1 basic block)
    Duration d = Duration.between(reportedAt, now);
    long mins = Math.max(0, d.toMinutes());
    if (mins < 60) {                       // node 4
        return mins + "m";                 // node 5 - "N phut"

    long hours = d.toHours();              // node 6 (tinh hours)
    if (hours < 24) {                      // node 7
        return hours + "h";                // node 8 - "N gio"
    }
    return d.toDays() + "d";               // node 9 - "N ngay"
}

public InterventionDecision evaluate(EvaluateRequest req) {
    List<InterventionSignal> signals = new ArrayList<>();
    if (req.onCall())   signals.add(ON_CALL);       // S1
    if (req.newPayee()) signals.add(NEW_PAYEE);     // S2
    boolean watchlistHit = watchRepository
        .findFirstByTenantIdAndValueOrderByAddedAtDesc(
            tenantId, req.destAccount())
        .isPresent();
    if (watchlistHit)   signals.add(WATCHLIST_HIT); // S3
    Decision decision;
    if (req.onCall() && req.newPayee() && watchlistHit) {
        decision = Decision.PAUSE;                  // D1
    } else if (watchlistHit) {
        decision = Decision.WARN;                   // D2
    } else {
        decision = Decision.ALLOW;
    }
    eventRepository.save(event);  // luu: name() -> CHU HOA
    return new InterventionDecision(   // tra ve: chu thuong
            decision.name().toLowerCase(Locale.ROOT),
            signals.stream()
                .map(InterventionSignal::wireName).toList(),
            message);
}

\begin{frame}{Độ phủ câu lệnh (Statement Coverage --- SC)}
  \begin{block}{Định nghĩa\sref{1,3}}
    \[
      SC = \frac{\text{số câu lệnh đã được thực thi bởi bộ test}}
                {\text{tổng số câu lệnh thực thi được}} \times 100\%
    \]
    Đạt 100\% khi \textbf{mỗi câu lệnh} (mỗi node của CFG) được thực thi
    ít nhất một lần.
  \end{block}

  \vspace{0.4em}
  \begin{itemize}
    \item Là tiêu chí \textbf{yếu nhất} --- mức tối thiểu mà mọi bộ unit test
      nên đạt cho code nghiệp vụ.
    \item Phát hiện được: dead code, câu lệnh ném exception chưa từng chạy.
    \item Công cụ đo tự động: JaCoCo (xem phần cuối buổi 3) --- hôm nay ta \textbf{tính tay}
      trên CFG để hiểu bản chất.
  \end{itemize}
\end{frame}

\begin{frame}[shrink=10]{Độ phủ nhánh (Branch/Decision Coverage --- BC)}
  \begin{block}{Nguyên tắc: Mọi ngã rẽ đều phải test cả 2 hướng}
    Tại mỗi câu lệnh rẽ nhánh (\texttt{if}, \texttt{while}, \texttt{for}), luồng code tách đôi thành \textbf{True} và \textbf{False} (kể cả khi code không viết chữ \texttt{else}).
    \[
      BC = \frac{\text{Số nhánh True / False đã test}}{\text{Tổng số nhánh True / False}} \times 100\%
    \]
  \end{block}

  \vspace{0.2em}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{alertblock}{So sánh với Statement Coverage (SC)}
      \small
      \begin{itemize}
        \item \textbf{100\% BC $\Rightarrow$ 100\% SC}: Đi hết mọi ngã rẽ thì chắc chắn ghé qua mọi dòng code.
        \item \textbf{Ngược lại KHÔNG đúng}: 100\% SC vẫn có thể bỏ sót nhánh False của câu \texttt{if} không có \texttt{else}.
      \end{itemize}
    \end{alertblock}

    \column{0.42\textwidth}
    \begin{exampleblock}{Quy tắc đếm nhanh}
      \small
      \begin{itemize}
        \item 1 câu \texttt{if} $\to$ \textbf{2 nhánh} (T / F)
        \item 3 câu \texttt{if} $\to$ \textbf{6 nhánh}
        \item \textbf{BTL yêu cầu:} $BC > 90\%$
        \item (SonarQube chuẩn: 80\%)
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

    thêm T4 để phủ rõ các nhánh S1--S3.

\section{Basis path testing và \texorpdfstring{$V(G)$}{V(G)}}
%=====================================================================

%--- SLIDE 28 ---------------------------------------------------------
\begin{frame}[shrink=5]{Độ phức tạp cyclomatic $V(G)$ --- McCabe}
  \begin{block}{Bản chất: $V(G)$ là gì?}
    $V(G)$ là thước đo \textbf{độ phức tạp logic} của một method.\\
    $\Rightarrow$ Trả lời ngay câu hỏi: \textbf{``Cần viết bao nhiêu test case?''}
  \end{block}

  \vspace{0.2em}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \textbf{Ý nghĩa với Tester:}
    \begin{itemize}\small
      \item \textbf{Đường đi cơ sở (Basis Paths)}: Thay vì test hàng trăm tổ hợp, chỉ cần chọn đúng $V(G)$ kịch bản chạy độc lập để vét sạch mọi ngã rẽ.
      \item \textbf{Số test case}: Cần đúng $V(G)$ test để phủ 100\% đường đi cơ sở, và $V(G)$ là \textbf{mốc tối đa} để phủ 100\% nhánh.
    \end{itemize}

    \column{0.45\textwidth}
    \begin{alertblock}{Thang đo chất lượng code}
      \small
      \begin{itemize}
        \item \textbf{1 -- 10}: \textcolor{green!60!black}{Code tốt}, gọn, dễ test.
        \item \textbf{11 -- 20}: \textcolor{orange!90!black}{Khá rối}, bắt đầu rủi ro.
        \item \textbf{$> 20$}: \textcolor{red!80!black}{Code rối như mớ bòng bong}, rất khó test!
      \end{itemize}
      \smallskip
      {\footnotesize \textbf{Quy tắc vàng:} Nếu $V(G) > 10 \Rightarrow$ bắt buộc phải tách nhỏ method (refactor).}
    \end{alertblock}
  \end{columns}
\end{frame}

%--- SLIDE 28b --------------------------------------------------------
\begin{frame}{Tính $V(G)$ bằng cách đếm decision}
  \begin{block}{Công thức dùng nhanh khi nhìn code}
    \large
    \[
      V(G) = D + 1
    \]
    \small
    $D$ là số điểm rẽ nhánh: \texttt{if}, \texttt{else if},
    \texttt{for}, \texttt{while}, \texttt{case}.
  \end{block}

  \vspace{0.4em}
  \begin{alertblock}{Cách nhớ}
    \begin{itemize}
      \item Chỉ đếm các điểm hỏi \textbf{có / không}.
      \item \texttt{else} không tính thêm vì là nhánh False có sẵn.
      \item Mỗi decision thêm một ngã rẽ; cộng thêm 1 cho đường đi ban đầu.
    \end{itemize}
  \end{alertblock}
\end{frame}

%--- SLIDE 29 ---------------------------------------------------------
\begin{frame}[shrink=8]{Ví dụ tính $V(G)$}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{\texttt{ageLabel()}}
      \begin{itemize}
        \item 3 decision
        \item $V(G) = D + 1 = 3 + 1 = \mathbf{4}$
        \item Basis path = 4 đường P1--P4 (slide trước)
          $\Rightarrow$ tối thiểu \textbf{4 test case}

      \begin{itemize}
        \item 2 decision: PAUSE? WARN?
        \item $V(G) = D + 1 = 2 + 1 = \mathbf{3}$
        \item Basis path = 3 đường PAUSE / WARN / ALLOW
          $\Rightarrow$ tối thiểu \textbf{3 test case}
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.5em}
  \begin{alertblock}{Chốt cách hiểu}
    Chọn \textbf{một phạm vi CFG} rồi tính theo phạm vi đó.
    Ở đây \texttt{evaluate()} chỉ tính phần quyết định chính.
  \end{alertblock}
\end{frame}

\section{Kiểm thử vòng lặp}
%=====================================================================

%--- SLIDE 36 ---------------------------------------------------------
\begin{frame}[shrink=12]{Vòng lặp trong luồng điều khiển --- Loop Testing}
  \begin{block}{Vì sao không thể test mọi số lần lặp?}
    Vòng lặp tạo ra vô số đường đi ($0, 1, 2, \dots$ lần lặp). Kỹ thuật \textbf{Loop Testing} chọn ra \textbf{các mốc giá trị biên} để kiểm thử.\sref{3}
  \end{block}

  \vspace{0.1em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{\small Quy ước $n$ và $m$}
      \small
      \begin{itemize}\setlength{\itemsep}{0.25em}
        \item \textbf{$n$ = Số lần lặp TỐI ĐA}:\\
          {\footnotesize Trần tối đa theo nghiệp vụ (VD: mảng max $n = 10$ phần tử, OTP thử max 5 lần).}
        \item \textbf{$m$ = Số lần lặp ĐIỂN HÌNH}:\\
          {\footnotesize Giá trị ở khoảng giữa $2 < m < n$ (VD: với $n = 10$, chọn $m = 4$).}
      \end{itemize}
    \end{block}

    \column{0.50\textwidth}
    \begin{exampleblock}{\small Ý nghĩa các mốc kiểm thử}
      \small
      \renewcommand{\arraystretch}{1.15}
      \begin{tabular}{ll}
        \textbf{0 lần} & Rỗng (phủ nhánh False) \\
        \textbf{1 lần} & Biên min (khởi tạo đúng?) \\
        \textbf{2 lần} & Bước lặp \texttt{i++} chạy đúng? \\
        \textbf{$m$ lần} & Kịch bản thông thường \\
        \textbf{$n-1, n$} & Chạm trần (lỗi off-by-one) \\
        \textbf{$n+1$} & Vượt trần (chặn tràn mảng) \\
      \end{tabular}
      \renewcommand{\arraystretch}{1}
    \end{exampleblock}
  \end{columns}
\end{frame}

%--- SLIDE 37 ---------------------------------------------------------
\begin{frame}[fragile,shrink=8]{Ví dụ: Vòng lặp đếm từ khóa lừa đảo trong DigiShield}
  \begin{block}{\small Hàm \texttt{classify()} trong \texttt{StubAiClient.java} (Module AI)}
    Kiểm tra một đoạn tin nhắn xem chứa bao nhiêu từ khóa lừa đảo trong bộ 10 từ khóa (\texttt{"password"}, \texttt{"otp"}, \texttt{"urgent"}...):
\begin{lstlisting}[language=java]

  \begin{alertblock}{\small Câu hỏi đặt ra cho Tester}
    Danh sách có $n = 10$ từ khóa. Ta nên test những kịch bản nào để vừa vét sạch lỗi vòng lặp và câu lệnh cộng dồn \texttt{hits++}, vừa không tốn công vô ích?
  \end{alertblock}
\end{frame}

\begin{frame}{Thiết kế ca kiểm thử theo nguyên tắc Loop Testing}
  \begin{columns}[T]
    \column{0.60\textwidth}
    \begin{block}{\small 5 kịch bản kiểm thử theo mốc lặp chuẩn}
      \scriptsize
      \renewcommand{\arraystretch}{1.2}
      \begin{tabularx}{\linewidth}{@{}c l X@{}}
        \toprule
        \textbf{Mốc lặp} & \textbf{Dữ liệu tin nhắn} & \textbf{Ý nghĩa kiểm thử} \\
        \midrule
        \textbf{0 lần} & \texttt{"Xin chao ngay moi"} & 0 hit $\to$ Trả về \texttt{"clean"} \\
        \textbf{1 lần} & \texttt{"Nhap password ngay"} & 1 hit $\to$ Biên min phát hiện \\

\section{Kiểm thử luồng dữ liệu (Data Flow Testing)}
%=====================================================================

\begin{frame}[shrink=6]{Data Flow Testing: lần theo giá trị của biến}
  \begin{block}{Câu hỏi chính}
    Control Flow hỏi: \textbf{test đã đi qua nhánh nào?}\\
    Data Flow hỏi: \textbf{giá trị tạo ra ở một chỗ có được dùng đúng ở chỗ sau không?}
  \end{block}

  \vspace{0.2em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{exampleblock}{Cách đọc đơn giản}
      \begin{enumerate}
        \item Tìm nơi biến được \textbf{gán}.
        \item Tìm nơi biến được \textbf{đọc lại}.
        \item Viết assert ở nơi có thể quan sát kết quả.
      \end{enumerate}
    \end{exampleblock}

    \column{0.48\textwidth}
    \begin{alertblock}{Lỗi thường gặp}
      \begin{itemize}
        \item Gán đúng nhưng không dùng.
        \item Dùng nhầm biến cũ.
        \item Response đúng nhưng DB event lưu sai.
      \end{itemize}
    \end{alertblock}
  \end{columns}

  \vspace{0.2em}
  \centering
  \large
  \textbf{Nói ngắn gọn:} không chỉ test ``đi đúng nhánh'', mà còn test
  ``dữ liệu đi đúng đường''.
\end{frame}

\begin{frame}[shrink=7]{Ví dụ Data Flow trong \texttt{evaluate()}}
  \begin{block}{Lần theo biến \texttt{watchlistHit}}
    Biến này cho biết tài khoản nhận tiền có nằm trong watchlist không.
  \end{block}

  \vspace{0.2em}
  \begin{columns}[T]
    \column{0.32\textwidth}
    \begin{exampleblock}{1) Gán}
      Repo tìm thấy tài khoản\\
      $\Rightarrow$ \texttt{watchlistHit = true}
    \end{exampleblock}

    \column{0.32\textwidth}
    \begin{exampleblock}{2) Dùng}
      \texttt{watchlistHit} được dùng để:
      \begin{itemize}
        \item thêm signal
        \item chọn PAUSE/WARN
      \end{itemize}
    \end{exampleblock}

    \column{0.32\textwidth}
    \begin{exampleblock}{3) Kiểm tra}
      Test nên assert:
      \begin{itemize}
        \item response có signal
        \item decision đúng
        \item event DB đúng
      \end{itemize}
    \end{exampleblock}
  \end{columns}

  \vspace{0.25em}
  \begin{alertblock}{Điểm khác với Branch Coverage}
    BC chỉ hỏi ``nhánh đã chạy chưa?'' Data Flow hỏi thêm:
    ``giá trị sau khi chạy nhánh đó có đi đúng tới response và DB không?''
  \end{alertblock}
\end{frame}

\begin{lstlisting}[basicstyle=\ttfamily\tiny]

  violationRules {
    rule {
      limit {
        counter = "LINE"
        value = "COVEREDRATIO"

    }
  }

\end{lstlisting}
      Tự động đánh rớt build (FAIL) nếu code mới kéo độ phủ tụt dưới 50\%!

    \textbf{Màu sắc trên giao diện JaCoCo:}
    \begin{itemize}\setlength{\itemsep}{0.25em}
      \item \textcolor{green!60!black}{\textbf{Kim cương Xanh lá}}: Toàn bộ các nhánh rẽ tại dòng lệnh này đã được kiểm thử.
      \item \textcolor{orange!85!black}{\textbf{Kim cương Vàng}}: Nhánh mới được phủ \textbf{một phần} (ví dụ biểu thức \texttt{\&\&} mới chỉ test trường hợp False, chưa có True).
      \item \textcolor{red!70!black}{\textbf{Kim cương Đỏ}}: Nhánh hoàn toàn chưa được kiểm thử (0\% coverage).
    \end{itemize}

    \column{0.48\textwidth}
    \begin{alertblock}{Độ phủ 100\% $\neq$ Hết lỗi!}
      \begin{itemize}\setlength{\itemsep}{0.2em}
        \item \textbf{Thiếu logic}: Dev quên không viết code kiểm tra số dư âm $\to$ JaCoCo vẫn báo xanh 100\% vì dòng code đó... chưa từng tồn tại!
        \item \textbf{Lỗi đặc tả}: Code chạy đúng thiết kế nhưng thiết kế bị sai nghiệp vụ.
      \end{itemize}
    \end{alertblock}
    \begin{block}{\small Nguyên tắc vàng}
      \footnotesize \textit{Coverage là công cụ đo xem code đã được chạm tới chưa, không chứng minh được code đã xử lý hết mọi ngoại lệ.}
    \end{block}

\begin{frame}[shrink=12]{Tóm tắt buổi 3}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Lý thuyết trọng tâm:}
    \begin{itemize}\setlength{\itemsep}{0.2em}
      \item \textbf{CFG}: Mô hình hóa luồng điều khiển thành đồ thị (Node, Edge, Decision node).
      \item \textbf{Độ phủ}: Câu lệnh (SC) $\subset$ Nhánh (BC) $\subset$ Đường đi (PC).
      \item \textbf{Cyclomatic $V(G)$}: đếm decision bằng công thức $V(G)=D+1$.
      \item \textbf{Loop Testing}: Kiểm thử vòng lặp có cấu trúc tại các mốc biên ($0, 1, 2, m, n$).
      \item \textbf{Data Flow}: Lần theo biến từ nơi gán đến nơi dùng.
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Thực hành trên DigiShield:}
    \begin{itemize}\setlength{\itemsep}{0.2em}
      \item Áp dụng CFG và $V(G)$ cho phần quyết định chính của \texttt{evaluate()} ($V(G)=3$).
      \item Bộ 4 test case đạt 100\% Độ phủ nhánh (BC) khớp với test thật trong repo.
      \item Lần theo biến để kiểm tra dữ liệu đi đúng từ biến trung gian đến DB event và response.
      \item Đo độ phủ tự động bằng công cụ chuẩn công nghiệp \textbf{JaCoCo}.
      \item Hiểu rõ giới hạn: Hộp trắng cần kết hợp hộp đen để bao quát yêu cầu.
    \end{itemize}
  \end{columns}
\end{frame}
