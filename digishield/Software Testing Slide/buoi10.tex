\begin{frame}{Vì sao cần kiểm thử UI tự động?}
  \setbeamerfont{itemize/enumerate body}{size=\footnotesize}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Chỉ test bằng tay thì sao?}
      \begin{itemize}
        \item Mỗi lần release phải bấm lại \textbf{toàn bộ} luồng:
              đăng nhập 4 vai trò, báo cáo phishing, watchlist\ldots
        \item Lặp đi lặp lại $\to$ nhàm chán $\to$ \textbf{bỏ sót}
        \item Regression thủ công tốn nhiều giờ mỗi lần sửa code
        \item Không chạy được trong CI/CD ban đêm
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{UI test tự động}
      \begin{itemize}
        \item Máy ``đóng vai'' người dùng: mở browser, gõ phím, click
        \item Chạy lại hàng trăm lần, kết quả \textbf{nhất quán}
        \item Phát hiện lỗi tích hợp frontend--backend mà unit/API test bỏ lọt
        \item Là lớp \textbf{smoke test} trước khi release
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.2em}
  {\small \textbf{Ví dụ lỗi chỉ UI test bắt được}: API \texttt{POST /auth/login}
  trả 200 đúng, nhưng frontend quên điều hướng sang dashboard $\to$ API test
  xanh, người dùng vẫn ``kẹt'' ở trang login.}
\end{frame}

\begin{frame}{Đặc điểm của tầng E2E --- viết ít nhưng viết đúng chỗ}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Vì sao E2E ``đắt''?}
      \begin{itemize}
        \item Cần \textbf{cả hệ thống chạy thật}: backend + frontend + browser
        \item Chậm: mỗi test vài giây đến vài chục giây
        \item Giòn: đổi giao diện nhỏ có thể làm vỡ test
        \item Lỗi khó cô lập: fail ở đâu trong cả chuỗi?
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Nên viết E2E cho gì?}
      \begin{itemize}
        \item Các luồng \textbf{sống còn} (critical path):
              đăng nhập, báo cáo phishing, chặn giao dịch
        \item Happy path chính + vài kịch bản tiêu cực quan trọng
        \item \textbf{Không} dùng E2E để kiểm tra mọi biên
              --- việc đó của unit test (EP/BVA buổi 8)
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{Với DigiShield}
    Buổi này: luồng \textbf{đăng nhập} (cửa ngõ của mọi persona).
    Buổi 11: kịch bản E2E nhiều bước (analyst xử lý báo cáo, watchlist).
  \end{exampleblock}
\end{frame}

\section{Selenium là gì?}
%=====================================================================

%--- SLIDE 8 ----------------------------------------------------------
\begin{frame}{Selenium --- lịch sử tóm tắt}
  \begin{center}
  \footnotesize
  \begin{tabularx}{\textwidth}{c>{\raggedright\arraybackslash}X}
    \toprule
    \textbf{Năm} & \textbf{Sự kiện} \\
    \midrule
    2004 & Jason Huggins (ThoughtWorks) tạo Selenium Core --- JavaScript tiêm vào trang \\
    2006 & Simon Stewart phát triển WebDriver --- điều khiển browser ``từ bên ngoài'' \\
    2009 & Hai dự án \textbf{công bố hợp nhất} (Selenium + WebDriver) \\
    2011 & Phát hành \textbf{Selenium 2.0 (Selenium WebDriver)} \\
    2018 & Giao thức WebDriver trở thành \textbf{chuẩn W3C} chính thức \\
    2021 & \textbf{Selenium 4}: dùng thuần giao thức W3C, bỏ JSON Wire cũ \\
    2022 & Selenium 4.6: \textbf{Selenium Manager} --- tự động tải driver \\
    \bottomrule
  \end{tabularx}
  \end{center}

  {\small \textbf{Vì sao tên ``Selenium''?} Đối thủ thời đó là công cụ thương
  mại Mercury; nguyên tố selen (Selenium) là\ldots{} thuốc giải độc thủy ngân
  (Mercury).\\[0.25em]
  \textbf{Điểm cần nhớ}: môn học dùng \textbf{Selenium 4.x} + Java --- giao
  thức chuẩn W3C, không cần tải \texttt{chromedriver} thủ công.}
\end{frame}

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}{Bộ công cụ Selenium --- ba thành phần}
  \begin{center}
  \small
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{p{2.9cm}p{6.3cm}p{3.4cm}}
    \toprule
    \textbf{Thành phần} & \textbf{Vai trò} & \textbf{Dùng khi} \\
    \midrule

    \textbf{IDE} & Extension Chrome/Firefox: ghi lại thao tác (record \& playback), xuất code & Prototype nhanh, người không lập trình \\

  \end{alertblock}

  \begin{center}
  \begin{tikzpicture}[

  \begin{itemize}
    \item Mỗi lệnh (\texttt{driver.get(...)}, \texttt{click()}\ldots) là một
          \textbf{HTTP request} gửi tới driver; driver dịch sang lệnh riêng của browser.
    \item Driver do \textbf{chính hãng browser} cung cấp:
          \texttt{chromedriver} (Google), \texttt{geckodriver} (Mozilla)\ldots
    \item Nhờ chuẩn W3C, \textbf{cùng một test code} chạy được trên
          Chrome, Firefox, Edge, Safari --- chỉ đổi driver.
  \end{itemize}
\end{frame}

\section{Các bước cài đặt Selenium}
%=====================================================================

%--- SLIDE 12 ---------------------------------------------------------
\begin{frame}{Cài đặt --- yêu cầu môi trường}
  \begin{block}{Checklist trước khi bắt đầu}
    \begin{enumerate}
      \item \textbf{JDK 17+} (máy đã có --- dùng cho DigiShield backend)
      \item \textbf{Google Chrome} bản mới (Selenium Manager sẽ tự khớp driver)
      \item \textbf{IDE}: IntelliJ IDEA / VS Code / Eclipse
      \item \textbf{Maven hoặc Gradle} --- buổi này trình bày cả hai
      \item Mã nguồn DigiShield (backend + frontend) chạy được ở local
    \end{enumerate}
  \end{block}

  \begin{alertblock}{Không cần}
    \begin{itemize}
      \item \textbf{Không} cần tải \texttt{chromedriver.exe} thủ công (Selenium $\geq$ 4.6)
      \item \textbf{Không} cần Docker cho buổi này (backend chạy profile dev với H2)

%--- SLIDE 25 ---------------------------------------------------------
\begin{frame}[fragile]{\texttt{WebElement} --- \texttt{findElement} và \texttt{findElements}}
  \begin{block}{Khái niệm}
    \textbf{Locator} = ``địa chỉ'' của một phần tử trong DOM.
    \texttt{findElement(By...)} trả về \texttt{WebElement} đầu tiên khớp;
    không thấy thì ném \texttt{NoSuchElementException} (test fail).
  \end{block}

\begin{lstlisting}[language=java]
// Mot element
WebElement email =
    driver.findElement(By.id("login-email"));

// Danh sach element - khong nem exception,
// tra ve list rong neu khong co gi khop
List<WebElement> buttons =
    driver.findElements(By.tagName("button"));
// Trang login co 4 nut persona + submit + SSO
\end{lstlisting}

  \begin{itemize}
    \item \texttt{findElements} + kiểm tra \texttt{isEmpty()} là cách
          kiểm tra ``element có tồn tại không'' mà không làm test nổ
  \end{itemize}
\end{frame}

\begin{frame}{Họ locator \texttt{By} --- tổng quan}

\begin{frame}{XPath --- cú pháp cần nhớ}
  \begin{center}
  \small
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{p{7.7cm}p{5.6cm}}
    \toprule
    \textbf{Cú pháp} & \textbf{Ý nghĩa} \\
    \midrule
    \texttt{//button} & Mọi \texttt{button} ở bất kỳ đâu \\
    \texttt{//input[@id='login-pw']} & \texttt{input} có thuộc tính id \\
    \texttt{//button[text()='Admin']} & Text \textbf{bằng đúng} ``Admin'' \\

    \texttt{//label[@for='login-pw']/..} & Đi \textbf{ngược lên} cha (CSS không làm được) \\

    \bottomrule
  \end{tabular}
  \end{center}

  \begin{block}{Khi nào cần XPath?}
    Khi phải tìm theo \textbf{text hiển thị} hoặc đi \textbf{ngược cây DOM} ---
    hai việc CSS selector không hỗ trợ. Còn lại, ưu tiên CSS (ngắn gọn, nhanh hơn).
  \end{block}
