% SLIDE 6 - Waterfall
%---------------------------------------------------------------------
\begin{frame}{Mô hình Waterfall (thác nước)}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Đặc điểm}
    \begin{itemize}
      \item Tuyến tính: mỗi giai đoạn hoàn tất mới sang giai đoạn kế
      \item Tài liệu hóa nặng, ký duyệt từng pha
      \item Kiểm thử là \textbf{một pha riêng, sau lập trình}
    \end{itemize}
    \textbf{Phù hợp khi}
    \begin{itemize}
      \item Yêu cầu ổn định, ít thay đổi
      \item Dự án có ràng buộc hợp đồng/pháp lý chặt
    \end{itemize}

    \column{0.5\textwidth}
    \begin{alertblock}{Hệ quả với kiểm thử}
      \begin{itemize}
        \item Lỗi yêu cầu/thiết kế bị phát hiện \textbf{rất muộn} --- chi phí sửa cao (quy tắc 1:10:100, Buổi 1)
        \item Kiểm thử bị \textbf{dồn ép} cuối dự án: trễ tiến độ thì cắt thời gian test
        \item Tester tham gia muộn, ít hiểu nghiệp vụ
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

  \end{tikzpicture}
  \end{center}

  \begin{itemize}
    \item Hệ thống được xây theo \textbf{từng phần tăng trưởng} (increment);
          mỗi vòng lặp là một ``Waterfall thu nhỏ'' \textbf{có kiểm thử riêng}.
    \item Kiểm thử diễn ra \textbf{nhiều lần, sớm hơn} so với Waterfall
          $\to$ phản hồi nhanh về chất lượng.
  \end{itemize}

  \begin{alertblock}{Hệ quả quan trọng}
    Chức năng của vòng lặp trước phải được \textbf{kiểm thử lại} ở vòng lặp sau
    (regression) --- vì code mới có thể phá vỡ code cũ. Regression tăng dần
    $\Rightarrow$ bắt buộc \textbf{tự động hóa}.
  \end{alertblock}
\end{frame}

    \end{itemize}

    \end{exampleblock}
  \end{columns}

\begin{frame}{Vị trí hoạt động kiểm thử trong từng mô hình}
  \begingroup\centering
  \small
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{p{2.2cm}p{4.6cm}p{4.9cm}}
    \toprule
    \textbf{Mô hình} & \textbf{Kiểm thử diễn ra khi nào?} & \textbf{Rủi ro / lưu ý} \\
    \midrule
    Waterfall & Một pha riêng, sau lập trình & Lỗi phát hiện muộn, test bị cắt xén khi trễ hạn \\
    \midrule
    V-model & \textit{Thiết kế} test từ đầu mỗi pha; \textit{thực thi} ở nhánh phải & Vẫn cứng nhắc trước thay đổi yêu cầu \\
    \midrule
    Iterative & Cuối mỗi vòng lặp & Khối lượng regression tăng dần theo vòng lặp \\
    \midrule
    Agile/Scrum & Liên tục trong Sprint, mọi story & Đòi hỏi tự động hóa và kỹ năng kỹ thuật của tester \\
    \bottomrule
  \end{tabular}\par\endgroup

  \vspace{0.3em}
  \begin{block}{Entry / Exit criteria \cite{istqb}}
    Mỗi giai đoạn có \textbf{Entry criteria} (điều kiện để bắt đầu) và
    \textbf{Exit criteria} (điều kiện để kết thúc). Chúng là ``cổng chất lượng''
    ngăn việc nhảy cóc --- ví dụ: không thực thi test khi môi trường chưa sẵn sàng.
  \end{block}
  \small Giai đoạn 3 và 4 có thể tiến hành \textit{song song}.
\end{frame}

%---------------------------------------------------------------------
% SLIDE 15 - STLC phase 1
%---------------------------------------------------------------------
\begin{frame}{Giai đoạn 1: Phân tích yêu cầu kiểm thử}
  \vspace*{-0.2em}
  \textbf{Hoạt động}: đọc đặc tả/user story, xác định \textit{cái gì} cần kiểm thử
  (testable requirements), đặt câu hỏi làm rõ, nhận diện yêu cầu phi chức năng.

  \vspace{0.2em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Entry criteria}
      \begin{itemize}
        \item Có tài liệu yêu cầu / user story
        \item Tiếp cận được BA/PO để hỏi đáp
        \item Có tài liệu kiến trúc (nếu đã có)
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Exit criteria}
      \begin{itemize}
        \item Danh sách \textbf{testable requirements}; RTM khởi tạo
        \item Đánh giá khả năng tự động hóa
        \item Các câu hỏi mở đã được giải đáp
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.1em}
  \begin{exampleblock}{\small DigiShield}
    \small Đọc yêu cầu ``can thiệp giao dịch đáng ngờ'': khi nào PAUSE, WARN,
    ALLOW? Nguồn watchlist là gì? $\to$ danh sách hạng mục cần kiểm thử
    cho module \texttt{interception}.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 16 - STLC phase 2
%---------------------------------------------------------------------
\begin{frame}{Giai đoạn 2: Lập kế hoạch kiểm thử}
  \textbf{Hoạt động}: viết Test Plan --- xác định phạm vi, cách tiếp cận, ước lượng
  công sức, phân công vai trò, chọn công cụ, phân tích rủi ro, lập lịch.

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Entry criteria}
      \begin{itemize}
        \item Tài liệu phân tích yêu cầu kiểm thử (GĐ 1)
        \item RTM khởi tạo
        \item Thông tin nguồn lực, ngân sách
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Exit criteria}
      \begin{itemize}
        \item Test Plan được duyệt/ký
        \item Ước lượng công sức được chốt
        \item Vai trò, trách nhiệm đã phân công
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{\small DigiShield}
    \small Test Plan cho module \texttt{interception} (mẫu rút gọn ở phần sau):
    phạm vi = \texttt{evaluate} + watchlist; công cụ = JUnit/Mockito, Postman;
    tiêu chí pass/fail; lịch theo tuần.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 17 - STLC phase 3
%---------------------------------------------------------------------
\begin{frame}{Giai đoạn 3: Thiết kế test case}
  \textbf{Hoạt động}: viết test case chi tiết (ID, bước, dữ liệu, kết quả mong đợi),
  chuẩn bị test data, cập nhật RTM (yêu cầu $\leftrightarrow$ test case),
  review chéo test case.

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Entry criteria}
      \begin{itemize}
        \item Test Plan đã duyệt
        \item Yêu cầu đủ chi tiết để thiết kế
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Exit criteria}
      \begin{itemize}
        \item Test case + test data được review, ký duyệt
        \item RTM phủ 100\% yêu cầu trong phạm vi
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{\small DigiShield}
    \small Từ quy tắc \texttt{onCall \&\& newPayee \&\& watchlistHit $\to$ PAUSE}
    thiết kế đủ test case phủ các nhánh PAUSE/WARN/ALLOW --- kỹ thuật cụ thể
    (bảng quyết định, độ phủ nhánh) học ở Buổi 3--4 và 8.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 18 - STLC phase 4
%---------------------------------------------------------------------
\begin{frame}[fragile]{Giai đoạn 4: Chuẩn bị môi trường kiểm thử}
  \textbf{Hoạt động}: dựng môi trường (server, DB, dữ liệu demo), cài công cụ,
  smoke test xác nhận môi trường dùng được.

  \vspace{0.3em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Entry criteria}
      \small Có thiết kế môi trường; có bản build cài đặt được; test data sẵn sàng.
    \end{block}
    \begin{block}{Exit criteria}
      \small Môi trường hoạt động, smoke test pass; test data đã nạp.
    \end{block}
    \column{0.50\textwidth}
    \begin{exampleblock}{\small DigiShield: 2 môi trường}
\begin{lstlisting}[language=bash,basicstyle=\ttfamily\scriptsize]
# Moi truong dev (H2, khong can Docker)
./gradlew :boot:app:bootRun \
  --args='--spring.profiles.active=dev'

# Moi truong prod-like (Docker)
docker compose -f deploy/compose/\
docker-compose.prodlike.yml up --build
\end{lstlisting}
    \end{exampleblock}
  \end{columns}

  \vspace{0.2em}
  \small Smoke test: mở \texttt{http://localhost:8080/api/v1} --- dữ liệu demo
  (6 user seed, vd. \texttt{admin@coquan.gov.vn}) tự nạp bởi \texttt{DevDataSeeder}.
\end{frame}

%---------------------------------------------------------------------
% SLIDE 19 - STLC phase 5
%---------------------------------------------------------------------
\begin{frame}{Giai đoạn 5: Thực thi kiểm thử}
  \textbf{Hoạt động}: chạy test case theo kế hoạch, ghi kết quả (pass/fail/blocked),
  báo defect kèm bằng chứng, re-test sau khi sửa, chạy regression.

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Entry criteria}
      \begin{itemize}
        \item Test case + test data sẵn sàng
        \item Môi trường đã smoke test pass
        \item Bản build cần test đã triển khai
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Exit criteria}
      \begin{itemize}
        \item Toàn bộ test case theo kế hoạch đã thực thi
        \item Defect đã được log và theo dõi
        \item Kết quả được ghi vào RTM
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{\small DigiShield}
    \small Tự động: \texttt{./gradlew test} rồi \texttt{./gradlew integrationTest};
    thủ công/API: chạy collection Postman trên \texttt{http://localhost:8080/api/v1};
    lỗi tìm được ghi thành bug report (mẫu ở phần sau).
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 20 - STLC phase 6
%---------------------------------------------------------------------
\begin{frame}{Giai đoạn 6: Đóng chu trình kiểm thử}
  \textbf{Hoạt động}: viết Test Summary Report, đánh giá theo exit criteria của cả
  chu trình, họp rút kinh nghiệm (lessons learned), lưu trữ testware để tái sử dụng.

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Entry criteria}
      \begin{itemize}
        \item Thực thi đã hoàn tất
        \item Kết quả test và danh sách defect đầy đủ
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Exit criteria}
      \begin{itemize}
        \item Test Summary Report được duyệt
        \item Defect còn mở có phương án xử lý
        \item Testware đã bàn giao/lưu trữ
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{block}{Nội dung Test Summary Report}
    \small Số test case thực thi, tỷ lệ pass/fail, độ phủ đạt được (JaCoCo),
    defect theo mức nghiêm trọng, rủi ro còn lại, \textbf{khuyến nghị release hay không}.
  \end{block}
\end{frame}

@ExtendWith(MockitoExtension.class)
class InterceptionServiceImplTest {
  @Mock AccountWatchEntryRepository watchRepo;
  @Mock InterventionEventRepository eventRepo;
  @InjectMocks InterceptionServiceImpl svc;

  @Test
  void evaluate_allSignals_pauses() {
    // Arrange: dich den nam trong watchlist
    when(watchRepo.findByTenantIdAndValue(
        TENANT_ID, DEST))
      .thenReturn(Optional.of(watchHit));
    // Act
    var d = svc.evaluate(
      new EvaluateRequest(userId, amount,
        DEST, true, true));
    // Assert: nhanh PAUSE duoc chon
    assertThat(d.decision())
      .isEqualTo("PAUSE");
  }
}
\end{lstlisting}

\begin{frame}{Kiểm thử chức năng và phi chức năng}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Chức năng (functional) --- ``làm gì''\vphantom{ỹể}}
      \small Kiểm tra hệ thống làm \textbf{đúng việc} được đặc tả.
      \begin{itemize}
        \item \small Evaluate trả về PAUSE khi đủ 3 tín hiệu rủi ro
        \item \small Triage xác nhận $\to$ trạng thái CONFIRMED
        \item \small Login sai mật khẩu $\to$ từ chối
      \end{itemize}
    \end{block}
    \column{0.5\textwidth}
    \begin{block}{Phi chức năng (non-functional) --- ``tốt thế nào''\vphantom{ỹể}}
      \small Kiểm tra \textbf{chất lượng} khi thực hiện việc đó.
      \begin{itemize}
        \item \small Hiệu năng: evaluate phản hồi $<$ 200ms
        \item \small Bảo mật: LEARNER gọi API của ANALYST bị 403; cách ly dữ liệu giữa các tenant
        \item \small Usability: form báo cáo dễ dùng trên di động
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  {\small \textbf{Ngoài ra}: kiểm thử \textbf{cấu trúc} (white-box --- độ phủ
  lệnh/nhánh, Buổi 3--4) và kiểm thử \textbf{liên quan thay đổi}
  (re-test, regression --- slide sau).}
\end{frame}

% SLIDE 29 - retest vs regression
%---------------------------------------------------------------------
\begin{frame}{Re-test (kiểm thử lại) và Regression (hồi quy)}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Re-test (confirmation testing) \cite{istqb}}
      Chạy lại \textbf{đúng test case đã fail} sau khi lỗi được sửa, để xác nhận
      lỗi \textbf{đã hết}.
      \begin{itemize}
        \item \small Cùng môi trường, cùng dữ liệu, cùng bước
        \item \small Bug chuyển từ Fixed $\to$ Verified
      \end{itemize}
    \end{block}
    \column{0.5\textwidth}
    \begin{block}{Regression testing}
      Chạy lại \textbf{các test case đã pass} quanh vùng thay đổi, để xác nhận thay đổi
      \textbf{không phá} chức năng cũ.
      \begin{itemize}
        \item \small Tập test tích lũy theo thời gian
        \item \small Bắt buộc tự động hóa, gắn vào CI
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{Ví dụ DigiShield}
    \small Dev sửa lỗi trong \texttt{evaluate()} khiến WARN trả về sai.
    \textbf{Re-test}: chạy lại test WARN đã fail. \textbf{Regression}: chạy lại
    \textbf{toàn bộ} \texttt{./gradlew test} (176 test) --- đảm bảo các nhánh
    PAUSE/ALLOW và 8 module còn lại không bị ảnh hưởng.
  \end{exampleblock}
\end{frame}

    \node[st,right=0.75cm of ver] (clo) {Closed};

    \draw[arr] (new) -- node[above,font=\scriptsize]{lead duyệt} (asg);
    \draw[arr] (asg) -- node[above,font=\scriptsize]{dev sửa} (fix);
    \draw[arr] (fix) -- node[above,font=\scriptsize]{tester re-test} (ver);
    \draw[arr] (ver) -- (clo);
    \draw[arr,dashed] (fix.south) to[out=-40,in=-140]
      node[below,font=\scriptsize]{re-test fail $\to$ Reopened} (asg.south);
    \node[st,below=0.9cm of new,fill=white,densely dotted] (rej) {Rejected /\\Duplicate};
    \draw[arr,dashed] (new.south) -- node[left,font=\scriptsize]{không phải lỗi} (rej.north);
  \end{tikzpicture}
  \end{center}

  \vspace{0.3em}
  \begin{itemize}
    \item \small \textbf{Severity} (mức nghiêm trọng với hệ thống) $\ne$ \textbf{Priority}
          (mức khẩn với kế hoạch) --- lỗi chính tả trên trang chủ: severity thấp, priority cao.
    \item \small Re-test chuyển Fixed $\to$ Verified; regression đảm bảo bản sửa không phá chỗ khác.
  \end{itemize}
\end{frame}
