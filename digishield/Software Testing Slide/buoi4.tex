\section{Độ phủ điều kiện và MC/DC}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 6
%---------------------------------------------------------------------
\begin{frame}{Decision coverage vs Condition coverage}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Độ phủ quyết định (decision)}
      Mỗi \textbf{quyết định} (toàn bộ biểu thức boolean trong
      \texttt{if}) nhận giá trị \textbf{true} và \textbf{false}
      ít nhất một lần.
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{Độ phủ điều kiện (condition)}
      Mỗi \textbf{điều kiện con} (atomic condition) trong quyết định
      nhận giá trị \textbf{true} và \textbf{false} ít nhất một lần.
    \end{block}
  \end{columns}

  \vspace{0.4em}
  \begin{exampleblock}{Ví dụ DigiShield}
    Quyết định: \texttt{onCall \&\& newPayee \&\& watchlistHit}\\
    Ba điều kiện con: \texttt{onCall}, \texttt{newPayee},
    \texttt{watchlistHit}.
  \end{exampleblock}

  \begin{alertblock}{Lưu ý}
    Condition coverage \textbf{không} bao hàm decision coverage
    và ngược lại --- xem phản ví dụ ở slide sau.
  \end{alertblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 7
%---------------------------------------------------------------------
\begin{frame}{Phản ví dụ: condition 100\% nhưng decision 50\% (1/2)}
  Ký hiệu: $A$ = \texttt{onCall}, $B$ = \texttt{newPayee},
  $C$ = \texttt{watchlistHit}; quyết định $D = A \land B \land C$.

  \vspace{0.3em}
  \begin{center}
  \begin{tabular}{lccccc}
    \toprule
    \textbf{Test} & $A$ & $B$ & $C$ & $D$ & \textbf{Kết quả} \\
    \midrule
    T1 & T & F & T & F & không PAUSE \\
    T2 & F & T & F & F & không PAUSE \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \begin{itemize}
    \item Mỗi điều kiện con đều đã nhận cả T và F
          $\Rightarrow$ \textbf{condition coverage = 100\%}.
    \item Nhưng $D$ luôn bằng F $\Rightarrow$ nhánh PAUSE
          \textbf{chưa bao giờ chạy}: decision coverage = 50\%.
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 7b
%---------------------------------------------------------------------
\begin{frame}{Phản ví dụ: condition 100\% nhưng decision 50\% (2/2)}
  \begin{alertblock}{Ghi chú Java: short-circuit}
    Toán tử \texttt{\&\&} là \textit{short-circuit}: với T2, $B$ và $C$
    không được đánh giá; với T1, $C$ cũng không (vì $A \land B$ đã F).
    Ở mức \textbf{thực thi} --- mức JaCoCo đo --- $C$ chưa bao giờ được
    đánh giá; ``100\%'' ở trên chỉ đúng theo bảng giá trị
    \textbf{lý thuyết}. JaCoCo đo ở bytecode nên ``nhìn thấy'' từng
    điều kiện con \textit{đã được đánh giá} --- và sẽ báo $C$ chưa phủ.
  \end{alertblock}
\end{frame}

\begin{frame}{Multiple condition coverage --- bảng chân trị $2^n$}
  Phủ \textbf{mọi tổ hợp} giá trị của $n$ điều kiện con: $2^3 = 8$ test
  cho \texttt{evaluate()}.

  \begin{center}
  \small
  \begin{tabular}{cccccl}
    \toprule
    \# & \texttt{onCall} & \texttt{newPayee} & \texttt{watchlistHit}
       & $A \land B \land C$ & \textbf{Quyết định} \\
    \midrule
    1 & T & T & T & T & PAUSE \\
    2 & T & T & F & F & ALLOW \\
    3 & T & F & T & F & WARN \\
    4 & T & F & F & F & ALLOW \\
    5 & F & T & T & F & WARN \\
    6 & F & T & F & F & ALLOW \\
    7 & F & F & T & F & WARN \\
    8 & F & F & F & F & ALLOW \\
    \bottomrule
  \end{tabular}
  \end{center}

  \begin{itemize}
    \item Đầy đủ nhất nhưng \textbf{bùng nổ tổ hợp}: $n = 10$ điều kiện
          $\Rightarrow$ 1024 test. Cần tiêu chí ``vừa đủ mạnh'': MC/DC.
  \end{itemize}
\end{frame}

% SLIDE 9
%---------------------------------------------------------------------
\begin{frame}{MC/DC --- Modified Condition / Decision Coverage}
  \begin{block}{Định nghĩa \cite{ammann,istqb}}
    Bộ test đạt MC/DC khi:
    \begin{enumerate}
      \item Mỗi quyết định nhận cả T và F;
      \item Mỗi điều kiện con nhận cả T và F;
      \item Mỗi điều kiện con được chứng minh là \textbf{ảnh hưởng
            độc lập} đến kết quả quyết định: tồn tại \textbf{cặp test}
            chỉ khác nhau ở điều kiện đó mà kết quả quyết định đổi.
    \end{enumerate}
  \end{block}

  \vspace{0.3em}
  \begin{exampleblock}{Chi phí}
    Với $n$ điều kiện con chỉ cần tối thiểu $n + 1$ test
    (thay vì $2^n$): với $n=3$ là \textbf{4 test} thay vì 8.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 10
%---------------------------------------------------------------------
\begin{frame}{Vì sao ngành hàng không (aviation) chọn MC/DC?}
  \begin{itemize}
    \item Chuẩn \textbf{DO-178C} \cite{do178c} (phần mềm điều khiển bay)
          \textbf{bắt buộc} MC/DC cho phần mềm mức A (Level A) --- lỗi có thể
          gây tai nạn thảm khốc.
    \item Lý do chọn MC/DC:
    \begin{itemize}
      \item Mạnh hơn hẳn decision/condition coverage: phát hiện lỗi
            ``dây điện đấu nhầm'' trong biểu thức logic
            (\texttt{\&\&} viết nhầm thành \texttt{||}, thiếu một điều kiện).
      \item Chi phí tuyến tính $n+1$, khả thi với biểu thức lớn.
    \end{itemize}
    \item Ứng dụng khác: ô tô (ISO 26262), y tế, đường sắt.
  \end{itemize}

  \begin{alertblock}{Liên hệ DigiShield}
    \texttt{evaluate()} quyết định \textbf{chặn hay cho phép một giao dịch
    chuyển tiền} --- sai nhánh nghĩa là hoặc chặn nhầm khách hàng,
    hoặc bỏ lọt giao dịch lừa đảo. Logic ``rẻ mà quan trọng'' như vậy
    rất đáng đầu tư MC/DC.
  \end{alertblock}
\end{frame}

\begin{frame}{MC/DC cho \texttt{onCall \&\& newPayee \&\& watchlistHit}}
  Tìm cặp dòng (trong bảng chân trị 8 dòng) chứng minh ảnh hưởng
  độc lập của từng điều kiện --- hai dòng chỉ khác nhau đúng một cột
  và kết quả $D$ đổi:

  \vspace{0.3em}
  \begin{center}
  \small
  \begin{tabular}{llll}
    \toprule
    \textbf{Điều kiện} & \textbf{Cặp dòng} & \textbf{Giá trị} & \textbf{$D$ đổi} \\

  \end{tabular}
  \end{center}

  \begin{exampleblock}{Bộ MC/DC tối thiểu}
    Hợp của các cặp: $\{\#1, \#2, \#3, \#5\}$ --- đúng $n+1 = 4$ test.
    Dòng \#1 được ``tái sử dụng'' trong cả 3 cặp.
  \end{exampleblock}
\end{frame}

\begin{frame}{So sánh các tiêu chí độ phủ điều kiện (1/2)}
  Áp dụng cho \texttt{evaluate()} --- chuỗi \texttt{if / else if / else}
  chứa quyết định 3 điều kiện con:

  \vspace{0.3em}
  \begin{center}
  \small
  \begin{tabular}{lcl}
    \toprule
    \textbf{Tiêu chí} & \textbf{Số test tối thiểu} & \textbf{Phát hiện} \\
    \midrule
    Statement coverage           & 3\(^*\) & dòng chưa chạy \\
    Decision (branch) coverage   & 3\(^*\) & nhánh chưa chạy \\
    Condition coverage           & 2\(^\dagger\) & điều kiện con chưa lật \\
    Condition/Decision           & 2\(^\dagger\) & cả hai mức trên \\
    \textbf{MC/DC}               & \textbf{4}\(^\dagger\) & vai trò từng điều kiện \\
    Multiple condition           & 8\(^\dagger\) & mọi tổ hợp \\
    \bottomrule
  \end{tabular}
  \end{center}
  {\scriptsize \(^*\) Tính trên \textbf{toàn chuỗi} \texttt{if / else if
  / else}: 3 lệnh gán / 3 lối ra PAUSE / WARN / ALLOW loại trừ nhau.\\
  \(^\dagger\) Tính \textbf{riêng trên quyết định phức}
  \texttt{onCall \&\& newPayee \&\& watchlistHit}; nếu tính trên toàn
  chuỗi thì Condition/Decision cũng cần $\geq 3$ test (vì bao hàm
  decision coverage).}
\end{frame}

\begin{frame}{Vì sao cần kiểm thử luồng dữ liệu?}
  \begin{alertblock}{Lỗ hổng của độ phủ luồng điều khiển}
    Branch coverage chỉ hỏi ``nhánh này \textbf{đã chạy} chưa?'' ---
    không hỏi ``giá trị được gán ở đây có \textbf{đến đúng nơi
    sử dụng} và \textbf{đúng nội dung} không?''
  \end{alertblock}

  \vspace{0.3em}
  \begin{exampleblock}{Ví dụ lỗi mà branch coverage bỏ lọt (DigiShield)}
    Trong \texttt{evaluate()}, giả sử lập trình viên quên
    \texttt{signals.add("WATCHLIST\_HIT")}: cả 3 nhánh
    PAUSE/WARN/ALLOW vẫn chạy đủ, test nhánh vẫn xanh ---
    nhưng event lưu xuống DB \textbf{thiếu tín hiệu}, đội SOC
    mất dấu vết điều tra.
  \end{exampleblock}

  \begin{block}{Ý tưởng Data Flow Testing}
    Theo dấu \textbf{từng biến}: từ nơi được gán (define)
    đến mọi nơi được dùng (use), và bắt buộc test đi qua các
    ``đường nối'' đó.
  \end{block}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 21
%---------------------------------------------------------------------
\begin{frame}{def, c-use, p-use}
  \begin{columns}[T]
    \column{0.31\textwidth}
    \begin{block}{def (define)}
      Biến được \textbf{gán giá trị}: khai báo kèm gán, phép gán,
      tham số vào hàm, đọc từ input.
    \end{block}

    \column{0.31\textwidth}
    \begin{block}{c-use}
      \textit{computation use} --- dùng trong \textbf{tính toán},
      vế phải phép gán, đối số lời gọi hàm, biểu thức return.
    \end{block}

    \column{0.31\textwidth}
    \begin{block}{p-use}
      \textit{predicate use} --- dùng trong \textbf{điều kiện
      rẽ nhánh} (if, while, toán tử ba ngôi).
    \end{block}
  \end{columns}

  \vspace{0.4em}
  \begin{exampleblock}{Nhận diện nhanh trên \texttt{evaluate()}}
    \begin{itemize}
      \item \texttt{watchlistHit = hit.isPresent()} --- def của
            \texttt{watchlistHit}, c-use của \texttt{hit}.
      \item \texttt{if (watchlistHit)} --- p-use của \texttt{watchlistHit}.
      \item \texttt{String.join(",", signals)} --- c-use của \texttt{signals}.
    \end{itemize}
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 22
%---------------------------------------------------------------------
\begin{frame}{DU-pair và DU-path}
  \begin{block}{DU-pair (cặp def--use)}
    Cặp $(d, u)$: vị trí $d$ định nghĩa biến $x$ và vị trí $u$ sử dụng
    $x$, sao cho tồn tại đường đi từ $d$ đến $u$ \textbf{không có
    định nghĩa lại} $x$ ở giữa (đường \textit{definition-clear}).
  \end{block}

  \vspace{0.3em}
  \begin{block}{DU-path}
    Một đường đi definition-clear \textbf{cụ thể} trong CFG
    từ $d$ đến $u$. Một DU-pair có thể có nhiều DU-path
    (đi qua các nhánh khác nhau).
  \end{block}

  \vspace{0.3em}
  \begin{itemize}
    \item Nếu giữa $d$ và $u$ có một phép gán khác cho $x$ ---
          def cũ bị \textbf{kill}, cặp $(d,u)$ không còn hiệu lực.
    \item Test một DU-pair = tạo dữ liệu vào sao cho chương trình
          \textbf{đi qua} $d$ rồi tới $u$, và \textbf{assert} giá trị tại $u$.
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------

%---------------------------------------------------------------------
\begin{frame}{Ba loại anomaly luồng dữ liệu}
  Anomaly = chuỗi def/use \textbf{bất thường}, thường là triệu chứng
  của bug hoặc code thừa:

  \vspace{0.3em}
  \begin{center}
  \footnotesize
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{>{\raggedright\arraybackslash}p{3.1cm}>{\raggedright\arraybackslash}p{4.6cm}>{\raggedright\arraybackslash}p{4.0cm}}
    \toprule
    \textbf{Loại} & \textbf{Mẫu} & \textbf{Triệu chứng của} \\
    \midrule
    define--define (dd) & gán rồi gán đè, chưa hề dùng & quên dùng giá trị đầu; sai thứ tự lệnh \\
    define--kill (dk)   & gán rồi ra khỏi scope, không dùng & biến thừa; logic bị xóa dở \\
    use-before-define (ud) & dùng khi chưa chắc đã gán & thiếu nhánh khởi tạo; NPE \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \begin{itemize}
    \item Anomaly phát hiện được bằng \textbf{phân tích tĩnh} ---
          không cần chạy chương trình.
    \item Java: compiler chặn ``dùng biến local chưa gán'', nhưng
          \textbf{không} chặn field/đối tượng \texttt{null} --- vẫn cần soi.
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 32
%---------------------------------------------------------------------
\begin{frame}[fragile]{Ví dụ anomaly và công cụ phát hiện}
  \setbeamerfont{itemize/enumerate body}{size=\small}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// 1. dd: gan de len khi chua dung -> mat gia tri
int score = computeScore(tenantId, userId);
score = 0;                    // bug? hay co y?

// 2. dk: dinh nghia roi khong ai dung
int unused = scores.size();   // code chet

// 3. ud: dung khi co the chua gan

if (highRisk) { risk = 90; }
return risk + 5;   // NPE khi highRisk = false
\end{lstlisting}

  \begin{itemize}
    \item IDE (IntelliJ) cảnh báo ngay: \textit{unused assignment},
          \textit{variable is never used}, \textit{unboxing may produce NPE}.
    \item CI của DigiShield chạy \textbf{Checkstyle} (buildSrc convention);
          có thể bổ sung SpotBugs / ErrorProne để bắt anomaly sâu hơn.
    \item Bài học: đọc warning của IDE = data flow analysis miễn phí.
  \end{itemize}
\end{frame}

\section{Giới hạn của độ phủ}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 42
%---------------------------------------------------------------------
\begin{frame}{100\% coverage $\neq$ hết lỗi}
  \begin{block}{Nguyên tắc ISTQB (buổi 1) \cite{istqb}}
    \#1 --- Kiểm thử chỉ chứng minh \textbf{sự hiện diện} của lỗi,
    không chứng minh sự vắng mặt. \#2 --- Kiểm thử vét cạn là
    \textbf{bất khả thi}. Coverage 100\% vẫn chỉ là một mẫu rất nhỏ
    của không gian input.
  \end{block}

  \vspace{0.3em}
  \textbf{Ba kiểu lỗi ``miễn nhiễm'' với coverage --- ví dụ DigiShield:}
  \begin{itemize}
    \item \small \textbf{Lỗi giá trị}: \texttt{computeScore()} coverage
          100\% với 1 test, nhưng nếu ai đổi \texttt{MAX\_RISK} thành 1000
          thì clamp sai mà test dòng lệnh vẫn xanh.
    \item \small \textbf{Lỗi thiếu code}: quên
          \texttt{signals.add("WATCHLIST\_HIT")} --- code không tồn tại
          thì không có dòng nào để ``chưa phủ''.
    \item \small \textbf{Test không assert}: gọi \texttt{evaluate()}
          mà không \texttt{assertThat} --- coverage tăng, giá trị bằng 0.
  \end{itemize}
\end{frame}
