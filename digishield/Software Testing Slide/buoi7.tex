%---------------------------------------------------------------------
\begin{frame}{Unit test với mock \emph{chưa} kiểm chứng được gì?}
  \begin{alertblock}{Mock trả lời đúng vì\ldots ta bảo nó trả lời như vậy}
    Mọi tương tác với hạ tầng đều bị thay bằng ``diễn viên đóng thế''.
  \end{alertblock}
  \begin{itemize}
    \item \textbf{SQL / JPA mapping}: câu query sinh ra có đúng không? Cột
      \texttt{tenant\_id} kiểu \texttt{uuid} có map được sang \texttt{UUID}?
    \item \textbf{Cấu hình database}: policy Row-Level Security của PostgreSQL
      có thực sự chặn tenant khác đọc dữ liệu? (mock không biết RLS là gì!)
    \item \textbf{Security}: \texttt{@PreAuthorize("hasRole('ANALYST')")} có
      chạy? Người chưa đăng nhập có bị chặn 401?
    \item \textbf{Wiring}: các bean có được Spring lắp ráp đúng? JSON
      serialize/deserialize qua HTTP có khớp DTO?
    \item \textbf{Sự kiện bất đồng bộ}: listener module khác có nhận được event?
  \end{itemize}
  \begin{exampleblock}{Kết luận}
    Cần một tầng test \textbf{tích hợp}: chạy nhiều thành phần thật cùng nhau.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 11 - @SpringBootTest là gì
%---------------------------------------------------------------------
\begin{frame}{\texttt{@SpringBootTest} --- khởi động ứng dụng thật để test}
  \begin{block}{Ý tưởng}
    Annotation của Spring Boot: khởi động \textbf{toàn bộ ApplicationContext}
    (bean, cấu hình, filter, listener\ldots) y như khi chạy thật, rồi cho phép
    \texttt{@Autowired} bất kỳ bean nào vào test.
  \end{block}
  \begin{itemize}
    \item Test chạy \emph{bên trong} JVM của ứng dụng --- khác Postman
      (buổi 9) gọi từ bên ngoài.
    \item Có thể ghi đè cấu hình ngay trên annotation:
      \texttt{@SpringBootTest(properties = \{\ldots\})}.
    \item Kết hợp \texttt{@MockitoBean} / \texttt{@TestConfiguration} để thay
      \emph{một vài} bean khó dựng (vd \texttt{JwtDecoder} gọi IdP ngoài).
  \end{itemize}
  \begin{alertblock}{Trade-off}
    Độ tin cậy cao nhất trong JVM --- nhưng khởi động context tốn
    \textbf{vài giây}; dùng có chọn lọc.
  \end{alertblock}
\end{frame}

% SLIDE 12 - webEnvironment

\begin{frame}[fragile]{\texttt{webEnvironment}: MOCK và RANDOM\_PORT}
  \begin{block}{MOCK (mặc định) --- không mở port thật}
\begin{lstlisting}[language=Java,style=code,basicstyle=\ttfamily\scriptsize,aboveskip=0pt]
@SpringBootTest              // webEnvironment = MOCK
@AutoConfigureMockMvc
class InterventionApiIT {
    @Autowired MockMvc mockMvc;
    // Goi qua MockMvc, khong co HTTP that
}
\end{lstlisting}
  \end{block}
  \begin{block}{RANDOM\_PORT --- server thật, port ngẫu nhiên}
\begin{lstlisting}[language=Java,style=code,basicstyle=\ttfamily\scriptsize,aboveskip=0pt]
@SpringBootTest(webEnvironment =
    SpringBootTest.WebEnvironment.RANDOM_PORT)
class InterventionHttpIT {
    @Autowired TestRestTemplate rest;
    @LocalServerPort int port;
    // Goi HTTP that toi localhost:port
}
\end{lstlisting}
  \end{block}
  \small \texttt{DEFINED\_PORT} (dùng port khai báo trong properties, mặc định
  8080): chỉ khi công cụ ngoài bắt buộc;
  dễ xung đột port trong CI.
\end{frame}

% SLIDE 13 - chi phí context + caching
%---------------------------------------------------------------------
\begin{frame}{Chi phí khởi động context và context caching}
  \begin{columns}[t]
    \column{0.52\textwidth}
    \textbf{Chi phí một lần khởi động:}
    \begin{itemize}\small
      \item Quét \& khởi tạo hàng trăm bean (DigiShield: 9 module nghiệp vụ)
      \item Kết nối datasource, chạy migration/DDL
      \item Dựng security filter chain, listener
      \item Thực tế: \textbf{5--20 giây} mỗi context
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Spring \emph{cache} context theo cấu hình:}
    \begin{itemize}\small
      \item Hai test class có \emph{cùng} cấu hình (properties, profiles,
        MockitoBean\ldots) $\Rightarrow$ dùng lại context, chỉ trả giá 1 lần
      \item Khác nhau 1 property thôi $\Rightarrow$ context \textbf{mới}
    \end{itemize}
  \end{columns}
  \vspace{0.3em}
  \begin{alertblock}{Hệ quả thiết kế}
    Chuẩn hóa cấu hình IT (dùng chung bộ \texttt{properties}, base class) để
    tối đa cache hit. Tránh \texttt{@DirtiesContext} nếu không thật cần ---
    nó vứt bỏ cache.
  \end{alertblock}
\end{frame}

@WebMvcTest(InterceptionController.class)

    @Autowired MockMvc mockMvc;

%---------------------------------------------------------------------
% SLIDE 27 - RLS policy SQL
%---------------------------------------------------------------------
\begin{frame}[fragile]{RLS hoạt động thế nào? (file thật \texttt{it/rls-setup.sql})}
\begin{lstlisting}[language=SQL,style=code,aboveskip=0pt]
ALTER TABLE phishing_report ENABLE ROW LEVEL SECURITY;
-- Ap dung ca voi table owner (superuser/BYPASSRLS van bo qua):
ALTER TABLE phishing_report FORCE ROW LEVEL SECURITY;

CREATE POLICY tenant_isolation ON phishing_report
  USING (tenant_id = NULLIF(
         current_setting('app.tenant_id', true), '')::uuid)
  WITH CHECK (tenant_id = NULLIF(
         current_setting('app.tenant_id', true), '')::uuid);
\end{lstlisting}
  \begin{itemize}\small
    \item \texttt{app.tenant\_id} là một \textbf{GUC} (biến cấu hình của phiên
      PostgreSQL). Production: aspect \texttt{RlsTenantAspect} gọi
      \texttt{set\_config('app.tenant\_id', ?, true)} đầu mỗi transaction.
    \item \texttt{USING}: lọc dòng khi SELECT/UPDATE/DELETE;
      \texttt{WITH CHECK}: chặn INSERT ``gắn nhãn'' tenant khác.
    \item Mọi query --- kể cả query quên filter --- đều bị policy áp lên
      \textbf{trước} mệnh đề WHERE của ứng dụng.
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 28 - FORCE RLS + fail-closed
%---------------------------------------------------------------------
\begin{frame}{Hai chi tiết quyết định chất lượng của test}
  \begin{alertblock}{1. Vì sao phải \texttt{FORCE ROW LEVEL SECURITY}?}
    \small Mặc định PostgreSQL \emph{bỏ qua} RLS cho \textbf{table owner};
    \texttt{FORCE} buộc policy áp cả lên owner --- nhưng \emph{không} chạm
    được superuser hay role có \texttt{BYPASSRLS}. User mặc định của
    Testcontainers lại là \textbf{chủ bảng kiêm superuser}, nên một mình
    \texttt{FORCE} chưa đủ: test còn phải kết nối bằng \texttt{normal\_user}
    \texttt{NOSUPERUSER NOBYPASSRLS} (slide sau). Thiếu hai lớp này, mọi dòng
    đều hiện ra, test cách ly \textbf{pass một cách vô nghĩa} --- lỗi kinh
    điển của test RLS.
  \end{alertblock}
  \begin{block}{2. Fail-closed: thiếu tenant thì thấy \textbf{0 dòng}}
    \small Khi \texttt{app.tenant\_id} chưa được set:
    \texttt{current\_setting(\ldots, true)} trả về NULL
    $\Rightarrow$ vị từ \texttt{tenant\_id = NULL} không bao giờ đúng
    $\Rightarrow$ \textbf{không dòng nào} hiển thị. Quên set tenant $\ne$ lộ
    toàn bộ dữ liệu --- hệ thống ``đóng an toàn''.
  \end{block}
  \small Một test tốt phải kiểm chứng \textbf{cả hai} hành vi này --- và
  \texttt{TenantIsolationIT} làm đúng như vậy.
\end{frame}

private <T> T runAsTenant(UUID tenantId, Supplier<T> work) {

  try {

  } finally {
    TenantContext.clear();
  }
}

  static final PostgreSQLContainer<?> POSTGRES =
      new PostgreSQLContainer<>("postgres:16-alpine");

  // Listener cai rieng cho test: gom moi event phat ra

    private final List<Object> events =
        new CopyOnWriteArrayList<>();
    @EventListener
    public void onEvent(Object event) { events.add(event); }
    public <T> List<T> ofType(Class<T> type) {
      return events.stream().filter(type::isInstance)
          .map(type::cast).toList();
    }
  }

@Test
void addedAccountIsFlaggedByCheck() throws Exception {
  // 1) POST them tai khoan vao watchlist -> 201 Created
  mockMvc.perform(post("/api/v1/account-watchlist")
          .contentType(MediaType.APPLICATION_JSON)
          .content("""
              {"type": "bank_account",
               "value": "9704-2222-9999",
               "risk_level": "high",
               "source": "A05"}"""))
      .andExpect(status().isCreated())
      .andExpect(jsonPath("$.value")
          .value("9704-2222-9999"));

  mockMvc.perform(get("/api/v1/account-watchlist/check")
          .param("value", "9704-2222-9999"))
      .andExpect(status().isOk())
      .andExpect(jsonPath("$.inWatchlist").value(true))
      .andExpect(jsonPath("$.riskLevel").value("HIGH"));
}
\end{lstlisting}
