\begin{frame}{Vì sao không dùng database thật trong unit test?}
  \begin{itemize}
    \item \textbf{Chậm}: khởi động PostgreSQL/H2 + Spring context mất vài giây;
      unit test cần chạy trong \textbf{mili-giây} (hàng trăm test mỗi lần build).
    \item \textbf{Không ổn định}: dữ liệu tồn dư giữa các test, thứ tự chạy
      ảnh hưởng kết quả (vi phạm tính \textit{isolated} của FIRST).
    \item \textbf{Khó dựng tình huống}: làm sao ép repository ném
      exception, hay trả về đúng một bản ghi watchlist "CONFIRMED"?
    \item \textbf{Sai mức kiểm thử}: unit test kiểm tra \textbf{logic quyết định}
      của service; việc JPA đọc/ghi đúng là chuyện của integration test (Buổi 7).
  \end{itemize}

  \begin{block}{Giải pháp}
    Thay dependency thật bằng \textbf{test double} --- đối tượng "đóng thế"
    do ta điều khiển hoàn toàn. Mockito là thư viện tạo test double
    phổ biến nhất trong hệ sinh thái Java.
  \end{block}
\end{frame}

\section{Test double và Mockito}
%=====================================================================

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}{Phân loại test double (G. Meszaros \cite{meszaros})}
  \begin{center}
  \footnotesize
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{>{\raggedright\arraybackslash}p{1.5cm}>{\raggedright\arraybackslash}p{5.6cm}>{\raggedright\arraybackslash}p{5.5cm}}
    \toprule
    \textbf{Loại} & \textbf{Mô tả} & \textbf{Ví dụ DigiShield} \\
    \midrule
    Dummy & Chỉ để lấp tham số, không bao giờ được dùng thật &
      \texttt{UUID} truyền cho DTO \\
    Stub & Trả về câu trả lời \textbf{định sẵn} cho lời gọi &
      repository trả \texttt{Optional.of(hit)} \\
    Fake & Cài đặt thật nhưng đơn giản hóa &
      \texttt{StubAiClient} thay Claude API \\
    Spy & Bọc đối tượng thật, \textbf{ghi lại} lời gọi &
      \texttt{@Spy} trên service thật \\
    Mock & Được lập trình cả \textbf{kỳ vọng về hành vi}: gọi gì, mấy lần &
      \texttt{verify(eventRepository)\allowbreak.save(...)} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \begin{exampleblock}{Lưu ý thuật ngữ}
    Trong Mockito, đối tượng \texttt{@Mock} dùng được như \textbf{cả stub lẫn mock}:
    \texttt{when(...)} để stub, \texttt{verify(...)} để kiểm tra hành vi.
  \end{exampleblock}
\end{frame}

%--- SLIDE 8 ----------------------------------------------------------
\begin{frame}{Stub vs Mock --- hai kiểu xác minh}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{State verification (dùng stub)}
      \begin{itemize}\small
        \item Cho dependency trả dữ liệu định sẵn
        \item Assert trên \textbf{kết quả trả về}
        \item Ví dụ: watchlist hit $\to$ decision phải là \texttt{PAUSE}
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Behaviour verification (dùng mock)}
      \begin{itemize}\small
        \item Kiểm tra service \textbf{đã gọi} dependency đúng cách
        \item Assert trên \textbf{tương tác}
        \item Ví dụ: \texttt{eventRepository.save()} được gọi đúng 1 lần
      \end{itemize}
    \end{block}
  \end{columns}

  \medskip
  \begin{alertblock}{Nguyên tắc chọn}
    Ưu tiên state verification khi method \textbf{trả về giá trị}; dùng
    behaviour verification cho \textbf{side effect} (ghi DB, publish event,
    gửi email) --- vì không có giá trị trả về nào để assert.
  \end{alertblock}
\end{frame}

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}{Mockito là gì?}
  \begin{block}{Mockito \cite{mockito}}
    Thư viện mã nguồn mở (\texttt{site.mockito.org}) tạo test double cho Java.
    Sinh \textbf{proxy class} lúc runtime (ByteBuddy) thay cho interface /
    class thật; mặc định mọi method trả \texttt{null} / \texttt{0} /
    \texttt{false} / collection rỗng.
  \end{block}

  \begin{itemize}
    \item Đi kèm sẵn trong \texttt{spring-boot-starter-test} --- DigiShield
      không cần khai báo thêm dependency nào.
    \item Tích hợp JUnit 5 qua \texttt{MockitoExtension}
      (gói \texttt{mockito-junit-jupiter}).
    \item Mock được interface \textbf{và} class thường; class/method
      \texttt{final}, \texttt{static} cần \texttt{mockito-inline} (hạn chế dùng).
  \end{itemize}

  \begin{exampleblock}{Trong repo DigiShield}
    176 unit test hiện có đều theo cùng một khuôn:
    \texttt{MockitoExtension} + \texttt{@Mock/@InjectMocks/@Captor} + AssertJ.
  \end{exampleblock}
\end{frame}

\section{Thiết lập Mockito}
%=====================================================================

%--- SLIDE 10 ---------------------------------------------------------
\begin{frame}[fragile]{Ba annotation cốt lõi}
  \begin{center}
  \small
  \begin{tabular}{p{2.8cm}p{9.6cm}}
    \toprule
    \textbf{Annotation} & \textbf{Ý nghĩa} \\
    \midrule
    \texttt{@Mock} & Tạo mock cho field --- thay dependency thật \\
    \texttt{@InjectMocks} & Tạo instance \textbf{thật} của class cần test,
      tự tiêm các \texttt{@Mock} vào qua constructor \\
    \texttt{@Captor} & Tạo \texttt{ArgumentCaptor} để bắt argument (xem sau) \\
    \bottomrule
  \end{tabular}
  \end{center}

\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize,aboveskip=0pt,belowskip=0pt]
@ExtendWith(MockitoExtension.class)  // kich hoat Mockito
class InterceptionServiceImplTest {
    // @Mock -> the vao cho 3 dependency
    // @InjectMocks -> new InterceptionServiceImpl(
    //                     watchRepository, eventRepository,
    //                     messages)
}
\end{lstlisting}

  \begin{alertblock}{Quên \texttt{@ExtendWith(MockitoExtension.class)}?}
    Các field \texttt{@Mock} sẽ là \texttt{null} $\to$
    \texttt{NullPointerException} ngay dòng đầu tiên của test.
  \end{alertblock}
\end{frame}

%--- SLIDE 12 ---------------------------------------------------------
\begin{frame}[fragile]{Setup và teardown --- trích test thật}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\footnotesize,aboveskip=0pt,belowskip=0pt]
@BeforeEach
void setUp() {
    // Service doc tenant tu TenantContext (ThreadLocal)
    TenantContext.set(TENANT_ID.toString());
    // save() tra ve chinh doi tuong duoc luu
    lenient().when(eventRepository.save(
            any(InterventionEvent.class)))
        .thenAnswer(inv -> inv.getArgument(0));
    // messages.get(key) tra ve chinh key
    lenient().when(messages.get(anyString()))
        .thenAnswer(inv -> inv.getArgument(0));
}

@AfterEach
void tearDown() {
    TenantContext.clear();  // don ThreadLocal sau moi test
}
\end{lstlisting}
\end{frame}

%--- SLIDE 13 ---------------------------------------------------------
\begin{frame}{Vì sao cần \texttt{lenient()}?}
  \begin{block}{Strict stubbing (mặc định từ Mockito 2+)}
    Mockito \textbf{fail test} nếu một stub được khai báo mà không được dùng:
    \texttt{UnnecessaryStubbingException}. Mục đích: phát hiện test "chết"
    và stub thừa.
  \end{block}

  \begin{itemize}
    \item Stub trong \texttt{@BeforeEach} áp dụng cho \textbf{mọi} test của class.
    \item Nhưng test \texttt{checkAccount\_...} không hề gọi
      \texttt{eventRepository.save()} $\to$ stub đó "thừa" với test này.
    \item \texttt{lenient()} đánh dấu: stub này \textit{được phép} không dùng.
  \end{itemize}

  \begin{exampleblock}{Kinh nghiệm}
    Chỉ \texttt{lenient()} cho stub dùng chung trong \texttt{@BeforeEach};
    stub riêng của từng test vẫn để strict --- nó là "máy dò" stub rác miễn phí.
  \end{exampleblock}
\end{frame}

\section{Stubbing}
%=====================================================================

%--- SLIDE 14 ---------------------------------------------------------
\begin{frame}[fragile]{Stubbing cơ bản: \texttt{when / thenReturn} (1/2)}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// Hai kich ban THAY THE nhau -- moi kich ban dat trong
// MOT test rieng (dat chung: stub sau ghi de stub truoc,
// strict stubbing bao UnnecessaryStubbing)

// Kich ban A: "khi duoc hoi tai khoan nay, tra ve mot
// ban ghi watchlist muc CONFIRMED"
AccountWatchEntry watchHit = new AccountWatchEntry(
    UUID.randomUUID(), TENANT_ID, WatchType.BANK_ACCOUNT,
    DEST_ACCOUNT, RiskLevel.CONFIRMED, "fraud-feed");

when(watchRepository
        .findByTenantIdAndValue(TENANT_ID, DEST_ACCOUNT))
    .thenReturn(Optional.of(watchHit));
\end{lstlisting}
\end{frame}

%--- SLIDE 14b --------------------------------------------------------
\begin{frame}[fragile]{Stubbing cơ bản: \texttt{when / thenReturn} (2/2)}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// Kich ban B: khong co trong watchlist
when(watchRepository
        .findByTenantIdAndValue(TENANT_ID, DEST_ACCOUNT))
    .thenReturn(Optional.empty());
\end{lstlisting}

  \begin{block}{Đọc như một câu điều kiện}
    \texttt{when(mock.method(args)).thenReturn(value)} ---
    "khi method được gọi với args này thì trả về value".
    Chưa stub $\to$ trả mặc định: object là \texttt{null}, nhưng
    \texttt{Optional} là \texttt{Optional.empty()}, collection là rỗng.
  \end{block}
\end{frame}

%--- SLIDE 15 ---------------------------------------------------------
\begin{frame}[fragile]{\texttt{thenThrow}, \texttt{thenAnswer}, stub nhiều lần gọi}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// Gia lap DB loi: lan goi nao cung nem exception
when(watchRepository.findByTenantIdAndValue(any(), any()))
    .thenThrow(new DataAccessResourceFailureException(
        "connection lost"));

\section{Anti-patterns}
%=====================================================================

%--- SLIDE 45 ---------------------------------------------------------
\begin{frame}[fragile]{Anti-pattern 1--2: over-mocking, test chép lại cài đặt}
  \begin{alertblock}{Over-mocking}
    Mock cả những thứ không cần: DTO, entity, \texttt{List}, class tiện ích.
    \texttt{EvaluateRequest} là record --- cứ \texttt{new} nó ra! Chỉ mock
    \textbf{dependency có side effect hoặc khó dựng} (DB, mạng, thời gian).
  \end{alertblock}

  \begin{alertblock}{Test chép lại cài đặt (mirror implementation)}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// XAU: lap lai cong thuc cua service trong test
int expected = Math.max(0, Math.min(100, 5 + weight));
assertThat(result.getValue()).isEqualTo(expected);

// TOT (nhu test that): tinh tay ky vong tu dac ta
// 2 click x 25 diem -> 5 + 50 = 55
assertThat(result.getValue()).isEqualTo(55);
\end{lstlisting}
    Copy công thức vào test $\to$ code sai thì test cũng sai theo,
    không bao giờ fail. Kỳ vọng phải đến từ \textbf{đặc tả}, không từ code.
  \end{alertblock}
\end{frame}

%--- SLIDE 46 ---------------------------------------------------------
\begin{frame}[fragile]{Anti-pattern 3--4: mock "trả lời mọi thứ", assert quá ít}
  \begin{alertblock}{Mock trả lời mọi thứ}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// XAU: any() moi noi -> test khong con kiem tra
// service truyen DUNG tenant/tai khoan hay khong
when(watchRepository.findByTenantIdAndValue(
        any(), any())).thenReturn(Optional.of(hit));
\end{lstlisting}
    Test thật của DigiShield stub với \texttt{TENANT\_ID, DEST\_ACCOUNT} cụ thể:
    nếu service truyền nhầm tenant, stub không khớp $\to$ test fail. Strict
    stubbing chỉ cứu được một phần --- hãy stub hẹp nhất có thể.
  \end{alertblock}

  \begin{alertblock}{Assert quá ít (test tăng coverage rỗng)}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\scriptsize]
// XAU: goi xong khong assert gi ca
interceptionService.evaluate(request);   // "no crash" ??
\end{lstlisting}
    Coverage 100\% nhưng không phát hiện lỗi nào. Mutation testing (PIT)
    sẽ vạch trần loại test này --- mỗi test tối thiểu một assert có ý nghĩa.
  \end{alertblock}
\end{frame}

      \item Bộ \textbf{unit test Mockito} cho service chính của module nhóm
        đang phụ trách (interception, reporting, auth, analytics,
        simulation\ldots):
        \begin{itemize}\small
          \item Mỗi nhánh quyết định một test method (như \texttt{evaluate()})
          \item Có ít nhất: 1 ArgumentCaptor, 1 test exception
            (type + message + side effect), 1 verify publish sự kiện
            hoặc \texttt{verifyNoInteractions}
          \item Tổ chức bằng \texttt{@Nested} + \texttt{@DisplayName} tiếng Việt
        \end{itemize}
      \item Ảnh chụp \textbf{JaCoCo report}: branch coverage của service
        được test $\geq 80\%$; ghi rõ nhánh nào chưa phủ và vì sao.
      \item Bảng ánh xạ: test method $\leftrightarrow$ nhánh/rule
        $\leftrightarrow$ kỹ thuật (stub / captor / verify) --- như slide
        "Tổng kết ca kiểm thử".
    \end{enumerate}
  \end{block}

  {\small \textbf{Định dạng}: push lên nhánh \texttt{feat/btl-unit-test-<nhóm>}
  của repo nhóm + nộp PDF báo cáo \texttt{Nhom\_XX\_Buoi06.pdf}.
  Hạn: trước Buổi 7.}
\end{frame}

\begin{frame}{Câu hỏi ôn tập}
  \begin{enumerate}\small
    \item Phân biệt stub và mock theo Meszaros. Trong
      \texttt{InterceptionServiceImplTest}, \texttt{watchRepository} đóng
      vai trò nào ở test PAUSE?
    \item Vì sao \texttt{@BeforeEach} của test đó phải dùng
      \texttt{lenient()}? Bỏ đi thì test nào sẽ fail và với exception gì?
    \item Viết lại \texttt{when(repo.findByTenantIdAndValue(TENANT\_ID,
      any()))} cho đúng quy tắc matcher.
    \item Khi nào bắt buộc dùng \texttt{ArgumentCaptor} thay vì assert
      giá trị trả về? Cho ví dụ từ \texttt{evaluate()}.
    \item Test "dismiss không publish event" bảo vệ chúng ta khỏi bug
      nghiệp vụ nào trong chuỗi reporting $\to$ analytics?
    \item \texttt{@Mock} và \texttt{@MockitoBean} khác nhau thế nào?
      Nếu dùng \texttt{@MockitoBean} trong test không có
      \texttt{@SpringBootTest} thì sao?
  \end{enumerate}
\end{frame}
