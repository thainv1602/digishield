\section{Giới thiệu JUnit}

%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}[shrink=8]{Vì sao phải tự động hóa unit test?}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Kiểm thử thủ công}
      \begin{itemize}
        \item Chạy app, nhập tay, quan sát mắt thường
        \item Lặp lại sau \textbf{mỗi lần sửa code} --- tốn công
        \item Không đo được độ phủ
        \item Dễ bỏ sót trường hợp biên
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Unit test tự động}
      \begin{itemize}
        \item Viết một lần, chạy hàng nghìn lần
        \item Chạy trong CI/CD trước khi merge
        \item Là ``lưới an toàn'' khi refactor
        \item Là \textbf{tài liệu sống} (living documentation) mô tả hành vi code
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.4em}
  \begin{exampleblock}{Trong DigiShield}
    Backend hiện có \textbf{296 test} trong 31 class test.
    Mỗi lần đẩy code, toàn bộ chạy lại bằng \texttt{./gradlew test} ---
    vài chục giây thay vì vài giờ kiểm thử tay.
  \end{exampleblock}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}[shrink=8]{JUnit là gì?}
  \begin{block}{Định nghĩa}
    JUnit là \textbf{framework kiểm thử đơn vị} (unit testing framework)
    cho Java: cung cấp annotation để đánh dấu test, assertion để
    kiểm tra kết quả, và bộ chạy (runner) để thực thi + báo cáo.\sref{5}
  \end{block}

  \vspace{0.35em}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Lịch sử:}
    \begin{itemize}
      \item 1997: Kent Beck \& Erich Gamma viết JUnit
            trên chuyến bay tới hội nghị OOPSLA
      \item Khởi nguồn họ framework \textbf{xUnit}
            (NUnit, PyTest, PHPUnit, \dots)
      \item 2006: JUnit 4 (annotation \texttt{@Test})
      \item 2017: JUnit 5 --- viết lại toàn bộ
    \end{itemize}

    \column{0.44\textwidth}
    \textbf{Vị thế hiện nay:}
    \begin{itemize}
      \item Chuẩn \textit{de facto} cho test Java
      \item Tích hợp sẵn trong Maven, Gradle, IntelliJ, Eclipse, CI/CD
      \item \texttt{spring-boot-\allowbreak starter-test} chọn JUnit 5 làm mặc định
    \end{itemize}
  \end{columns}
\end{frame}

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}{JUnit 4 \(\to\) JUnit 5: có gì thay đổi?\sref{5}}
  \footnotesize
  \renewcommand{\arraystretch}{1.05}
  \setlength{\tabcolsep}{4pt}
  \begin{tabularx}{\textwidth}{l>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X}
    \toprule
    & \textbf{JUnit 4 (2006)} & \textbf{JUnit 5 (2017--nay)} \\
    \midrule
    Kiến trúc & 1 file jar duy nhất & 3 phần: Platform + Jupiter + Vintage \\
    Java tối thiểu & Java 5 & Java 8+ (lambda) \\
    Trước/sau mỗi test & \texttt{@Before}/\allowbreak\texttt{@After} & \texttt{@BeforeEach}/\allowbreak\texttt{@AfterEach} \\
    Trước/sau tất cả & \texttt{@BeforeClass}/\allowbreak\texttt{@AfterClass} & \texttt{@BeforeAll}/\allowbreak\texttt{@AfterAll} \\
    Bỏ qua test & \texttt{@Ignore} & \texttt{@Disabled} \\
    Mở rộng & \texttt{@RunWith} (1 runner) & \texttt{@ExtendWith} (nhiều extension) \\
    Test theo tham số & Cồng kềnh (runner riêng) & \texttt{@ParameterizedTest} tích hợp \\
    Test lồng nhau & Không & \texttt{@Nested} \\
    \bottomrule
  \end{tabularx}
  \renewcommand{\arraystretch}{1}

  \vspace{0.3em}
  {\small \textbf{Lưu ý khi đọc tài liệu trên mạng}: nhiều bài viết cũ vẫn dùng
  \texttt{@Before}, \texttt{@RunWith} của JUnit 4 --- các annotation này
  \textbf{không tồn tại} trong JUnit 5 (Jupiter).}
\end{frame}

\section{Các bước cài đặt JUnit}

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}{Các bước cài đặt JUnit --- tổng quan}
  \begin{enumerate}
    \setlength{\itemsep}{0.4em}
    \item \textbf{Khai báo dependency} JUnit Jupiter trong file build
          (\texttt{pom.xml} với Maven, \texttt{build.gradle.kts} với Gradle)
    \item \textbf{Bảo đảm build tool dùng JUnit Platform} để chạy test
          (Maven Surefire 2.22+; Gradle \texttt{useJUnitPlatform()}
          hoặc \texttt{useJUnitJupiter()})
    \item \textbf{Tạo thư mục} \texttt{src/test/java} song song với
          \texttt{src/main/java}
    \item \textbf{Viết class test} theo quy ước \texttt{*Test.java}
    \item \textbf{Chạy}: \texttt{mvn test} / \texttt{./gradlew test} / nút Run trong IDE
  \end{enumerate}

  \vspace{0.4em}
  \begin{exampleblock}{Hai con đường trong môn học}
    \textbf{Maven}: cho project riêng của nhóm (nếu nhóm tự chọn đề tài).\\
    \textbf{Gradle}: theo đúng cấu hình thật của DigiShield --- dùng cho BTL.
  \end{exampleblock}
\end{frame}

%--- SLIDE 10 ---------------------------------------------------------
\begin{frame}[fragile,shrink=8]{Cài đặt với Maven --- \texttt{pom.xml}}
\begin{lstlisting}[language=xml]
<!-- 1. Dependency JUnit Jupiter (scope test) -->
<dependency>
  <groupId>org.junit.jupiter</groupId>
  <artifactId>junit-jupiter</artifactId>
  <version>5.11.4</version>
  <scope>test</scope>
</dependency>

<!-- 2. Surefire 2.22+ de ho tro JUnit Platform -->
<plugin>
  <groupId>org.apache.maven.plugins</groupId>
  <artifactId>maven-surefire-plugin</artifactId>
  <version>3.5.2</version>
</plugin>
\end{lstlisting}
  \vspace{-0.2em}
  {\small Chạy: \texttt{mvn test}. Với project Spring Boot chỉ cần
  \texttt{spring-boot-starter-test} --- đã gói sẵn JUnit 5.}
\end{frame}

%--- SLIDE 11 ---------------------------------------------------------
\begin{frame}[fragile,shrink=8]{Cài đặt với Gradle --- project thường}
\begin{lstlisting}[style=nonumber]
// build.gradle.kts
dependencies {
    // BOM giu cac artifact JUnit cung phien ban
    testImplementation(platform(
        "org.junit:junit-bom:5.11.4"))
    testImplementation(
        "org.junit.jupiter:junit-jupiter")
    testRuntimeOnly(
        "org.junit.platform:junit-platform-launcher")
}

tasks.test {
    // Bat buoc: chay test tren JUnit Platform
    useJUnitPlatform()
}
\end{lstlisting}
  {\small Thiếu \texttt{useJUnitPlatform()} là lỗi kinh điển:
  Gradle biên dịch test nhưng \textbf{không chạy} test nào cả.}
\end{frame}

%--- SLIDE 12 ---------------------------------------------------------
\begin{frame}[fragile,shrink=8]{Cài đặt trong DigiShield --- convention cho 9 module}
  \begin{block}{\small Trích \texttt{buildSrc/src/main/kotlin/\allowbreak digishield.spring-\allowbreak module-\allowbreak conventions.gradle.kts}}
\begin{lstlisting}[style=nonumber]
dependencies {
    // ... dependency production ...
    "testImplementation"("org.springframework.boot:"
        + "spring-boot-starter-test")
    "testImplementation"("org.springframework.modulith:"
        + "spring-modulith-starter-test")
    "testRuntimeOnly"("org.junit.platform:"

    useJUnitPlatform()
}
\end{lstlisting}
  \end{block}
  \vspace{-0.2em}
  {\small Convention plugin áp dụng cho cả 9 module nghiệp vụ
  (\texttt{auth}, \texttt{ai}, \texttt{reporting}, \dots) --- khai báo
  \textbf{một lần}, mọi module đều có JUnit 5.}
\end{frame}

%--- SLIDE 13 ---------------------------------------------------------
\begin{frame}{\texttt{spring-boot-starter-test} kéo theo những gì?}
  \renewcommand{\arraystretch}{1.2}
  \begin{tabularx}{\textwidth}{lX}
    \toprule
    \textbf{Thư viện} & \textbf{Vai trò} \\
    \midrule

  \vspace{0.5em}
  \begin{exampleblock}{Ý nghĩa}

    của cả 3 buổi 5--6--7. Không cần khai báo JUnit/Mockito/AssertJ riêng lẻ.
  \end{exampleblock}
\end{frame}

%--- SLIDE 14 ---------------------------------------------------------
\begin{frame}[fragile,shrink=12]{DigiShield: JVM Test Suites --- tách unit và integration\sref{7}}
  \begin{block}{\small Trích \texttt{boot/app/build.gradle.kts} (cấu hình thật)}
\begin{lstlisting}[style=nonumber]
testing {
  suites {
    // Suite unit test mac dinh -> JUnit Jupiter
    getByName<JvmTestSuite>("test") {
      useJUnitJupiter()
    }
    // Suite integration test: Spring + Testcontainers
    register<JvmTestSuite>("integrationTest") {
      useJUnitJupiter()
      // ... dependency rieng: Testcontainers, ...
      targets { all { testTask.configure {
        // IT cham (can Docker) chay sau UT nhanh
        shouldRunAfter(tasks.named("test"))
      } } }
    }
  }
}
\end{lstlisting}
  \end{block}
\end{frame}

String ageLabel(Instant reportedAt, Instant now) {
    if (reportedAt == null) {
        return null;
    }
    Duration d = Duration.between(reportedAt, now);
    long mins = Math.max(0, d.toMinutes());
    if (mins < 60) {
        return mins + "p";    // vd "5p"  (p = phut)
    }
    long hours = d.toHours();
    if (hours < 24) {
        return hours + "h";   // vd "3h"
    }
    return d.toDays() + "d";  // vd "2d"
}
\end{lstlisting}
  \end{block}

\begin{frame}[shrink=8]{AssertJ --- assertion kiểu fluent}
  \begin{block}{AssertJ là gì?\sref{6}}
    Thư viện assertion đi kèm \texttt{spring-boot-starter-test}, cú pháp
    \textbf{chuỗi} (fluent): \texttt{assertThat(actual).isEqualTo(expected)}.

  \end{block}

  \vspace{0.35em}
  \textbf{Vì sao đội DigiShield chọn AssertJ?}
  \begin{itemize}
    \item Đọc như câu văn: \texttt{assertThat(reasons).hasSize(2)}
    \item Không nhầm thứ tự expected/actual như \texttt{assertEquals}
    \item IDE gợi ý đúng method theo \textbf{kiểu dữ liệu}
          (String có \texttt{isNotBlank}, List có \texttt{contains}, \dots)
    \item Thông báo lỗi giàu ngữ cảnh: in phần tử thiếu/thừa của collection
  \end{itemize}

  \vspace{0.35em}
  \begin{exampleblock}{Import duy nhất cần nhớ}
    \texttt{import static org.assertj.core.api.Assertions.assertThat;}
  \end{exampleblock}
\end{frame}

    long spamHits   = countHits(text, SPAM_SIGNALS);

public ModerationView moderate(String content) {
    String text = content == null
        ? "" : content.toLowerCase(Locale.ROOT);

    List<String> reasons = new ArrayList<>();
    for (String word : UNSAFE_WORDS) {
        if (text.contains(word)) {
            reasons.add("Noi dung chua tu: " + word);
        }
    }
    String verdict;
    if (reasons.isEmpty())        verdict = "pass";
    else if (reasons.size() >= 2) verdict = "block";
    else                          verdict = "flag";
    return new ModerationView(
        verdict, List.copyOf(reasons));
}
\end{lstlisting}

  1 \(\to\) \texttt{flag}; \(\ge 2 \to\) \texttt{block} ---
  tương ứng ba nhánh code, đúng kiểu độ phủ nhánh đã học ở hộp trắng.}

\begin{frame}[fragile,shrink=8]{Vấn đề: nhiều test chỉ khác dữ liệu}

    \column{0.52\textwidth}
\begin{lstlisting}[language=java,style=nonumber]
@Test

@Test

      Copy--paste test chỉ đổi số liệu:
      \begin{itemize}
        \item Dài dòng, khó đọc
        \item Sửa logic phải sửa \(n\) chỗ
        \item Dễ quên bổ sung ca biên mới
      \end{itemize}
    \end{alertblock}
    \begin{block}{Giải pháp}
      \texttt{@ParameterizedTest}: \textbf{một} method,
      \textbf{nhiều} bộ dữ liệu --- mỗi bộ hiện thành
      một test riêng trong report.
    \end{block}
  \end{columns}
\end{frame}
