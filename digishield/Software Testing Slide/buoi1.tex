\section{Kiểm thử phần mềm là gì?}
%=====================================================================

%---------------------------------------------------------------------
\begin{frame}{Định nghĩa}
  \begin{block}{IEEE 610.12-1990 \cite{ieee610}}
    Kiểm thử phần mềm là quá trình \textbf{vận hành} một hệ thống hoặc thành phần
    trong các điều kiện được chỉ định, \textbf{quan sát} và \textbf{ghi lại kết quả},
    và \textbf{đánh giá} một số khía cạnh của hệ thống hoặc thành phần đó.
  \end{block}

  \vspace{0.8em}
  \begin{columns}
    \column{0.48\textwidth}
    \textbf{Kiểm thử KHÔNG phải là:}
    \begin{itemize}
      \item Chứng minh phần mềm đúng
      \item Đảm bảo không còn lỗi
      \item Chỉ chạy chương trình xem có lỗi không
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Kiểm thử LÀ:}
    \begin{itemize}
      \item Tìm kiếm lỗi một cách có hệ thống
      \item Đánh giá chất lượng phần mềm
      \item Cung cấp thông tin để ra quyết định
    \end{itemize}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Mục tiêu của kiểm thử}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Đánh giá chất lượng}
      \begin{itemize}
        \item Đo mức độ đáp ứng yêu cầu chức năng và phi chức năng
        \item Cung cấp thông tin cho quyết định phát hành
        \item Xây dựng niềm tin vào sản phẩm
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Phát hiện lỗi}
      \begin{itemize}
        \item Tìm khiếm khuyết (defect) \textbf{trước} người dùng cuối
        \item Ngăn lỗi lan sang giai đoạn sau
        \item Giảm rủi ro vận hành và chi phí sửa chữa
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.5em}
  \begin{exampleblock}{Quan điểm của Myers (The Art of Software Testing) \cite{myers}}
    Một test case \textbf{tốt} là test case có xác suất cao \textbf{tìm ra một lỗi
    chưa được phát hiện} --- không phải test case chứng tỏ chương trình chạy đúng.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Error --- Fault (Defect) --- Failure}
  \begin{center}
  \begin{tikzpicture}[
    box/.style={rectangle,draw=darkgray,fill=lightgray,rounded corners=2pt,
                minimum width=2.5cm,minimum height=1cm,align=center,font=\small},
    arr/.style={-{Stealth[length=5pt]},darkgray,thick}
  ]
    \node[box] (error) at (0,0) {\textbf{Error}\\(Lỗi con người)};
    \node[box] (defect) at (4.2,0) {\textbf{Fault / Defect}\\(Trong mã nguồn)};
    \node[box] (failure) at (8.4,0) {\textbf{Failure}\\(Khi thực thi)};

    \draw[arr] (error.east) -- node[above,font=\scriptsize]{gây ra} (defect.west);
    \draw[arr] (defect.east) -- node[above,font=\scriptsize]{biểu hiện} (failure.west);
  \end{tikzpicture}
  \end{center}

  \vspace{0.5em}
  \begin{exampleblock}{Ví dụ trên DigiShield --- phân trang nhật ký can thiệp}
    \begin{itemize}
      \item \textbf{Error}: lập trình viên \textit{quên} rằng tham số \texttt{size}
            do client gửi lên cần được giới hạn (clamp)
      \item \textbf{Fault}: \texttt{listInterventions(page, size)} dùng thẳng
            \texttt{size} để truy vấn, không chặn trên 100
      \item \textbf{Failure}: client gọi \texttt{GET /interventions?size=1000}
            $\to$ hệ thống trả về 1000 bản ghi, quá tải bộ nhớ và băng thông
    \end{itemize}
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Mã thật của DigiShield đã ``vá'' fault đó thế nào?}
  \begin{block}{\small \texttt{modules/interception/.../InterceptionServiceImpl.java}}
\begin{lstlisting}[language=java]
public List<InterventionEventView> listInterventions(
    int page, int size) {
  // Chan duoi: page toi thieu la 1
  int safePage = Math.max(page, 1);
  // Chan hai dau: 1 <= size <= 100
  int safeSize = Math.min(Math.max(size, 1), 100);
  // ... truy van voi safePage, safeSize
}
\end{lstlisting}
  \end{block}

  \vspace{0.2em}
  \small
  \textbf{Câu hỏi kiểm thử}: những giá trị \texttt{size} nào đáng để thử?
  $0$, $1$, $2$, $99$, $100$, $101$? --- Đây chính là \textbf{phân tích giá trị biên
  (BVA)}, sẽ học ở Buổi 8.
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Tại sao cần kiểm thử? --- Chi phí lỗi}
  \begin{block}{Quy tắc 1 : 10 : 100}
    Chi phí sửa lỗi tăng theo cấp bậc khi phát hiện muộn hơn trong vòng đời.
  \end{block}

  \vspace{0.2em}
  \begin{center}
  \begin{tikzpicture}[scale=0.85]
    \small
    \draw[->,darkgray,thick] (0,0) -- (10.5,0) node[right]{\scriptsize Giai đoạn};
    \draw[->,darkgray,thick] (0,0) -- (0,4.2) node[above]{\scriptsize Chi phí};

    \foreach \x/\label/\h in {
      1/Yêu cầu/0.4,
      2.5/Thiết kế/0.8,
      4/Lập trình/1.4,
      5.5/Kiểm thử/2.2,
      7/Triển khai/3.2,
      8.5/Vận hành/4.0
    }{
      \draw[fill=darkgray!70,darkgray] (\x-0.4,0) rectangle (\x+0.4,\h);
      \node[font=\tiny,rotate=30,anchor=north east] at (\x+0.3,-0.08) {\label};
    }
  \end{tikzpicture}

%---------------------------------------------------------------------
\begin{frame}{Verification và Validation}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Verification --- Xác minh}
      \emph{``Are we building the product \textbf{right}?''}
      \begin{itemize}
        \item Sản phẩm có đúng với \textbf{đặc tả} không?
        \item Review tài liệu, thiết kế, mã nguồn
        \item Thực hiện được từ rất sớm
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Validation --- Thẩm định}
      \emph{``Are we building the \textbf{right} product?''}
      \begin{itemize}
        \item Sản phẩm có đáp ứng \textbf{nhu cầu người dùng} không?
        \item Chạy thử hệ thống trong bối cảnh thật
        \item Thường thực hiện ở giai đoạn sau
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.2em}
  \begin{exampleblock}{Ví dụ DigiShield}
    \small
    \textbf{Verification}: quyết định can thiệp có đúng quy tắc đặc tả
    (đang nghe điện thoại + người nhận mới + tài khoản trong danh sách theo dõi
    $\to$ tạm dừng)?\\
    \textbf{Validation}: màn hình cảnh báo có thực sự \textbf{ngăn được} người dùng
    lớn tuổi chuyển tiền cho kẻ lừa đảo hay không?
  \end{exampleblock}
\end{frame}

\section{Bảy nguyên tắc kiểm thử (ISTQB)}
%=====================================================================

%---------------------------------------------------------------------
\begin{frame}{Tổng quan 7 nguyên tắc ISTQB}
  \begin{enumerate}
    \item Kiểm thử cho thấy sự \textbf{hiện diện} của lỗi, không phải sự \textbf{vắng mặt}
    \item Kiểm thử \textbf{vét cạn} (exhaustive testing) là bất khả thi
    \item Kiểm thử \textbf{sớm} (early testing) tiết kiệm chi phí
    \item \textbf{Phân cụm} lỗi (defect clustering)
    \item \textbf{Nghịch lý thuốc trừ sâu} (pesticide paradox)
    \item Kiểm thử phụ thuộc vào \textbf{ngữ cảnh} (context-dependent)
    \item \textbf{Ngụy biện} ``không có lỗi'' (absence-of-errors fallacy)
  \end{enumerate}

  \vspace{0.5em}
  \begin{block}{Lưu ý}
    Các nguyên tắc này không phải quy tắc cứng nhắc mà là \textbf{hướng dẫn tư duy}
    cho tester khi ra quyết định (theo ISTQB Foundation Syllabus \cite{istqb}).
  \end{block}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 1: Kiểm thử cho thấy sự hiện diện của lỗi}
  \begin{block}{Phát biểu}
    Kiểm thử có thể chứng minh phần mềm \textbf{có lỗi}, nhưng \textbf{không thể}
    chứng minh phần mềm hoàn toàn không có lỗi.
  \end{block}

  \vspace{0.5em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Hệ quả thực tế:}
    \begin{itemize}
      \item Khi tất cả test case pass, không có nghĩa phần mềm đúng
      \item Cần liên tục thiết kế test case mới
      \item Mục tiêu: \textit{giảm rủi ro}, không phải \textit{đảm bảo hoàn hảo}
    \end{itemize}

    \column{0.5\textwidth}
    \begin{exampleblock}{Ví dụ DigiShield}
      \small \texttt{InterceptionServiceImpl} hiện có 5 unit test đều pass ---
      nhưng điều đó không chứng minh logic can thiệp đúng với \textbf{mọi} tổ hợp
      tín hiệu, số tiền và trạng thái cuộc gọi có thể xảy ra.
    \end{exampleblock}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 2: Kiểm thử vét cạn là bất khả thi}
  \begin{block}{Vấn đề}
    Số tổ hợp đầu vào của bất kỳ hệ thống thực tế (non-trivial) nào cũng
    \textbf{vô hạn hoặc quá lớn} --- không thể kiểm thử hết mọi trường hợp.
  \end{block}

  \vspace{0.5em}
  \begin{exampleblock}{Ví dụ định lượng --- DigiShield}
    API \texttt{POST /api/v1/interventions/evaluate} nhận:
    \begin{itemize}
      \item \texttt{amount}: số thập phân (BigDecimal) --- miền giá trị khổng lồ
      \item \texttt{destAccount}: chuỗi số tài khoản bất kỳ
      \item \texttt{onCall}, \texttt{newPayee}: 2 boolean --- $2 \times 2 = 4$ tổ hợp
    \end{itemize}
    Chỉ riêng \texttt{amount} và \texttt{destAccount} đã cho số tổ hợp
    $> 10^{50}$ --- không thể thử hết.
  \end{exampleblock}

  \vspace{0.3em}
  \textbf{Giải pháp}: ưu tiên test case theo \textbf{rủi ro} và
  \textbf{tầm quan trọng nghiệp vụ} (kỹ thuật EP, BVA --- Buổi 8).
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 3 \& 4: Kiểm thử sớm và Phân cụm lỗi}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Nguyên tắc 3: Kiểm thử sớm}
      \begin{itemize}
        \item Bắt đầu kiểm thử ngay từ giai đoạn phân tích yêu cầu
        \item Review đặc tả, review thiết kế --- đây là kiểm thử tĩnh
        \item Chi phí sửa lỗi tăng theo cấp bậc khi phát hiện muộn
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Nguyên tắc 4: Phân cụm lỗi (defect clustering)}
      Nguyên lý Pareto (80/20): phần lớn lỗi tập trung ở
      \textbf{một số ít module}.

      \vspace{0.3em}
      \begin{exampleblock}{\small DigiShield}
        \small Module \texttt{interception} và \texttt{analytics} chứa nhiều
        logic điều kiện, biên số học nhất $\to$ đáng được ưu tiên kiểm thử
        hơn module \texttt{tenancy} chủ yếu CRUD.
      \end{exampleblock}
    \end{block}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Nguyên tắc 5: Nghịch lý thuốc trừ sâu}
  \begin{block}{Phát biểu (pesticide paradox)}
    Nếu cùng một tập test case được chạy lặp đi lặp lại, chúng sẽ
    \textbf{mất khả năng phát hiện lỗi mới} theo thời gian.
  \end{block}

  \vspace{-0.2em}
  \begin{center}
  \begin{tikzpicture}[scale=0.74]
    \draw[->,darkgray] (0,0) -- (8,0) node[right]{\small Số lần chạy};
    \draw[->,darkgray] (0,0) -- (0,3.5) node[above]{\small Lỗi tìm được};
    \draw[darkgray,thick] plot[smooth] coordinates {
      (0.5,3.0)(1.5,2.5)(2.5,1.8)(3.5,1.2)(4.5,0.7)(5.5,0.4)(7,0.2)
    };
    \node[font=\small] at (4,-0.45) {Hiệu quả giảm dần theo thời gian};
  \end{tikzpicture}
  \end{center}
  \vspace{-0.4em}

  \small\textbf{Giải pháp}: Định kỳ \textbf{xem xét và bổ sung} test case mới.
  Ví dụ DigiShield: kẻ lừa đảo đổi chiêu thức liên tục $\to$ bộ test cho
  \texttt{StubAiClient} (phân loại threat/spam/clean) phải được cập nhật
  theo mẫu lừa đảo mới.
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 6 \& 7}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Nguyên tắc 6: Phụ thuộc ngữ cảnh}
      Không có cách tiếp cận kiểm thử duy nhất cho mọi dự án.

      \vspace{0.3em}
      \begin{tabularx}{\columnwidth}{l>{\raggedright\arraybackslash}X}
        \small\textbf{Hệ thống} & \small\textbf{Ưu tiên} \\
        \midrule
        \small Y tế & \small Safety, reliability \\
        \small Chống lừa đảo & \small Security, accuracy \\
        \small Game & \small Gameplay, trải nghiệm chơi \\
        \small E-commerce & \small Performance, UX \\
      \end{tabularx}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Nguyên tắc 7: Ngụy biện ``không có lỗi''}
      \textit{Absence-of-errors fallacy}: một hệ thống vượt qua toàn bộ test case
      nhưng \textbf{không đáp ứng nhu cầu người dùng} vẫn là hệ thống thất bại.

      \vspace{0.3em}
      \begin{exampleblock}{\small Ví dụ DigiShield}
        \small Màn hình cảnh báo giao dịch hiển thị đúng 100\% test case,
        nhưng nội dung khó hiểu với người lớn tuổi $\to$ họ vẫn bấm
        ``Tiếp tục'' và mất tiền.
      \end{exampleblock}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile,shrink=8]{Nhìn thử một class sẽ gặp lại nhiều lần}
  \begin{block}{\small \texttt{InterceptionServiceImpl.evaluate()} (rút gọn)}
\begin{lstlisting}[language=java,basicstyle=\ttfamily\footnotesize]
public InterventionDecision evaluate(EvaluateRequest req) {
  boolean watchlistHit = watchlist.contains(
      req.destAccount());
  // Quy tac nghiep vu chong lua dao
  if (req.onCall() && req.newPayee() && watchlistHit) {
    return pause(req);   // Tam dung giao dich
  } else if (watchlistHit) {
    return warn(req);    // Canh bao nguoi dung
  }
  return allow(req);     // Cho phep
}
\end{lstlisting}
  \end{block}

  \vspace{0.2em}
  \small Ba đầu ra \texttt{PAUSE / WARN / ALLOW} --- đây là ví dụ ``chủ lực''
  xuyên suốt môn học: đồ thị luồng điều khiển (Buổi 3), bảng quyết định (Buổi 8),
  unit test với Mockito (Buổi 6), API test (Buổi 9).
\end{frame}

void pausesWhenOnCallNewPayeeAndWatchlisted() {
