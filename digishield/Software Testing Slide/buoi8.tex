\section{Equivalence Partitioning}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 9
%---------------------------------------------------------------------
\begin{frame}[shrink=10]{Equivalence Partitioning --- Nguyên lý}
  \begin{block}{Định nghĩa}
    EP chia miền giá trị đầu vào thành các \textbf{lớp tương đương}
    (equivalence class) sao cho các giá trị trong cùng một lớp
    được hệ thống \textbf{xử lý theo cùng một cách}.
  \end{block}

  \vspace{0.2em}
  \begin{block}{Nguyên tắc chọn đại diện}
    Nếu một giá trị trong lớp phát hiện lỗi, thì các giá trị khác
    trong cùng lớp cũng sẽ phát hiện lỗi đó.\\[0.1em]
    $\Rightarrow$ Chỉ cần kiểm thử \textbf{một đại diện} từ mỗi lớp.
  \end{block}

  \vspace{0.15em}
  \small
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
    \textbf{Phân loại lớp:}
    \begin{itemize}
      \item \textbf{Hợp lệ}: hệ thống chấp nhận
      \item \textbf{Không hợp lệ}: hệ thống từ chối / xử lý riêng
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Lợi ích:}
    \begin{itemize}
      \item Giảm số test case
      \item Bao phủ đại diện mọi tình huống
      \item Dễ lý giải và review
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{EP --- Quy trình áp dụng}
  \begin{enumerate}
    \item \textbf{Xác định các trường đầu vào} cần kiểm thử.
    \vspace{0.2em}
    \item \textbf{Với mỗi trường}, xác định tất cả ràng buộc từ đặc tả:
      kiểu dữ liệu, phạm vi (min/max), định dạng, bắt buộc/tùy chọn.
    \vspace{0.2em}
    \item \textbf{Phân lớp} thành lớp hợp lệ và không hợp lệ theo từng ràng buộc.
    \vspace{0.2em}
    \item \textbf{Chọn một đại diện} từ mỗi lớp.
    \vspace{0.2em}
    \item \textbf{Kết hợp} đại diện từ nhiều trường thành test case hoàn chỉnh.
  \end{enumerate}

  \vspace{0.3em}
  \begin{exampleblock}{Áp dụng ngay trên DigiShield}
    Ba ví dụ: (a) nhãn tuổi báo cáo \texttt{ageLabel}, (b) tham số
    \texttt{size} của \texttt{GET /interventions}, (c) trường
    \texttt{action} của \texttt{POST /sim/events}.
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile,shrink=8]{Ví dụ (a): nhãn tuổi báo cáo phishing}
  \begin{block}{Đặc tả (từ javadoc \texttt{ReportingServiceImpl})}
    Mỗi báo cáo trong SOC Inbox hiển thị \textbf{nhãn tuổi} gọn:
    dưới 60 phút $\to$ \texttt{"Np"} (phút); từ 1 giờ đến dưới 24 giờ
    $\to$ \texttt{"Nh"}; từ 24 giờ trở lên $\to$ \texttt{"Nd"}.
  \end{block}

\begin{lstlisting}[language=java]
// ReportingServiceImpl.java (module reporting)
private String ageLabel(Instant reportedAt, Instant now) {
    if (reportedAt == null) return null;
    Duration d = Duration.between(reportedAt, now);
    long mins = Math.max(0, d.toMinutes());
    if (mins < 60) return mins + "p";  // duoi 1 gio
    long hours = d.toHours();
    if (hours < 24) return hours + "h"; // duoi 1 ngay
    return d.toDays() + "d";            // tu 1 ngay
}
\end{lstlisting}

  {\small Tester hộp đen \textbf{không nhìn} code này --- chỉ nhìn đặc tả.
  Ta chiếu code để đối chiếu sau khi phân lớp.}
\end{frame}

\begin{frame}{Decision Table Testing --- Nguyên lý}
  \begin{block}{Khi nào dùng?}
    Khi kết quả đầu ra phụ thuộc vào \textbf{tổ hợp} của nhiều điều kiện
    đầu vào. Một điều kiện đơn lẻ không đủ để xác định hành vi hệ thống.
  \end{block}

  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \textbf{Cấu trúc một Decision Table:}
    \begin{itemize}
      \item \textbf{Conditions} (hàng trên): điều kiện đầu vào
      \item \textbf{Actions} (hàng dưới): hành động / kết quả
      \item \textbf{Rules} (cột): mỗi cột một tổ hợp điều kiện
      \item $n$ điều kiện Boolean $\Rightarrow$ tối đa $2^n$ rules
    \end{itemize}

    \column{0.42\textwidth}
    \begin{block}{Ký hiệu}
      {\small
      \begin{tabular}{ll}
        \texttt{T} & Điều kiện đúng \\
        \texttt{F} & Điều kiện sai \\
        \texttt{-} & Không quan trọng \\

\section{Error Guessing \& Exploratory Testing}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 42
%---------------------------------------------------------------------
\begin{frame}[shrink=6]{Error Guessing --- đoán lỗi bằng kinh nghiệm}
  \begin{block}{Ý tưởng}
    Tester giàu kinh nghiệm \textbf{đoán} nơi dễ có lỗi dựa trên các
    dạng lỗi kinh điển, rồi thiết kế test nhắm thẳng vào đó.
    Bổ trợ --- không thay thế --- các kỹ thuật có hệ thống.
  \end{block}

  \vspace{0.2em}
  \small
  \textbf{Checklist đoán lỗi áp lên DigiShield:}
  \begin{itemize}
    \item Chuỗi rỗng / null / toàn khoảng trắng: email \texttt{"~~~"}
      (3 dấu cách) qua được \texttt{isBlank()} không?
    \item Unicode, emoji, tiếng Việt có dấu trong \texttt{payload} báo cáo phishing
    \item Số âm, số 0, \texttt{Integer.MAX\_VALUE} cho \texttt{page}, \texttt{size}
    \item Trùng lặp: gửi cùng một request hai lần (POST /users cùng email
      --- DigiShield chọn upsert idempotent, có đúng ý spec?)
    \item Chuỗi rất dài (payload 1MB), ký tự đặc biệt SQL/HTML
      (\texttt{' OR 1=1}, \texttt{<script>})
    \item Sai kiểu JSON: \texttt{"size": "abc"}, thiếu dấu ngoặc
  \end{itemize}

  \vspace{0.2em}
  {\small Mẹo: giữ một \textbf{fault catalogue} của nhóm --- mỗi defect tìm được
  thêm một dòng, dùng lại cho module sau.}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 43

\begin{frame}{Exploratory Testing --- kiểm thử khám phá}
  \begin{block}{Ý tưởng}
    Học --- thiết kế --- thực thi --- đánh giá diễn ra \textbf{đồng thời}.
    Không có kịch bản viết trước; tester vừa dùng hệ thống vừa đào sâu
    theo những gì quan sát được.
  \end{block}

  \vspace{0.3em}
  \begin{columns}[T]
    \column{0.5\textwidth}
    {\small\textbf{Session-based:} mỗi phiên 60--90 phút có:
    \begin{itemize}
      \item \textbf{Charter}: mục tiêu khám phá
      \item \textbf{Time-box}: giới hạn thời gian
      \item \textbf{Ghi chú}: bug, câu hỏi, ý tưởng test mới
    \end{itemize}}
    \column{0.46\textwidth}
    \begin{exampleblock}{Charter mẫu cho DigiShield}
      \small ``Khám phá SOC Inbox (\texttt{/soc/inbox}) với vai ANALYST:
      tìm cách làm nhãn tuổi, bộ lọc trạng thái và nút triage
      hiển thị sai hoặc mâu thuẫn nhau.''
    \end{exampleblock}
  \end{columns}

  \vspace{0.3em}
  {\small Kết quả phiên khám phá thường \textbf{quay lại bổ sung} cho các kỹ
  thuật hệ thống: một quan sát lạ $\to$ một lớp tương đương mới hoặc một rule mới.}
\end{frame}

\begin{frame}[shrink=6]{Bài tập 2 (nhóm): Decision Table cho \texttt{moderate()}}
  \begin{block}{Đặc tả \texttt{POST /ai/moderate} (role ANALYST)}
    Kiểm duyệt nội dung mẫu phishing trước khi dùng cho đào tạo:
    không chứa từ không an toàn $\to$ \texttt{"pass"};
    chứa \textbf{đúng 1} từ $\to$ \texttt{"flag"};
    chứa \textbf{từ 2 từ trở lên} $\to$ \texttt{"block"}.
    (Danh sách từ: malware, ransomware, exploit, ddos, \ldots)
  \end{block}

  \vspace{0.2em}
  \small
  \textbf{Yêu cầu:}
  \begin{enumerate}
    \item Đặt 2 điều kiện: C1 = ``có $\geq 1$ từ không an toàn'',
      C2 = ``có $\geq 2$ từ không an toàn''. Lập bảng đầy đủ $2^2 = 4$ rules.
    \item Phát hiện rule \textbf{bất khả thi} (infeasible): C1=F, C2=T
      --- vì sao không thể xảy ra? Xử lý thế nào trong bảng?
    \item Mỗi rule khả thi $\to$ một test case với \texttt{content} cụ thể
      (tự chọn từ trong danh sách).
    \item Bonus (BVA trên số đếm): 0, 1, 2, 3 từ --- biên nằm ở đâu?
  \end{enumerate}
\end{frame}

\begin{frame}{Câu hỏi ôn tập}
  \begin{enumerate}
    \item EP và BVA khác nhau thế nào? Với \texttt{size} 1..100, liệt kê đầy đủ
      các điểm BVA 2-value.
    \item DigiShield ``clamp'' \texttt{size} thay vì trả 400. Nêu một lập luận
      ủng hộ và một lập luận phản đối; tester nên làm gì?
    \item Bảng quyết định đầy đủ của \texttt{evaluate()} có mấy rule?
      Vì sao rule (onCall=T, newPayee=T, watchlistHit=F) đáng đưa ra
      thảo luận với product owner?
    \item Vì sao phải test cả \textbf{invalid transitions}? Cho ví dụ một
      invalid transition của \texttt{PhishingReport} mà code hiện tại
      \textbf{không} chặn.
    \item Use Case Testing phát hiện loại lỗi gì mà EP/BVA bỏ sót?
      Minh họa bằng pipeline triage $\to$ risk $\to$ enroll.
    \item Test hộp đen tìm ra lỗi ``email rỗng trả 500'': đây là lỗi của code
      hay của hợp đồng API? Ai cần sửa gì?
  \end{enumerate}
\end{frame}
