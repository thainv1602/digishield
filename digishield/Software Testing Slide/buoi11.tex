\section{Ba loại wait trong Selenium}
%=====================================================================

%---------------------------------------------------------------------
\begin{frame}[fragile]{Cách làm sai: \texttt{Thread.sleep}}
\begin{lstlisting}[language=Java]
loginButton.click();
Thread.sleep(3000);   // TUYET DOI TRANH!
assertTrue(driver.getCurrentUrl()
                 .contains("/dashboard"));
\end{lstlisting}

  \begin{alertblock}{Vì sao TUYỆT ĐỐI tránh \texttt{Thread.sleep}?}
    \begin{itemize}
      \item \textbf{Chậm}: luôn đợi đủ 3s dù trang xong sau 0.7s
            --- 100 test $\times$ 3s = 5 phút lãng phí
      \item \textbf{Vẫn flaky}: máy CI chậm, trang mất 3.5s
            $\Rightarrow$ vẫn fail
      \item Không diễn đạt \textbf{ý định}: đợi cái gì? vì sao 3s?
      \item Cộng dồn: mỗi lần fail lại tăng số giây $\Rightarrow$
            suite E2E phình to vô hạn
    \end{itemize}
  \end{alertblock}

  \vspace{-0.45em}
  \begin{block}{Nguyên tắc}
    Chờ theo \textbf{điều kiện} (condition-based), không chờ theo
    \textbf{thời gian cố định} (time-based).
  \end{block}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Loại 1: Implicit wait --- đặt một lần cho toàn driver (1/2)}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\scriptsize]
WebDriver driver = new ChromeDriver();
// Dat MOT lan, ap dung cho MOI findElement
driver.manage().timeouts()
      .implicitlyWait(Duration.ofSeconds(5));
// Tu day: findElement se poll toi da 5s
// truoc khi nem NoSuchElementException
WebElement email =
    driver.findElement(By.id("login-email"));
\end{lstlisting}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Loại 1: Implicit wait --- đặt một lần cho toàn driver (2/2)}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{exampleblock}{Ưu điểm}
      \begin{itemize}
        \item Cấu hình 1 dòng duy nhất
        \item Giảm fail do element chưa xuất hiện
      \end{itemize}
    \end{exampleblock}
    \column{0.52\textwidth}
    \begin{alertblock}{Nhược điểm}
      \begin{itemize}
        \item Chỉ chờ element \textbf{tồn tại trong DOM},
              không chờ visible/clickable/URL đổi
        \item Áp dụng mù quáng cho mọi lệnh tìm
        \item Làm test tiêu cực (element phải biến mất) chậm hẳn
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Loại 2: Explicit wait --- \texttt{WebDriverWait} (khuyến nghị)}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\scriptsize]
import org.openqa.selenium.support.ui.WebDriverWait;
import org.openqa.selenium.support.ui.ExpectedConditions;

// Cho DEN KHI dieu kien dung, toi da 5 giay
WebDriverWait wait =
    new WebDriverWait(driver, Duration.ofSeconds(5));

wait.until(ExpectedConditions
    .urlContains("/dashboard"));

WebElement input = wait.until(ExpectedConditions
    .visibilityOfElementLocated(
        By.id("login-email")));
\end{lstlisting}

  \begin{block}{Đặc điểm}
    \begin{itemize}
      \item Chờ \textbf{đúng điều kiện cần} tại \textbf{đúng chỗ cần}
      \item Poll mỗi 500ms (mặc định); điều kiện đúng là đi tiếp ngay
      \item Hết timeout $\Rightarrow$ \texttt{TimeoutException} với
            thông điệp rõ ràng
    \end{itemize}
  \end{block}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Demo: sửa test login buổi 10 bằng explicit wait (1/2)}
  \begin{alertblock}{Trước (buổi 10) --- fail vì \texttt{setTimeout} 600ms}
\begin{lstlisting}[language=Java]
driver.findElement(By.cssSelector(
    "button[type='submit']")).click();
// Assertion chay ngay lap tuc -> URL van /login

%---------------------------------------------------------------------
\begin{frame}[fragile]{Demo: sửa test login buổi 10 bằng explicit wait (2/2)}
  \begin{exampleblock}{Sau (buổi 11) --- chờ điều kiện URL đổi}
\begin{lstlisting}[language=Java]
driver.findElement(By.cssSelector(
    "button[type='submit']")).click();
// Cho toi da 5s cho den khi URL chua /dashboard
new WebDriverWait(driver, Duration.ofSeconds(5))
    .until(ExpectedConditions
        .urlContains("/dashboard"));   // PASS on dinh
\end{lstlisting}
  \end{exampleblock}

  \begin{itemize}
    \item 600ms trễ + thời gian render $<$ 5s timeout $\Rightarrow$
          test pass, và chỉ chờ \textbf{đúng bằng thời gian thực tế}
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Loại 3: Fluent wait --- tùy biến polling và ngoại lệ}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\scriptsize]
Wait<WebDriver> wait =
  new FluentWait<>(driver)
    .withTimeout(Duration.ofSeconds(10))
    .pollingEvery(Duration.ofMillis(200))
    .ignoring(NoSuchElementException.class)
    .withMessage("Watchlist row not visible");
WebElement row = wait.until(d ->
    d.findElement(By.xpath(
        "//div[text()='0909123456']")));
\end{lstlisting}

  \begin{block}{Khi nào dùng fluent wait?}
    \begin{itemize}
      \item Cần chu kỳ poll khác 500ms (UI cập nhật rất nhanh/chậm)
      \item Cần \texttt{ignoring} thêm ngoại lệ
            (vd \texttt{StaleElementReferenceException})
      \item Điều kiện lambda tùy biến không có trong
            \texttt{ExpectedConditions}
    \end{itemize}
  \end{block}
  \begin{itemize}
    \item \texttt{WebDriverWait} chính là \texttt{FluentWait} cấu hình sẵn
  \end{itemize}
\end{frame}

\begin{frame}{So sánh các cơ chế chờ}
  \begin{center}
  \footnotesize
  \begin{tabular}{p{3.4cm}p{3.1cm}p{3.0cm}p{2.6cm}}
    \toprule
    \textbf{Tiêu chí} & \textbf{Thread.sleep} & \textbf{Implicit}
      & \textbf{Explicit/Fluent} \\
    \midrule
    Chờ theo        & thời gian cố định & element có trong DOM
      & \textbf{điều kiện bất kỳ} \\
    Phạm vi         & tại chỗ           & toàn driver
      & tại chỗ, có chủ đích \\
    Tốc độ          & luôn chờ đủ       & dừng sớm khi thấy
      & dừng sớm khi đúng \\
    URL/visible/clickable & không       & không
      & \textbf{có} \\
    Độ ổn định      & kém               & trung bình
      & \textbf{cao} \\
    Khuyến nghị     & \textbf{cấm}      & hạn chế
      & \textbf{mặc định dùng} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \begin{alertblock}{Không trộn implicit và explicit wait}
    Trộn hai loại làm thời gian chờ thực tế \textbf{khó dự đoán}
    (có thể cộng dồn bất thường). Quy ước của lớp: implicit = 0,
    mọi chỗ cần chờ dùng \texttt{WebDriverWait}.
  \end{alertblock}
\end{frame}

\section{ExpectedConditions và stale element}
%=====================================================================

%---------------------------------------------------------------------
\begin{frame}{Danh mục \texttt{ExpectedConditions} hay dùng}
  \begin{center}
  \footnotesize
  \begin{tabular}{p{7.5cm}p{5.6cm}}
    \toprule
    \textbf{Điều kiện} & \textbf{Chờ đến khi\ldots} \\
    \midrule
    \texttt{visibilityOfElementLocated(by)}
      & có trong DOM \textbf{và} hiển thị \\
    \texttt{elementToBeClickable(by)}
      & hiển thị \textbf{và} enabled --- sắp \texttt{click()} \\
    \texttt{urlContains(text)}
      & URL chứa chuỗi con --- điều hướng SPA \\
    \texttt{textToBePresentInElementLocated(by, s)}
      & văn bản \texttt{s} xuất hiện trong element \\
    \texttt{presenceOfElementLocated(by)}
      & có trong DOM (chưa chắc hiển thị) \\
    \texttt{invisibilityOfElementLocated(by)}
      & element biến mất --- spinner tắt \\
    \texttt{numberOfElementsToBeMoreThan(by, n)}
      & số element $>$ n --- bảng đổ dữ liệu \\
    \bottomrule
  \end{tabular}
  \end{center}

  \begin{block}{Chọn điều kiện theo ý định kiểm chứng}
    \textbf{đọc} nội dung $\to$ visibility; \textbf{bấm} $\to$
    clickable; xác nhận \textbf{chuyển trang} $\to$ urlContains.
  \end{block}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[fragile]{Ví dụ trên DigiShield: chờ bảng watchlist đổ dữ liệu}
  Trang \texttt{/soc/watchlist} tải dữ liệu qua TanStack Query
  (hiển thị ``Đang tải watchlist\ldots'' trước khi có hàng):

\begin{lstlisting}[language=Java,basicstyle=\ttfamily\scriptsize]
WebDriverWait wait =
    new WebDriverWait(driver, Duration.ofSeconds(5));

// 1. Cho o tim kiem Quick Check hien thi
WebElement search = wait.until(
  ExpectedConditions.visibilityOfElementLocated(
    By.cssSelector("form input[type='text']")));

// 2. Cho nut "Kiem tra" bam duoc roi moi click
wait.until(ExpectedConditions.elementToBeClickable(
    By.cssSelector("form button[type='submit']")))
  .click();

// 3. Cho hang du lieu cu the xuat hien trong bang
wait.until(ExpectedConditions
  .visibilityOfElementLocated(By.xpath(

\begin{frame}[fragile]{\texttt{StaleElementReferenceException} trong SPA React}
  \begin{block}{Hiện tượng}
    Đã \texttt{findElement} thành công, lát sau gọi
    \texttt{getText()}/\texttt{click()} thì nổ
    \texttt{StaleElementReferenceException}.
  \end{block}

\begin{lstlisting}[language=Java]
WebElement row = driver.findElement(rowLocator);
// ... React re-render bang (data moi ve) ...
row.getText();   // StaleElementReferenceException!
\end{lstlisting}

  \begin{alertblock}{Nguyên nhân --- đặc thù SPA React}
    \begin{itemize}
      \item \texttt{WebElement} là \textbf{con trỏ tới node DOM cụ thể}
      \item React re-render (state đổi, query refetch) $\Rightarrow$
            node cũ bị \textbf{gỡ khỏi DOM}, thay bằng node mới
            trông y hệt
      \item Con trỏ cũ thành ``mồ côi'' $\Rightarrow$ stale
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{Page Object Model --- nguyên tắc}
  \vspace*{-0.4em}
  \begin{block}{Ba nguyên tắc cốt lõi}

\begin{frame}[fragile]{\texttt{PageFactory} và \texttt{@FindBy} --- giới thiệu nhanh}

public class LoginPageF {
  @FindBy(id = "login-email")

\begin{frame}{\texttt{@FindBy} so với \texttt{By} thủ công}
  \begin{center}
  \footnotesize
  \begin{tabular}{p{3.4cm}p{4.0cm}p{4.2cm}}
    \toprule

      & \textbf{By + WebDriverWait} \\

      "http://localhost:8080/api/v1";
  static final String TENANT =
      "11111111-1111-1111-1111-111111111111";

      HttpClient.newHttpClient();

      .uri(URI.create(API + path))
      .header("Content-Type", "application/json")
      // dev: DevSecurityConfig permitAll

        HttpResponse.BodyHandlers.ofString());

public class ScreenshotOnFailure implements TestWatcher {
  @Override
  public void testFailed(ExtensionContext ctx, Throwable cause) {
    WebDriver driver = DriverFactory.current();
    if (driver == null) return;
    try {
      File shot = ((TakesScreenshot) driver).getScreenshotAs(OutputType.FILE);
      Path dir = Path.of("build", "screenshots");
      Files.createDirectories(dir);
      String name = ctx.getDisplayName()
          .replaceAll("[^a-zA-Z0-9_-]", "_");
      Files.copy(shot.toPath(), dir.resolve(name + ".png"),
          StandardCopyOption.REPLACE_EXISTING);
    } catch (IOException e) {
      System.err.println("No screenshot: " + e);
    }
  }
}
