\section{Equivalence Partitioning}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 7
%---------------------------------------------------------------------
\begin{frame}[shrink=10]{Equivalence Partitioning --- Nguyên lý}
  \begin{block}{Định nghĩa}
    EP chia miền giá trị đầu vào thành các \textbf{lớp tương đương}
    (equivalence class) sao cho các giá trị trong cùng một lớp
    được hệ thống \textbf{xử lý theo cùng một cách} \cite{istqb-ctfl,myers-testing}.
  \end{block}

  \vspace{0.2em}
  \begin{block}{Nguyên tắc chọn đại diện}
    Nếu một giá trị trong lớp phát hiện lỗi, thì các giá trị khác
    trong cùng lớp cũng sẽ phát hiện lỗi đó.\\[0.1em]
    $\Rightarrow$ Chỉ cần kiểm thử \textbf{một đại diện} từ mỗi lớp.
  \end{block}

  \vspace{0.15em}
  \small
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
    \textbf{Phân loại lớp:}
    \begin{itemize}
      \item \textbf{Hợp lệ}: hệ thống chấp nhận
      \item \textbf{Không hợp lệ}: hệ thống từ chối
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Lợi ích:}
    \begin{itemize}
      \item Giảm số test case
      \item Bao phủ đại diện mọi tình huống
      \item Dễ lý giải và review
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{EP --- Quy trình áp dụng}
  \begin{enumerate}

        \item Định dạng (regex, pattern)
        \item Bắt buộc / tùy chọn
      \end{itemize}

    \vspace{0.3em}
    \item \textbf{Phân lớp} thành lớp hợp lệ và không hợp lệ theo từng ràng buộc.

    \vspace{0.3em}
    \item \textbf{Kết hợp} đại diện từ nhiều trường để tạo test case hoàn chỉnh.

  \begin{alertblock}{Lưu ý khi kết hợp nhiều trường}
    Để test lớp không hợp lệ của một trường, hãy giữ các trường khác
    ở giá trị \textbf{hợp lệ}. Nếu không, không rõ lỗi do trường nào gây ra.
  \end{alertblock}
\end{frame}

\section{Boundary Value Analysis}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 11
%---------------------------------------------------------------------
\begin{frame}{Boundary Value Analysis --- Nguyên lý}
  \begin{block}{Động lực}
    Kinh nghiệm cho thấy lỗi \textbf{thường xuất hiện tại vùng biên}
    giữa các lớp tương đương hơn là ở giữa lớp.
    Lập trình viên dễ viết sai \texttt{<} thành \texttt{<=},
    hay \texttt{>} thành \texttt{>=}.
  \end{block}

  \vspace{0.5em}
  \begin{block}{BVA 2-value (ISTQB Foundation)}
    Với mỗi biên, kiểm thử \textbf{hai} giá trị:
    giá trị ngay tại biên (\textit{on-point}) và
    giá trị sát biên phía đối diện (\textit{off-point}).
  \end{block}

  \begin{block}{BVA 3-value (ISTQB Advanced)}
    Kiểm thử \textbf{ba} giá trị cho mỗi biên:
    \textit{below-point}, \textit{on-point}, \textit{above-point}.
  \end{block}
\end{frame}

\begin{frame}{BVA --- Kết hợp với EP}

\begin{frame}{Decision Table Testing --- Nguyên lý}
  \begin{block}{Khi nào dùng?}
    Khi kết quả đầu ra phụ thuộc vào \textbf{tổ hợp} của nhiều điều kiện
    đầu vào. Một điều kiện đơn lẻ không đủ để xác định hành vi hệ thống.
  \end{block}

  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \textbf{Cấu trúc một Decision Table:}
    \begin{itemize}
      \item \textbf{Conditions} (hàng trên): các điều kiện đầu vào
      \item \textbf{Actions} (hàng dưới): các hành động/kết quả
      \item \textbf{Rules} (cột): mỗi cột là một tổ hợp điều kiện
      \item Với $n$ điều kiện Boolean: tối đa $2^n$ rules
    \end{itemize}

    \column{0.42\textwidth}
    \begin{block}{Ký hiệu}
      {\small
      \begin{tabular}{ll}
        \texttt{T} & Điều kiện đúng \\
        \texttt{F} & Điều kiện sai \\
        \texttt{-} & Không quan trọng \\
        $\checkmark$ & Hành động xảy ra \\
        (trống) & Không xảy ra \\
      \end{tabular}
      }
    \end{block}
  \end{columns}

  \vspace{-0.2em}
  {\scriptsize Tham khảo: \cite{istqb-ctfl,myers-testing}.}
\end{frame}

      Mỗi \textbf{chuyển tiếp hợp lệ} được thực thi ít nhất một lần.\\

      Mọi \textbf{cặp} chuyển tiếp liên tiếp được bao phủ.\\

      \end{enumerate}

      \begin{enumerate}
        \setcounter{enumi}{5}
        \item SHIPPED $\to$ cancel $\to$ Error
        \item DELIVERED $\to$ confirm $\to$ Error
      \end{enumerate}
