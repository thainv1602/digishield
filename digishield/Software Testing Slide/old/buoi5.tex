        \item Công cụ: Postman, Newman
      \end{itemize}
    \end{block}

\section{REST API Fundamentals}
%=====================================================================

%--- SLIDE 4 ----------------------------------------------------------
\begin{frame}{HTTP Request --- Cấu trúc\sref{rfc9110}}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \begin{block}{Thành phần của HTTP Request}
      \begin{itemize}
        \item \textbf{Method}: GET, POST, PUT, PATCH, DELETE
        \item \textbf{URL}: scheme + host + path + query string
        \item \textbf{Headers}: metadata như Content-Type, Authorization
        \item \textbf{Body}: dữ liệu gửi đi, thường là JSON
      \end{itemize}
    \end{block}

    \column{0.46\textwidth}
    \begin{block}{HTTP Methods trong hệ thống thực hành}
      \scriptsize
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{lll}
        \toprule
        \textbf{Method} & \textbf{Endpoint} & \textbf{Mục đích} \\
        \midrule
        GET    & /api/product   & Lấy danh sách \\
        POST   & /api/product   & Tạo sản phẩm \\
        POST   & /api/order     & Đặt hàng \\
        GET    & /api/inventory & Check kho \\
        \bottomrule
      \end{tabular}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{HTTP Request --- Ví dụ đặt hàng\sref{rfc9110}}
  \begin{exampleblock}{Request \texttt{POST /api/order}}
\begin{lstlisting}[style=json,language={}]
POST http://localhost:8080/api/order
Content-Type: application/json
Authorization: Bearer {token}

{
  "skuCode": "PROD-001",
  "quantity": 3
}
\end{lstlisting}
  \end{exampleblock}

  \begin{block}{Điểm cần kiểm thử}
    Kiểm tra đủ method, URL, headers bắt buộc và body JSON đúng schema.
  \end{block}
\end{frame}

%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}{HTTP Status Codes --- 2xx và 4xx\sref{rfc9110}}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \begin{block}{2xx --- Thành công}
      \small
      \begin{tabular}{p{2.1cm}p{3.25cm}}
        \texttt{200 OK} & GET thành công \\
        \texttt{201 Created} & POST tạo tài nguyên \\
        \texttt{204 No Content} & DELETE thành công \\
      \end{tabular}
    \end{block}

    \column{0.46\textwidth}
    \begin{block}{4xx --- Lỗi phía client}
      \small
      \begin{tabular}{p{2.35cm}p{3.0cm}}
        \texttt{400 Bad Request} & Input không hợp lệ \\
        \texttt{401 Unauthorized} & Chưa xác thực \\
        \texttt{403 Forbidden} & Không có quyền \\
        \texttt{404 Not Found} & Không tìm thấy tài nguyên \\
        \texttt{422 Invalid} & Lỗi nghiệp vụ \\
        \texttt{429 Too Many Req} & Rate limit \\
      \end{tabular}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{HTTP Status Codes --- 5xx và nguyên tắc test\sref{rfc9110}}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{5xx --- Lỗi phía server}
      \small
      \begin{tabular}{p{2.55cm}p{2.8cm}}
        \texttt{500 Internal Error} & Lỗi server không xác định \\
        \texttt{503 Unavailable} & Service không phản hồi \\
        \texttt{504 Gateway Timeout} & Timeout qua gateway \\
      \end{tabular}
    \end{block}

    \column{0.48\textwidth}
    \begin{alertblock}{Nguyên tắc kiểm thử}
      Mỗi API endpoint cần được test với \textbf{cả hai nhóm}:
      \begin{itemize}
        \item Trường hợp thành công: 2xx.
        \item Trường hợp lỗi có thể xảy ra: 4xx/5xx.
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}{REST API --- Kiểm thử hộp đen ở tầng HTTP\sref{istqb}}
  \begin{block}{API Test là Black-box Testing ở tầng tích hợp}
    Không quan tâm implementation bên trong. Chỉ quan tâm:
    request hợp lệ $\to$ response đúng đặc tả không?
  \end{block}

  \vspace{0.2em}
  \begin{center}
  \scriptsize
  \setlength{\tabcolsep}{3pt}
  \begin{tabular}{p{2.4cm}p{2.1cm}p{2.7cm}p{3.2cm}}
    \toprule
    \textbf{Khía cạnh} & \textbf{Kỹ thuật} & \textbf{Ví dụ} & \textbf{Assertion} \\
    \midrule
    Status code & EP / BVA & 201, 400, 404 & \texttt{status === 201} \\
    Schema & Cấu trúc & Có field \texttt{id} & \texttt{has property "id"} \\
    Giá trị & Kiểm tra miền & \texttt{price > 0} & \texttt{price > 0} \\
    Thời gian & Hiệu năng & $< 500$ms & \texttt{responseTime < 500} \\
    Headers & Metadata & Content-Type & \texttt{header === "application/json"} \\
    Error body & Thông báo lỗi & Message rõ & \texttt{body.message exists} \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\section{Postman --- Tổng quan}

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}{Postman --- Giao diện và các thành phần chính\sref{postman}}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Các thành phần cốt lõi}
      \begin{itemize}
        \item \textbf{Workspace}: không gian làm việc (cá nhân / nhóm)
        \item \textbf{Collection}: nhóm các request có liên quan
        \item \textbf{Folder}: phân cấp trong Collection
        \item \textbf{Request}: một HTTP call cụ thể
        \item \textbf{Environment}: tập biến theo môi trường
        \item \textbf{Variable}: giá trị có thể thay đổi theo context
        \item \textbf{Scripts}: JavaScript chạy trước/sau request
        \item \textbf{Tests}: assertions chạy sau response
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Luồng làm việc cơ bản}
      \begin{enumerate}
        \item Tạo Collection
        \item Tạo Environment (dev/staging/prod)
        \item Tạo Request trong Collection
        \item Viết Pre-request Script (nếu cần)
        \item Gửi request
        \item Viết Tests assertions
        \item Chạy Collection Runner
        \item Export để dùng với Newman
      \end{enumerate}
    \end{block}
  \end{columns}
\end{frame}

%--- SLIDE 8 ----------------------------------------------------------
\begin{frame}[fragile]{Postman --- Cấu trúc một Request\sref{postman}}
  \begin{columns}
    \column{0.52\textwidth}
    \textbf{Tab \textit{Params}}: query string parameters\\
    \textbf{Tab \textit{Headers}}: HTTP headers\\
    \textbf{Tab \textit{Body}}: request body (raw JSON)\\
    \textbf{Tab \textit{Pre-request Script}}: JS trước khi gửi\\
    \textbf{Tab \textit{Tests}}: JS sau khi nhận response\\

    \vspace{0.5em}
    \begin{exampleblock}{Request \texttt{POST /api/order}}
\begin{lstlisting}[style=json,language={}]
URL: {{base_url}}/api/order

Body (raw JSON):
{
  "skuCode":  "{{sku_code}}",
  "quantity": {{quantity}}
}
\end{lstlisting}
    \end{exampleblock}

    \column{0.46\textwidth}
    \begin{block}{Cú pháp biến trong Postman}
      \texttt{\{\{tên\_biến\}\}} --- biến được thay thế
      tự động khi gửi request.

      \vspace{0.3em}
      Nguồn biến (ưu tiên từ cao đến thấp):
      \begin{enumerate}
        \item Data file (CSV/JSON)
        \item Environment variable
        \item Collection variable
        \item Global variable
      \end{enumerate}

      \vspace{0.3em}
      \textbf{Ví dụ:}\\
      \texttt{\{\{base\_url\}\}} $\to$ \texttt{http://localhost:8080}\\
      \texttt{\{\{product\_id\}\}} $\to$ \texttt{64abc...}
    \end{block}
  \end{columns}
\end{frame}

\section{Scripts: Pre-request và Tests}
%=====================================================================

%--- SLIDE 12 ----------------------------------------------------------
\begin{frame}{Pre-request Script --- Mục đích}
  \begin{block}{Dùng để làm gì?}
    \begin{itemize}
      \item Tạo dữ liệu ngẫu nhiên: UUID, timestamp, mã sản phẩm.
      \item Tính toán giá trị động: hash, signature, token.
      \item Lấy token xác thực khi không dùng Keycloak GUI.
      \item Set biến phụ thuộc vào logic trước khi request chạy.
    \end{itemize}
  \end{block}

  \begin{exampleblock}{Collection-level Pre-request Script}
    Viết script ở cấp \textbf{Collection} thay vì từng request để tránh lặp.
    Script này chạy trước \textit{mọi} request trong Collection.
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Pre-request Script --- Ví dụ code}
  \begin{block}{Tạo dữ liệu động cho \texttt{POST /api/product}}
\begin{lstlisting}[language=Java,style=nonumber]
// Tao ten san pham ngau nhien de tranh duplicate
const randomSuffix = Math.random().toString(36).substring(2, 8);
pm.environment.set("product_name", `TestProduct-${randomSuffix}`);

// Tao UUID cho orderNumber kiem thu (neu can)
const uuid = require('uuid');
pm.environment.set("test_uuid", uuid.v4());

// Lay unix timestamp hien tai
pm.environment.set("timestamp", Date.now());
\end{lstlisting}
  \end{block}
\end{frame}

%--- SLIDE 13 ----------------------------------------------------------
\begin{frame}[fragile]{Tests Script --- Cấu trúc assertion}
  \begin{block}{Cấu trúc cơ bản}
\begin{lstlisting}[language=Java,style=nonumber]
pm.test("Tên mô tả assertion", function() {
    // Assertion expression - throw neu fail
    pm.expect(giaTri).to.equal(ketQuaMongDoi);
});
\end{lstlisting}
  \end{block}

  \begin{block}{Snippet nhanh trong Postman}
    \begin{itemize}
      \item \textit{Status code: Code is 200}
      \item \textit{Response body: JSON value check}
      \item \textit{Response time is less than 200ms}
    \end{itemize}
    Bấm vào snippet để chèn code mẫu vào Tests Script.
  \end{block}
\end{frame}

\begin{frame}{Tests Script --- Các object thường dùng}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Response}
      \begin{itemize}
        \item \texttt{pm.response}: đối tượng response
        \item \texttt{pm.response.json()}: parse body JSON
        \item \texttt{pm.response.code}: HTTP status code

\begin{frame}[shrink=6]{Các lỗi phổ biến khi dùng Postman}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{alertblock}{Lỗi 1: Connection refused}
      \textbf{Nguyên nhân}: service chưa chạy hoặc sai port.\\
      \textbf{Xử lý}: kiểm tra \texttt{docker ps}, xác nhận port đúng.
    \end{alertblock}

    \begin{alertblock}{Lỗi 2: Biến \texttt{\{\{base\_url\}\}} không được thay thế}
      \textbf{Nguyên nhân}: chưa chọn Environment hoặc biến chưa được set.\\
      \textbf{Xử lý}: click dropdown chọn Environment \textit{dev-local}.
    \end{alertblock}

    \begin{alertblock}{Lỗi 3: Data chaining bị null}
      \textbf{Nguyên nhân}: request trước fail nên không \texttt{set()} được biến.\\
      \textbf{Xử lý}: sửa request trước chạy được trước.
    \end{alertblock}

    \column{0.5\textwidth}
    \begin{alertblock}{Lỗi 4: Tests Script không chạy}
      \textbf{Nguyên nhân}: lỗi JavaScript (syntax error) trong Tests Script.\\
      \textbf{Xử lý}: xem Postman Console (View $\to$ Show Postman Console).
    \end{alertblock}

    \begin{alertblock}{Lỗi 5: 401 Unauthorized}
      \textbf{Nguyên nhân}: Profile B không tắt Keycloak hoàn toàn.\\
      \textbf{Xử lý}: thêm header \texttt{Authorization: Bearer test}
      hoặc thêm \texttt{spring.security.enabled=false}
      vào \texttt{application-test.properties}.
    \end{alertblock}

    \begin{exampleblock}{Postman Console}
      View $\to$ Show Postman Console (\texttt{Ctrl+Alt+C}):\\
      xem log, lỗi JavaScript, request/response raw.
    \end{exampleblock}
  \end{columns}

\begin{frame}[shrink=8]{So sánh API Test với Unit Test}
  \begin{center}
  \small
  \begin{tabular}{p{2.5cm}p{4.1cm}p{3.9cm}}
    \toprule
    \textbf{Khía cạnh} & \textbf{Unit Test (JUnit)} & \textbf{API Test (Postman)} \\
    \midrule
    Phạm vi       & Một class/method               & Toàn endpoint từ HTTP \\
    Dependencies  & Tất cả mock                    & Thật (DB, Kafka, ...) \\
    Tốc độ        & Mili-giây                      & Mili-giây đến giây \\
    Phát hiện     & Lỗi logic nội tại              & Lỗi tích hợp, routing, serialization \\
    Khi nào chạy  & Mỗi lần build                  & Trước khi merge / deploy \\
    Ngôn ngữ      & Java                           & JavaScript (assertion) \\
    Môi trường    & Không cần Docker               & Cần Docker chạy \\
    Kết quả       & JaCoCo coverage report         & Pass/fail theo assertion \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \begin{exampleblock}{Ví dụ lỗi chỉ API Test phát hiện được}
    Unit test: \texttt{orderService.placeOrder()} trả về đúng. \\
    API test: \texttt{POST /api/order} trả về 500 vì \texttt{@RequestBody}
    thiếu annotation \texttt{@Valid} trong Controller $\to$ validation không chạy.
  \end{exampleblock}
\end{frame}

\begin{frame}{Câu hỏi ôn tập}
  \begin{enumerate}
    \item Tại sao cần API Test khi đã có Unit Test?
      Nêu 2 ví dụ lỗi chỉ API test phát hiện được.

    \item Sự khác biệt giữa Environment variable và Collection variable
      trong Postman là gì? Khi nào dùng loại nào?

    \item Pre-request Script chạy ở thời điểm nào trong vòng đời
      của một request? Nêu 2 trường hợp thực tế cần dùng.

    \item Nếu request POST /api/product fail (status 500),
      các request sau trong Collection Runner sẽ bị ảnh hưởng
      như thế nào? Làm sao để dừng sớm (\textit{bail on failure})?

    \item Sự khác biệt giữa \texttt{pm.response.to.have.status(201)}
      và \texttt{pm.expect(pm.response.code).to.equal(201)} là gì?
      Cả hai có cho kết quả như nhau không?

    \item Tại sao cần kiểm tra cả \texttt{status code} lẫn \texttt{response body}
      khi test error cases? Chỉ kiểm tra status code có đủ không?
  \end{enumerate}
