%--- SLIDE 4 ----------------------------------------------------------
\begin{frame}[shrink=10]{Kiểm thử hộp trắng là gì?}
  \begin{block}{Định nghĩa}
    Kiểm thử hộp trắng (White-box testing) thiết kế test case
    \textbf{dựa trên cấu trúc bên trong} của phần mềm:
    mã nguồn, luồng điều khiển, luồng dữ liệu.\sref{1}
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Còn gọi là:}
    \begin{itemize}
      \item Structural testing
      \item Glass-box testing
      \item Clear-box testing
      \item Code-based testing
    \end{itemize}
    \vspace{0.4em}
    \textbf{Người thực hiện:}\\
    Developer hoặc tester có kiến thức lập trình.

    \column{0.5\textwidth}
    \textbf{Mục tiêu:}
    \begin{itemize}
      \item Đảm bảo mọi câu lệnh được thực thi ít nhất một lần
      \item Đảm bảo mọi nhánh (true/false) đều được đi qua
      \item Phát hiện dead code, logic sai trong cài đặt
      \item Đo lường chất lượng bộ test bằng \textbf{coverage}
    \end{itemize}
  \end{columns}
\end{frame}

%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}[fragile,shrink=15]{Đoạn code dùng xuyên suốt buổi học}
  \begin{block}{\small \texttt{InventoryService.isInStock()} --- phiên bản giản lược}
\begin{lstlisting}[language=java]
public boolean isInStock(String skuCode, int quantity) {  // L1
    if (skuCode == null || skuCode.isEmpty()) {            // L2
        throw new IllegalArgumentException("SKU invalid"); // L3
    }                                                      // L4
    if (quantity <= 0) {                                   // L5
        throw new IllegalArgumentException("Qty invalid"); // L6
    }                                                      // L7
    Inventory inv = repository.findBySkuCode(skuCode);     // L8
    if (inv == null) {                                     // L9
        return false;                                      // L10
    }                                                      // L11
    return inv.getQuantity() >= quantity;                  // L12
}
\end{lstlisting}
  \end{block}
  \vspace{-0.3em}
  {\small Method này có \textbf{3 điểm quyết định} (L2, L5, L9)
  và \textbf{5 đường dẫn} có thể đi qua.}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}[shrink=12]{Statement Coverage (SC)}
  \begin{block}{Định nghĩa}
    \[
      SC = \frac{\text{Số câu lệnh được thực thi}}{\text{Tổng số câu lệnh thực thi được}} \times 100\%
    \]
    Mỗi câu lệnh phải được thực thi \textbf{ít nhất một lần}.\sref{1}
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Áp dụng cho \texttt{isInStock()}:}

    Để đạt 100\% SC, cần ít nhất một test case đi qua tất cả
    12 dòng.

    \vspace{0.3em}
    \begin{exampleblock}{Test case đạt 100\% SC}
      Một test case duy nhất:\\
      \texttt{skuCode="A"}, \texttt{qty=2},
      \texttt{inventory.qty=5}\\
      $\Rightarrow$ đi qua: L1, L2, L5, L8, L9, L12\\
      $\Rightarrow$ SC = 6/8 = 75\% (L3, L6, L10 chưa được đi qua)
    \end{exampleblock}

    \column{0.45\textwidth}
    \begin{alertblock}{Hạn chế của SC}
      100\% Statement Coverage \textbf{không đảm bảo} phát hiện lỗi
      trong nhánh sai (false branch).

      \vspace{0.3em}
      Ví dụ: nếu điều kiện \texttt{qty <= 0} bị viết thành
      \texttt{qty < 0}, SC vẫn 100\% nhưng không phát hiện lỗi
      với \texttt{qty = 0}.
    \end{alertblock}
  \end{columns}
\end{frame}

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}[shrink=12]{Branch Coverage (BC)}
  \begin{block}{Định nghĩa}
    \[
      BC = \frac{\text{Số nhánh được thực thi}}{\text{Tổng số nhánh}} \times 100\%
    \]
    Mỗi quyết định (if, while, for, switch, ternary) phải có
    \textbf{cả nhánh true và false} được thực thi ít nhất một lần.\sref{1}
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Các nhánh trong \texttt{isInStock()}:}
    \begin{center}
    \small
    \begin{tabular}{lll}
      \toprule
      \textbf{Điều kiện} & \textbf{True} & \textbf{False} \\
      \midrule
      L2: skuCode null/empty & L3 (throw) & L5 \\
      L5: quantity $\leq$ 0  & L6 (throw) & L8 \\
      L9: inv == null        & L10 (false) & L12 \\
      \bottomrule
    \end{tabular}
    \end{center}

    Tổng: \textbf{6 nhánh} cần bao phủ.

    \column{0.45\textwidth}
    \begin{exampleblock}{Test cases đạt 100\% BC}
      \textbf{TC-B1}: \texttt{skuCode=null} $\to$ nhánh T của L2\\[0.2em]
      \textbf{TC-B2}: \texttt{qty=0} $\to$ nhánh T của L5\\[0.2em]
      \textbf{TC-B3}: SKU không tồn tại $\to$ nhánh T của L9\\[0.2em]
      \textbf{TC-B4}: \texttt{skuCode="A"}, \texttt{qty=2},
      \texttt{inv.qty=5} $\to$ tất cả nhánh F\\
    \end{exampleblock}
  \end{columns}
\end{frame}

%--- SLIDE 8 ----------------------------------------------------------
\begin{frame}[fragile,shrink=12]{Branch Coverage --- Tại sao quan trọng hơn SC?}
  \begin{block}{BC $\geq$ SC về độ mạnh}
    100\% Branch Coverage \textbf{đảm bảo} 100\% Statement Coverage,
    nhưng không ngược lại.
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{exampleblock}{Ví dụ lỗi chỉ BC phát hiện}
\begin{lstlisting}[language=java,style=nonumber]
// Giam gia neu qty >= 10
// Code sai:
if (quantity > 10) {
    price *= 0.9;
}
\end{lstlisting}
      \begin{itemize}
        \item \texttt{qty=15}: SC đạt, nhánh F chưa test
        \item \texttt{qty=10}: BC phát hiện lỗi
      \end{itemize}
    \end{exampleblock}

    \column{0.5\textwidth}
    \begin{block}{Ngưỡng trong thực tế công nghiệp}
      \begin{tabular}{ll}
        \toprule
        \textbf{Ngành} & \textbf{Ngưỡng BC} \\
        \midrule
        Safety-critical & 100\% \\
        Fintech, Banking & 80--90\% \\
        E-commerce       & 70--80\% \\
        SonarQube default & 80\% \\
        Môn học này       & \textbf{70\%} \\
        \bottomrule
      \end{tabular}
    \end{block}
  \end{columns}
\end{frame}

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}[shrink=22]{Condition Coverage (CC)}
  \begin{block}{Định nghĩa}
    Mỗi \textbf{điều kiện nguyên tố} (atomic condition) trong biểu thức
    Boolean phức tạp phải nhận cả giá trị \texttt{true} và \texttt{false}.\sref{1}
  \end{block}

  \vspace{0.1em}
  \begin{exampleblock}{Ví dụ: L2 trong \texttt{isInStock()}}
    Điều kiện phức hợp: \texttt{skuCode == null || skuCode.isEmpty()}

    Gồm \textbf{2 điều kiện nguyên tố}:
    \begin{itemize}
      \item $C_1$: \texttt{skuCode == null}
      \item $C_2$: \texttt{skuCode.isEmpty()}
    \end{itemize}

    \begin{center}
    \small
    \begin{tabular}{cccl}
      \toprule
      $C_1$ (null) & $C_2$ (empty) & \textbf{Kết quả} & \textbf{Test case} \\
      \midrule
      T & -- & T (short-circuit) & \texttt{skuCode = null} \\
      F & T  & T & \texttt{skuCode = ""} \\
      F & F  & F & \texttt{skuCode = "PROD-A"} \\
      \bottomrule
    \end{tabular}
    \end{center}
  \end{exampleblock}

  {\small\textbf{Lưu ý:} Branch Coverage chưa đủ với điều kiện phức hợp; hệ an toàn cao thường cần MC/DC.}
\end{frame}

%--- SLIDE 10 ----------------------------------------------------------
\begin{frame}[shrink=12]{Path Coverage}
  \begin{block}{Định nghĩa}
    Mỗi \textbf{đường dẫn độc lập} từ đầu vào đến đầu ra của method
    phải được thực thi ít nhất một lần.
  \end{block}

  \vspace{0.3em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Các path trong \texttt{isInStock()}:}
    \begin{enumerate}
      \item L2(T) $\to$ L3 (throw SKU invalid)
      \item L2(F) $\to$ L5(T) $\to$ L6 (throw Qty invalid)
      \item L2(F) $\to$ L5(F) $\to$ L8 $\to$ L9(T) $\to$ L10 (return false)
      \item L2(F) $\to$ L5(F) $\to$ L8 $\to$ L9(F) $\to$ L12 (return false/true)
    \end{enumerate}

    \column{0.5\textwidth}
    \begin{block}{Tại sao Path Coverage không thực tế?}
      \begin{itemize}
        \item Mỗi vòng lặp nhân số path lên $\times n$
        \item Vòng lặp $n$ lần: $n+1$ paths
        \item Hai vòng lặp lồng nhau: $(n+1)^2$ paths
        \item Hệ thống thực tế: hàng triệu paths
      \end{itemize}
    \end{block}
    \begin{exampleblock}{Giải pháp thực tế}
      Dùng \textbf{Cyclomatic Complexity} để xác định số path
      \textit{độc lập tuyến tính} tối thiểu cần test.
    \end{exampleblock}
  \end{columns}
\end{frame}

  \large
  \renewcommand{\arraystretch}{1.18}
  \begin{tabularx}{\textwidth}{lXXl}
    \toprule
    \textbf{Tiêu chí} & \textbf{Yêu cầu} & \textbf{Phát hiện lỗi} & \textbf{Chi phí} \\

  \end{tabularx}

  \begin{block}{Quan hệ độ mạnh}
    \Large
    \[
      \text{SC} \subset \text{BC} \subset \text{CC} \subset \text{MC/DC} \subset \text{Path}
    \]
  \end{block}
\end{frame}

%--- SLIDE 12 ----------------------------------------------------------
\begin{frame}[fragile,shrink=12]{Control Flow Graph --- Định nghĩa}
  \begin{block}{Định nghĩa}
    CFG $G = (N, E)$ là đồ thị có hướng trong đó:
    \begin{itemize}
      \item $N$: tập \textbf{node} --- mỗi node là một khối câu lệnh
            (basic block) thực thi tuần tự
      \item $E$: tập \textbf{cạnh} --- mỗi cạnh biểu diễn
            luồng điều khiển có thể xảy ra
    \end{itemize}
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.48\textwidth}
    \textbf{Các loại node đặc biệt:}
    \begin{itemize}
      \item \textbf{Entry node}: điểm vào duy nhất
      \item \textbf{Exit node}: điểm thoát (có thể nhiều)
      \item \textbf{Decision node}: node có 2 cạnh ra (if, while, for)
      \item \textbf{Junction node}: node có nhiều cạnh vào
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Cyclomatic Complexity:}
    \[
      V(G) = E - N + 2P
    \]
    $E$: số cạnh, $N$: số node, $P$: số thành phần liên thông (thường = 1).

    \vspace{0.3em}
    $V(G)$ = số đường dẫn độc lập tuyến tính tối thiểu cần test.

    \vspace{0.3em}
    $V(G)$ = số decision node + 1.
  \end{columns}
\end{frame}

%--- SLIDE 13 ----------------------------------------------------------
\begin{frame}[fragile,shrink=18]{CFG --- Vẽ cho \texttt{isInStock()}}
  \begin{columns}
    \column{0.38\textwidth}
    \begin{center}

      \draw[arr] (n2) -- node[lbl,right]{F} (n4);

    \column{0.58\textwidth}
    \textbf{Tính $V(G)$:}
    \begin{itemize}
      \item $N = 9$ node
      \item $E = 8$ cạnh
      \item $P = 1$
      \item $V(G) = 8 - 9 + 2 \times 1 = \mathbf{1}$
    \end{itemize}

    \vspace{0.3em}
    \textit{Cách nhanh hơn:} $V(G)$ = số decision node + 1 = $3 + 1 = \mathbf{4}$

    \vspace{0.5em}
    \textbf{4 đường dẫn độc lập tuyến tính:}
    \begin{enumerate}
      \item $1 \to 2(T) \to 3$
      \item $1 \to 2(F) \to 4(T) \to 5$
      \item $1 \to 2(F) \to 4(F) \to 6 \to 7(T) \to 8$
      \item $1 \to 2(F) \to 4(F) \to 6 \to 7(F) \to 9$
    \end{enumerate}

        \item \textbf{p-use}: dùng trong điều kiện rẽ nhánh

      Một cặp gồm:
      \begin{itemize}
        \item vị trí biến được định nghĩa
        \item vị trí biến được sử dụng
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

// L8: DEF(inv)
Inventory inv = repository.findBySkuCode(skuCode);

// L9: p-USE(inv)
if (inv == null) {
    return false;
}

// L12: c-USE(inv)
return inv.getQuantity() >= quantity;
\end{lstlisting}
  \end{exampleblock}

  \vspace{0.3em}
  \Large
  \begin{block}{Cặp def-use cần test}
    \(L8, L9\) và \(L8, L12\): cần test case cho cả hai cặp.
  \end{block}
\end{frame}

\begin{frame}[fragile,shrink=12]{Mockito --- \texttt{when/thenReturn} và \texttt{verify}}
\begin{lstlisting}[language=java]
@Test
void isInStock_SufficientQuantity_ShouldReturnTrue() {
    // ARRANGE: lap trinh hanh vi cua mock
    Inventory mockInv = new Inventory("PROD-A", 10);
    when(repository.findBySkuCode("PROD-A"))       // (1)
        .thenReturn(mockInv);

    // ACT
    boolean result = inventoryService.isInStock("PROD-A", 5);

    // ASSERT: kiem tra ket qua
    assertTrue(result);                             // (2)

    // VERIFY: xac minh mock da duoc goi dung cach
    verify(repository, times(1))                   // (3)
        .findBySkuCode("PROD-A");
    verify(repository, never())                    // (4)
        .save(any());
}
\end{lstlisting}
  \vspace{-0.3em}
  {\small (1) Stub: khi gọi với "PROD-A" thì trả về mockInv \quad
          (2) Assertion trên kết quả\\
          (3) Verify: method được gọi đúng 1 lần \quad
          (4) Verify: method này KHÔNG được gọi}
\end{frame}

    assertEquals("Apple laptop",             saved.getDescription());

\begin{frame}[shrink=12]{Nguyên tắc viết unit test tốt --- FIRST}
  \begin{block}{FIRST principles}
    \begin{itemize}
      \item \textbf{F}ast: chạy trong mili-giây
      \item \textbf{I}ndependent: không phụ thuộc vào thứ tự
      \item \textbf{R}epeatable: luôn cho cùng kết quả
      \item \textbf{S}elf-validating: pass/fail rõ ràng, không cần nhìn log
      \item \textbf{T}imely: viết cùng lúc với production code
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}[shrink=12]{Nguyên tắc viết unit test tốt --- AAA và tên test}
  \begin{block}{AAA pattern}
    \textbf{A}rrange --- \textbf{A}ct --- \textbf{A}ssert\\[0.3em]
    \begin{itemize}
      \item \textbf{Arrange}: chuẩn bị dữ liệu, stub mock
      \item \textbf{Act}: gọi method cần test
      \item \textbf{Assert}: kiểm tra kết quả
    \end{itemize}
    Mỗi test method nên kiểm thử \textbf{một khía cạnh} duy nhất.
  \end{block}

  \begin{alertblock}{Đặt tên test method}
    \texttt{methodName\_condition\_expectedBehavior}\\[0.2em]
    Ví dụ: \texttt{isInStock\_NullSku\_ShouldThrow}
  \end{alertblock}
\end{frame}
