\section{Newman CLI}
%=====================================================================

%--- SLIDE 4 ----------------------------------------------------------
\begin{frame}[fragile]{Newman --- Tổng quan\sref{newman}}
  \begin{block}{Newman là gì?}
    Newman là \textbf{command-line collection runner} cho Postman.
    Cho phép chạy Postman Collection từ terminal mà không cần
    giao diện đồ họa, phù hợp để tích hợp vào CI/CD pipeline.
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Collection Runner (Postman GUI) vs Newman:}
    \begin{center}
    \small
    \begin{tabular}{lcc}
      \toprule
      \textbf{Tính năng} & \textbf{Runner} & \textbf{Newman} \\
      \midrule
      Cần GUI       & Có  & Không \\
      CI/CD tích hợp & Khó & Dễ \\
      HTML report   & Không & Có (plugin) \\
      Data file CSV & Có  & Có \\
      Headless      & Không & Có \\
      Script automation & Không & Có \\
      \bottomrule
    \end{tabular}
    \end{center}

    \column{0.5\textwidth}
    \begin{block}{Cài đặt}
\begin{lstlisting}[style=bash]
# Cai Newman core
npm install -g newman

# Cai them HTML reporter
npm install -g \
  newman-reporter-htmlextra

# Kiem tra
newman --version
\end{lstlisting}
    \end{block}
  \end{columns}
\end{frame}

%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}[fragile]{Newman --- Lệnh cơ bản}
  \begin{block}{Cú pháp tổng quát}
\begin{lstlisting}[style=bash]
newman run <collection> [options]
\end{lstlisting}
  \end{block}

  \begin{block}{Chạy với Environment}
\begin{lstlisting}[style=bash]
newman run E-Commerce-API.postman_collection.json \
    --environment dev-local.postman_environment.json
\end{lstlisting}
  \end{block}

  \begin{block}{Các option quan trọng}
\begin{lstlisting}[style=bash]
newman run collection.json \
    -e environment.json          \  # environment file
    -d test-data.csv             \  # data file
    -n 3                         \  # so lan lap (iterations)
    --timeout-request 5000       \  # timeout moi request (ms)
    --delay-request 200          \  # delay giua cac request (ms)
    --bail                       \  # dung ngay khi co loi
    --reporters cli,htmlextra    \  # reporters
    --reporter-htmlextra-export report.html
\end{lstlisting}
  \end{block}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}[fragile]{Newman --- Đọc kết quả CLI}
\begin{lstlisting}[style=bash]
newman run collection.json -e env.json

  \begin{columns}

      \texttt{0}: tất cả test pass\\
      \texttt{1}: có test fail hoặc lỗi\\[0.3em]
      CI/CD pipeline dùng exit code để quyết định
      build pass hay fail.
    \end{block}

    \column{0.5\textwidth}
    \begin{alertblock}{Khi có assertion fail}
      Newman in ra rõ request nào fail,
      assertion nào sai, expected vs actual.
      Không cần mở Postman để debug.
    \end{alertblock}
  \end{columns}
\end{frame}

\section{Data-driven Testing}

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}{Data-driven Testing --- Nguyên lý\sref{postman-learning}}
  \begin{block}{Định nghĩa}
    Data-driven testing (DDT) tách biệt \textbf{logic test}
    khỏi \textbf{dữ liệu test}. Cùng một test script chạy
    nhiều lần với các bộ dữ liệu khác nhau từ file ngoài.
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Không dùng DDT --- lặp code:}
    \begin{itemize}
      \item Request ``Order qty=1'': body hardcode
      \item Request ``Order qty=500'': body hardcode
      \item Request ``Order qty=1000'': body hardcode
      \item Request ``Order qty=0'': body hardcode
      \item ... (N request cho N bộ dữ liệu)
    \end{itemize}
    $\Rightarrow$ 8 request riêng lẻ, khó bảo trì.

    \column{0.5\textwidth}
    \textbf{Dùng DDT --- một request, nhiều dữ liệu:}
    \begin{itemize}
      \item 1 request ``Order'' duy nhất
      \item File \texttt{order-data.csv}: 8 dòng
      \item Newman chạy request đó 8 lần
      \item Mỗi lần dùng một dòng dữ liệu
    \end{itemize}
    $\Rightarrow$ 1 request, dễ thêm/bớt test case.

  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{Liên hệ với Buổi 3}
    Tương tự \texttt{@ParameterizedTest} trong JUnit 5 ---
    cùng logic test, nhiều bộ dữ liệu.
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Data-driven Testing --- Assertions động}
  \begin{block}{Tests Script cho data-driven request}
\begin{lstlisting}[style=nonumber]
const expStatus = parseInt(pm.iterationData.get("expectedStatus"));
const desc      = pm.iterationData.get("description");
const body      = pm.response.json();

// Assertion 1: status code dong voi du lieu
pm.test(`[${desc}] Status code is ${expStatus}`, function() {
    pm.expect(pm.response.code).to.equal(expStatus);
});

// Assertion 2: phan ung response time
pm.test(`[${desc}] Response time < 500ms`, function() {
    pm.expect(pm.response.responseTime).to.be.below(500);
});

// Assertion 3: conditional - neu thanh cong phai co orderNumber
pm.test(`[${desc}] Success response has orderNumber`, function() {
    if (expStatus === 201) {
        pm.expect(body).to.have.property("orderNumber");
        pm.expect(body.orderNumber).to.not.be.empty;
    } else {
        pm.expect(body).to.have.property("message");
    }
});
\end{lstlisting}
  \end{block}
\end{frame}

\section{Negative Testing hệ thống}
%=====================================================================

%--- SLIDE 15 ----------------------------------------------------------
\begin{frame}{Negative Testing --- Phân loại\sref{istqb}}
  \begin{block}{Định nghĩa}
    Negative testing kiểm thử hành vi của hệ thống khi nhận
    \textbf{đầu vào không mong đợi, không hợp lệ, hoặc ngoài
    vùng thiết kế}. Mục tiêu: hệ thống phải xử lý gracefully,
    không crash, và trả về lỗi rõ ràng.
  \end{block}

  \vspace{0.4em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Phân loại negative test:}
    \begin{enumerate}
      \item \textbf{Input validation}: trường rỗng, null,
            sai kiểu, ngoài phạm vi
      \item \textbf{Business rule violation}: đặt hàng
            khi hết stock, tạo duplicate
      \item \textbf{Missing required fields}: bỏ trống
            trường bắt buộc
      \item \textbf{Wrong Content-Type}: gửi XML khi API
            yêu cầu JSON
      \item \textbf{Malformed JSON}: JSON không hợp lệ
      \item \textbf{Extra fields}: gửi thêm field không
            có trong schema
    \end{enumerate}

    \column{0.5\textwidth}
    \begin{alertblock}{Tại sao negative testing quan trọng?}
      \begin{itemize}
        \item Hacker thường tấn công qua input không hợp lệ
        \item Lỗi 500 khi input sai = thông tin nhạy cảm bị lộ
        \item API trả về lỗi mơ hồ = khó debug cho client
        \item Business rule không được enforce = dữ liệu sai
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

%--- SLIDE 16 ----------------------------------------------------------
\begin{frame}[fragile]{Negative Testing --- Input Validation}
  \begin{block}{Nhóm 1: Missing required fields}
\begin{lstlisting}[style=bash]
# Request 1: Thieu ca skuCode va quantity
POST /api/order
Body: {}
Expected: 400, message mo ta truong nao con thieu

# Request 2: Chi co skuCode, thieu quantity
POST /api/order
Body: {"skuCode": "IPHONE-15"}
Expected: 400

# Request 3: Chi co quantity, thieu skuCode
POST /api/order
Body: {"quantity": 5}
Expected: 400

%--- SLIDE 25 ----------------------------------------------------------
\begin{frame}[fragile]{GraphQL --- Tổng quan\sref{graphql}}
  \begin{block}{GraphQL là gì?}
    GraphQL là ngôn ngữ truy vấn cho API, được Facebook phát triển
    năm 2015. Thay vì nhiều endpoint như REST, GraphQL chỉ có
    \textbf{một endpoint duy nhất} và client tự chỉ định
    dữ liệu nào cần lấy.
  \end{block}

  \vspace{0.3em}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{center}
    \resizebox{\linewidth}{!}{%
    \begin{tikzpicture}[
      bx/.style={rectangle,draw=darkgray,fill=lightgray,
                 rounded corners=2pt,minimum width=2.8cm,
                 minimum height=0.65cm,font=\small,align=center},
      arr/.style={-{Stealth[length=4pt]},darkgray}
    ]
      \node[font=\small\bfseries] at (1.4,3.8) {REST};
      \node[bx] (c) at (1.4,3.0) {Client};
      \node[bx,right=0.3cm of c,minimum width=1.2cm]
            (e1) at (3.3,3.5) {/products};
      \node[bx,right=0.3cm of c,minimum width=1.2cm]
            (e2) at (3.3,3.0) {/orders};
      \node[bx,right=0.3cm of c,minimum width=1.2cm]
            (e3) at (3.3,2.5) {/inventory};
      \draw[arr] (c) -- (e1);
      \draw[arr] (c) -- (e2);
      \draw[arr] (c) -- (e3);

      \node[font=\small\bfseries] at (7,3.8) {GraphQL};
      \node[bx] (c2) at (6,3.0) {Client};
      \node[bx,fill=darkgray,text=white] (gql) at (8.5,3.0) {/graphql};
      \draw[arr] (c2) -- (gql)
        node[midway,above,font=\scriptsize]{query\{...\}};
    \end{tikzpicture}%
    }
    \end{center}

    \column{0.5\textwidth}
    \textbf{Đặc điểm:}
    \begin{itemize}
      \item \textbf{Một endpoint}: \texttt{POST /graphql}
      \item \textbf{Client chỉ định}: lấy đúng field cần
      \item \textbf{Strongly typed}: schema định nghĩa trước
      \item \textbf{Query}: đọc dữ liệu
      \item \textbf{Mutation}: ghi dữ liệu
      \item \textbf{Subscription}: real-time updates
    \end{itemize}
  \end{columns}
\end{frame}

%--- SLIDE 26 ----------------------------------------------------------
\begin{frame}[fragile]{GraphQL vs REST --- So sánh thực tế\sref{graphql}}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{REST: lấy product và orders}
\begin{lstlisting}[style=bash]
# 3 round-trips
GET /api/product/123
# Returns: id, name, desc, price,
#          category, images, ...

GET /api/orders?productId=123
# Returns: orders array

GET /api/inventory/IPHONE-15
# Returns: quantity, warehouse, ...
\end{lstlisting}
      \textit{Over-fetching}: nhận nhiều field hơn cần.\\
      \textit{Under-fetching}: phải gọi nhiều endpoint.
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{GraphQL: cùng dữ liệu, 1 request}
\begin{lstlisting}[style=bash]
# 1 request duy nhat
POST /graphql
{
  "query": "
    {
      product(id: \"123\") {
        id
        name
        price
      }
      orders(productId: \"123\") {
        orderNumber
        quantity
      }
    }
  "
}
\end{lstlisting}
      Client lấy đúng field cần, một lần.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Kiểm thử GraphQL trong Postman}
  \begin{block}{Cấu hình request GraphQL}

%--- SLIDE 28 ----------------------------------------------------------
\begin{frame}{REST vs GraphQL --- Khi nào dùng cái nào?}
  \begin{center}
  \small
  \begin{tabular}{p{3cm}p{3.5cm}p{3.5cm}}
    \toprule
    \textbf{Tiêu chí} & \textbf{REST} & \textbf{GraphQL} \\

    \bottomrule
  \end{tabular}
  \end{center}

    GraphQL \textbf{luôn trả về HTTP 200} --- kể cả khi có lỗi.

    Assertion phải kiểm tra body, không chỉ status code.
  \end{alertblock}
\end{frame}

\begin{frame}{Đánh giá chất lượng API dựa trên kết quả test}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Các chỉ số cần theo dõi}
      \begin{itemize}
        \item \textbf{Pass rate}: $\geq 95\%$ cho happy path, $100\%$ cho negative test
        \item \textbf{Response time P95}: $< 500$ms
        \item \textbf{Negative handling}: mọi bad input đều trả về 4xx (không có 500)
        \item \textbf{Error message quality}: có mô tả lỗi rõ ràng
        \item \textbf{Consistency}: cùng loại lỗi trả về cùng status code
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{exampleblock}{Defect nào cần ưu tiên sửa?}
      \textbf{Critical} (sửa ngay):
      \begin{itemize}
        \item Input không hợp lệ trả về 500
        \item SQL injection không bị chặn
      \end{itemize}
      \textbf{High}:
      \begin{itemize}
        \item Error message không mô tả lỗi
        \item Response time $> 2$s
      \end{itemize}
      \textbf{Medium}:
      \begin{itemize}
        \item Status code không đúng chuẩn (dùng 200 thay 201)
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Kỹ thuật nâng cao --- pm.sendRequest}
  \begin{block}{Gửi request phụ bên trong Pre-request Script}
    Đôi khi cần lấy dữ liệu từ request khác trước khi gửi request chính.
  \end{block}

\begin{lstlisting}[style=nonumber]
// Pre-request Script: lay danh sach product truoc
pm.sendRequest({
    url:    pm.environment.get("base_url") + "/api/product",
    method: "GET",
    header: { "Content-Type": "application/json" }
}, function(err, res) {
    if (err) {
        console.error("Pre-request failed:", err);
        return;
    }
    const products = res.json();
    if (products && products.length > 0) {
        // Lay SKU cua san pham dau tien de dung trong request chinh
        pm.environment.set("dynamic_sku", products[0].name);
        console.log("Set dynamic_sku:", products[0].name);
    }
});
\end{lstlisting}

  \begin{exampleblock}{Ứng dụng}
    Dùng \texttt{pm.sendRequest} để lấy token xác thực,
    lấy danh sách resource hiện có, hoặc kiểm tra precondition
    trước khi thực thi test chính.
  \end{exampleblock}
\end{frame}
