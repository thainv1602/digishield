\section{@Mock vs @MockBean}
%=====================================================================

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}[fragile,shrink=12]{@Mock vs @MockBean --- Phân biệt cốt lõi\sref{mockito}}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{\texttt{@Mock} (Mockito thuần)}
      \begin{itemize}
        \item Tạo mock \textbf{ngoài Spring context}
        \item Dùng với \texttt{@ExtendWith(
    MockitoExtension.class)}
        \item Không khởi động ApplicationContext
        \item Tốc độ: \textbf{mili-giây}
        \item Dùng cho: \textbf{unit test}
      \end{itemize}
\begin{lstlisting}[language=java,style=small]
@ExtendWith(
    MockitoExtension.class)
class OrderServiceTest {
  @Mock
  OrderRepository repo;
  @InjectMocks
  OrderService service;
}
\end{lstlisting}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{\texttt{@MockBean} (Spring Boot)}
      \begin{itemize}
        \item Tạo mock \textbf{và đăng ký vào ApplicationContext}
        \item Dùng với \texttt{@SpringBootTest} hoặc \texttt{@WebMvcTest}
        \item Khởi động (một phần) Spring context
        \item Tốc độ: \textbf{vài giây}
        \item Dùng cho: \textbf{integration test}
      \end{itemize}
\begin{lstlisting}[language=java,style=small]
@SpringBootTest
class OrderControllerIT {
  @MockBean
  InventoryClient client;
  @Autowired
  OrderService service;
}
\end{lstlisting}
    \end{block}
  \end{columns}
\end{frame}

%--- SLIDE 8 ----------------------------------------------------------
\begin{frame}[shrink=12]{@Mock vs @MockBean --- Khi nào dùng cái nào?}
  \begin{center}
  \scriptsize
  \begin{tabular}{p{5cm}cc}
    \toprule
    \textbf{Tình huống} & \textbf{@Mock} & \textbf{@MockBean} \\
    \midrule
    Unit test một Service class & \checkmark & \\
    Test Controller với \texttt{@WebMvcTest} & & \checkmark \\
    Test có \texttt{@SpringBootTest} & & \checkmark \\
    Test có \texttt{@DataJpaTest} & & \checkmark \\
    Cần Spring Security context & & \checkmark \\
    Không cần Spring context & \checkmark & \\
    Pipeline CI cần chạy nhanh & \checkmark & \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \begin{alertblock}{Lỗi thường gặp khi nhầm lẫn}
    Dùng \texttt{@Mock} trong class có \texttt{@SpringBootTest}:
    mock được tạo nhưng \textbf{không được inject} vào Spring bean
    $\to$ service vẫn dùng dependency thật $\to$ test fail hoặc
    kết nối database thật.
  \end{alertblock}
\end{frame}

\section{Mock Feign Client và KafkaTemplate}
%=====================================================================

%--- SLIDE 9 ----------------------------------------------------------
\begin{frame}[fragile]{Mock Feign Client --- \texttt{InventoryClient}\sref{mockito}}
  \begin{block}{Feign Client là interface, Mockito mock được không?}
    Có. Mockito tạo proxy implementation của bất kỳ interface hoặc
    non-final class nào. Feign Client là interface nên mock trực tiếp.
  \end{block}

\begin{lstlisting}[language=java]
@ExtendWith(
    MockitoExtension.class)
class OrderServiceTest {

    @Mock private OrderRepository    orderRepository;
    @Mock private InventoryClient    inventoryClient;  // Feign interface
    @Mock private KafkaTemplate<String, OrderPlacedEvent> kafkaTemplate;

    @InjectMocks private OrderService orderService;

    @Test
    void placeOrder_InStock_ShouldCallInventoryWithCorrectArgs() {
        // Stub: khi goi isInStock("SKU-A", 3) thi tra ve true
        when(inventoryClient.isInStock("SKU-A", 3))
            .thenReturn(true);

        orderService.placeOrder(new OrderRequest("SKU-A", 3));

        // Verify Feign duoc goi dung 1 lan, dung tham so
        verify(inventoryClient, times(1))
            .isInStock("SKU-A", 3);
    }
}
\end{lstlisting}
\end{frame}

%--- SLIDE 11 ----------------------------------------------------------
\begin{frame}[fragile]{Stubbing nâng cao --- Nhiều lần gọi khác nhau}
\begin{lstlisting}[language=java]
// Lan goi thu nhat tra true, lan thu hai tra false
when(inventoryClient.isInStock(anyString(), anyInt()))
    .thenReturn(true)
    .thenReturn(false);

// Lan goi thu nhat throw exception, lan thu hai tra ket qua
when(inventoryClient.isInStock("FLAKY-SKU", anyInt()))
    .thenThrow(new RuntimeException("timeout"))
    .thenReturn(true);

// Stub dua tren gia tri tham so cu the
when(inventoryClient.isInStock(eq("SKU-A"), anyInt()))
    .thenReturn(true);
when(inventoryClient.isInStock(eq("SKU-B"), anyInt()))
    .thenReturn(false);
when(inventoryClient.isInStock(eq("SKU-NULL"), anyInt()))
    .thenThrow(new RuntimeException("SKU not found"));
\end{lstlisting}

  \begin{block}{doReturn vs thenReturn}
    \texttt{thenReturn}: cách thông thường, không dùng được với \texttt{@Spy} khi method void.\\
    \texttt{doReturn(...).when(mock).method()}: dùng khi mock void method hoặc khi dùng \texttt{@Spy}.
  \end{block}
\end{frame}

%--- SLIDE 13 ----------------------------------------------------------
\begin{frame}[fragile]{ArgumentCaptor --- Xác minh Order được save\sref{mockito}}
\begin{lstlisting}[language=java]
@Test
void placeOrder_Success_ShouldSaveOrderWithCorrectFields() {
    when(inventoryClient.isInStock("SKU-PRO", 5))
        .thenReturn(true);

    orderService.placeOrder(new OrderRequest("SKU-PRO", 5));

    // Bat doi tuong Order truyen vao save()
    ArgumentCaptor<Order> orderCaptor =
        ArgumentCaptor.forClass(Order.class);
    verify(orderRepository).save(orderCaptor.capture());

    Order saved = orderCaptor.getValue();
    assertAll("Order fields",
        () -> assertEquals("SKU-PRO",  saved.getSkuCode()),
        () -> assertEquals(5,           saved.getQuantity()),
        () -> assertNotNull(            saved.getOrderNumber()),
        () -> assertFalse(
                saved.getOrderNumber().isBlank())
    );
}
\end{lstlisting}
\end{frame}

@Test
void placeOrder_Success_ShouldPublishCorrectKafkaEvent() {

    orderService.placeOrder(new OrderRequest("SKU-X", 3));

}

\begin{frame}[fragile,shrink=22]{ArgumentCaptor --- Nhiều lần gọi}
\begin{lstlisting}[language=java]
// Gia su placeOrder co the xu ly nhieu don hang
// (hoac test service khac goi save() nhieu lan)
@Test
void batchOrder_ShouldSaveAllOrders() {
    when(inventoryClient.isInStock(anyString(), anyInt()))
        .thenReturn(true);

    orderService.placeOrder(new OrderRequest("SKU-A", 1));
    orderService.placeOrder(new OrderRequest("SKU-B", 2));

    // Lay TAT CA cac lan goi save()
    ArgumentCaptor<Order> captor =
        ArgumentCaptor.forClass(Order.class);
    verify(orderRepository, times(2)).save(captor.capture());

    List<Order> allOrders = captor.getAllValues(); // (!)
    assertEquals(2, allOrders.size());
    assertEquals("SKU-A", allOrders.get(0).getSkuCode());
    assertEquals("SKU-B", allOrders.get(1).getSkuCode());
}
\end{lstlisting}

  \begin{block}{getAllValues() vs getValue()}
    \texttt{getValue()}: lấy argument của lần gọi \textit{cuối cùng}.\\
    \texttt{getAllValues()}: lấy \textit{danh sách} argument của tất cả các lần gọi.
  \end{block}
\end{frame}

void placeOrder_OutOfStock_ShouldThrowAndHaveNoSideEffects() {

\begin{frame}[fragile]{Data Flow Testing --- Phân tích \texttt{placeOrder()}\sref{istqb}}

  \small

\end{frame}

%--- SLIDE 25 ----------------------------------------------------------
\begin{frame}[shrink=20]{Chiến lược đạt coverage mục tiêu}
  \begin{block}{Quy trình lặp để tăng coverage}
    \begin{enumerate}
      \item Chạy \texttt{mvn test} $\to$ xem JaCoCo report
      \item Mở class cần cải thiện trong report
      \item Tìm dòng màu \textbf{vàng} (branch partly covered) và \textbf{đỏ} (uncovered)
      \item Xác định: test case nào sẽ đi vào nhánh đó?
      \item Viết test case, chạy lại
      \item Lặp đến khi đạt ngưỡng
    \end{enumerate}
  \end{block}

  \vspace{0.3em}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{exampleblock}{Ví dụ: dòng vàng tại L3}
      Điều kiện: \texttt{request.getQuantity() == null || request.getQuantity() <= 0}\\[0.3em]
      Màu vàng: nhánh \texttt{quantity == null} chưa test\\[0.3em]
      Thêm test: \texttt{new OrderRequest("SKU", null)}\\
      $\Rightarrow$ dòng chuyển sang xanh
    \end{exampleblock}

    \column{0.5\textwidth}
    \begin{alertblock}{Không nên làm}
      Viết test chỉ để tăng số coverage mà không có assertion thực sự.
      Coverage tăng nhưng không phát hiện được lỗi.\\[0.3em]
      \textit{Mutation testing} (PIT) sẽ phát hiện
      những test như vậy.
    \end{alertblock}
  \end{columns}
\end{frame}
