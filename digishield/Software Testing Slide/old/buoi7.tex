%--- SLIDE 5 ----------------------------------------------------------
\begin{frame}{So sánh chi tiết: Unit Test vs Integration Test}
  \begin{center}
  \small
  \begin{tabular}{p{2.5cm}p{3.5cm}p{4.5cm}}
    \toprule
    \textbf{Thuộc tính} & \textbf{Unit Test} & \textbf{Integration Test} \\
    \midrule
    Phạm vi          & Một class/method & Nhiều layer cộng tác \\
    Dependencies     & Tất cả mock       & DB, Kafka, HTTP thật \\
    Tốc độ           & $<$ 10ms          & 3--30 giây (Testcontainers) \\
    Isolation        & Hoàn toàn cô lập  & Phụ thuộc môi trường \\
    Độ tin cậy       & Cao (không flaky) & Có thể flaky nếu setup sai \\
    Phát hiện lỗi    & Logic nội tại     & Tương tác layer, mapping ORM, transaction \\
    Khi fail         & Rõ nguồn gốc      & Có thể khó debug \\
    Chi phí CI       & Thấp              & Cao hơn (cần Docker) \\
    \midrule
    Tỉ lệ đề nghị   & 70\%              & 20\% \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \begin{exampleblock}{Khi nào tuyệt đối cần integration test?}
    \begin{itemize}
      \item ORM mapping phức tạp (cascade, lazy loading, N+1 query)
      \item Custom JPQL / native SQL query
      \item Kafka producer/consumer end-to-end
      \item Transaction rollback behavior
    \end{itemize}
  \end{exampleblock}
\end{frame}

%--- SLIDE 6 ----------------------------------------------------------
\begin{frame}[fragile]{\texttt{@SpringBootTest} --- MOCK và RANDOM\_PORT\sref{spring-boot-testing}}
  \begin{block}{Chế độ 1: MOCK (mặc định) --- không cần port thật}
\begin{lstlisting}[language=Java,style=nonumber]
@SpringBootTest  // mac dinh: webEnvironment = MOCK
@AutoConfigureMockMvc
class ProductControllerTest {
    @Autowired MockMvc mockMvc;
    // Dung MockMvc.perform() thay vi HTTP thuc su
}
\end{lstlisting}
  \end{block}

  \begin{block}{Chế độ 2: RANDOM\_PORT --- port ngẫu nhiên}
\begin{lstlisting}[language=Java,style=nonumber]
@SpringBootTest(webEnvironment = RANDOM_PORT)
class ProductControllerIntegrationTest {
    @Autowired TestRestTemplate restTemplate;
    @LocalServerPort int port;
    // Goi HTTP thuc su den localhost:port
}
\end{lstlisting}
  \end{block}

\end{frame}

\begin{frame}[fragile]{\texttt{@SpringBootTest} --- DEFINED\_PORT\sref{spring-boot-testing}}
  \begin{block}{Chế độ 3: DEFINED\_PORT --- port cố định (8080)}
\begin{lstlisting}[language=Java,style=nonumber]
@SpringBootTest(webEnvironment = DEFINED_PORT)
// Dung khi can goi tu Selenium / external tools
// Co the xung dot neu port da bi chiem
\end{lstlisting}
  \end{block}

  \begin{alertblock}{Lưu ý khi chọn port cố định}
    Chỉ dùng khi công cụ bên ngoài bắt buộc gọi vào một port biết trước.
    Trong CI/CD, nên ưu tiên \texttt{RANDOM\_PORT} để tránh xung đột port.
  \end{alertblock}
\end{frame}

%--- SLIDE 7 ----------------------------------------------------------
\begin{frame}[fragile]{Slice Testing --- Web layer\sref{spring-boot-testing}}
    \begin{block}{\texttt{@WebMvcTest} --- Chỉ Web Layer}

    // H2 mac dinh (hoac DB that)
    // Khong load Service/Controller
}
\end{lstlisting}
      \textbf{Dùng khi}: test custom query,
      JPQL, native SQL, constraint.
    \end{block}
\end{frame}

%--- SLIDE 8 ----------------------------------------------------------
\begin{frame}{Chọn loại test phù hợp}
  \begin{center}
  \small
  \begin{tabular}{p{4cm}p{3cm}p{2.5cm}}
    \toprule
    \textbf{Muốn kiểm thử} & \textbf{Loại test} & \textbf{Tốc độ} \\
    \midrule
    Logic trong Service class         & Unit + Mockito        & $<$ 10ms \\
    Routing và request mapping        & @WebMvcTest           & $\sim$500ms \\
    Custom JPA query với H2           & @DataJpaTest          & $\sim$1s \\
    Repository với MongoDB thật       & @DataMongoTest + TC   & $\sim$5s \\
    Controller + Service + DB thật    & @SpringBootTest + TC  & $\sim$10s \\
    Toàn bộ app bao gồm Kafka         & @SpringBootTest + TC  & $\sim$20s \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \begin{block}{TC = Testcontainers}
    Testcontainers cần Docker đang chạy. Lần đầu pull image sẽ chậm.
    Các lần sau dùng image đã cache, nhanh hơn nhiều.
  \end{block}

  \begin{alertblock}{Nguyên tắc}
    Dùng loại test \textbf{nhẹ nhất} có thể xác minh được điều cần kiểm thử.
    \texttt{@SpringBootTest} full context chỉ khi thực sự cần.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{\texttt{@DynamicPropertySource} --- Inject cấu hình container\sref{spring-framework-dynamicproperty}}
  \begin{block}{Vấn đề: port thay đổi mỗi lần chạy}
    Container được gán port ngẫu nhiên. Không thể hardcode trong
    \texttt{application.properties}. Cần cơ chế động.
  \end{block}

\begin{lstlisting}[language=Java]
@SpringBootTest(webEnvironment = RANDOM_PORT)
@Testcontainers
class ProductControllerIntegrationTest {

    @Container                                       // (1)
    static MongoDBContainer mongoDBContainer =
        new MongoDBContainer("mongo:6.0");

    @DynamicPropertySource                           // (2)
    static void setMongoProperties(DynamicPropertyRegistry registry) {
        registry.add(
            "spring.data.mongodb.uri",              // (3) ten property
            mongoDBContainer::getReplicaSetUrl      // (4) ham lay gia tri
        );
    }
    // ...
}
\end{lstlisting}

  \vspace{-0.2em}
  {\small
  (1) \texttt{static}: container chia sẻ cho tất cả test trong class\\
  (2) Chạy \textit{trước} khi Spring context được tạo\\
  (3) Override property trong \texttt{application.properties}\\
  (4) Lambda trả về URL với port thực của container
  }
\end{frame}

\section{Integration Test cho ProductController}
%=====================================================================

%--- SLIDE 13 ----------------------------------------------------------
\begin{frame}[fragile]{Integration Test --- \texttt{ProductController} với MongoDB thật}
\begin{lstlisting}[language=Java]
@SpringBootTest(webEnvironment = RANDOM_PORT)
@Testcontainers
class ProductControllerIntegrationTest {

    @Container
    static MongoDBContainer mongoDBContainer =
        new MongoDBContainer("mongo:6.0");

    @Autowired TestRestTemplate restTemplate;
    @Autowired ProductRepository productRepository;

    @DynamicPropertySource
    static void setProperties(DynamicPropertyRegistry registry) {
        registry.add("spring.data.mongodb.uri",
                     mongoDBContainer::getReplicaSetUrl);
    }

    @BeforeEach
    void setUp() {
        productRepository.deleteAll();   // (!) dam bao DB sach truoc moi test
    }

    // Test methods...
}
\end{lstlisting}

  \begin{alertblock}{\texttt{@BeforeEach deleteAll()}}
    Với container-per-class, dữ liệu từ test trước còn lại trong DB.

@Test

    \column{0.55\textwidth}
    \textbf{Kết quả chạy thành công:}
    \begin{itemize}
      \item Tất cả \texttt{@Test} trong IT class pass
      \item Không có \texttt{BeforeEach} fail
      \item Log hiển thị container khởi động thành công
      \item Response time hợp lý ($<$ 5s cho mỗi test)
    \end{itemize}

    \vspace{0.3em}
    \textbf{Lỗi thường gặp và cách xử lý:}
    \begin{itemize}
      \item \texttt{ContainerLaunchException}: Docker không chạy
        $\to$ khởi động Docker Desktop
      \item \texttt{DataIntegrityViolationException}: thiếu \texttt{@BeforeEach deleteAll()}
        $\to$ thêm dọn dẹp trước mỗi test
      \item Test flaky (đôi pass đôi fail): do data còn sót
        $\to$ dùng \texttt{@Transactional} rollback hoặc \texttt{deleteAll()}
    \end{itemize}

        \item Dùng \texttt{static} container\\
