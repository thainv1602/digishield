\section{Kiểm thử phần mềm là gì?}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 6
%---------------------------------------------------------------------
\begin{frame}{Định nghĩa}
  \begin{block}{IEEE 610.12-1990}
    Kiểm thử phần mềm là quá trình \textbf{vận hành} một hệ thống hoặc thành phần
    trong các điều kiện được chỉ định, \textbf{quan sát} và \textbf{ghi lại kết quả},
    và \textbf{đánh giá} một số khía cạnh của hệ thống hoặc thành phần đó.
  \end{block}

  \vspace{0.8em}
  \begin{columns}
    \column{0.48\textwidth}
    \textbf{Kiểm thử KHÔNG phải là:}
    \begin{itemize}
      \item Chứng minh phần mềm đúng
      \item Đảm bảo không còn lỗi
      \item Chỉ chạy chương trình xem có lỗi không
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Kiểm thử LÀ:}
    \begin{itemize}
      \item Tìm kiếm lỗi một cách có hệ thống
      \item Đánh giá chất lượng phần mềm
      \item Cung cấp thông tin để ra quyết định
    \end{itemize}
  \end{columns}
\end{frame}

\section{Bảy nguyên tắc kiểm thử (ISTQB)}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 10
%---------------------------------------------------------------------
\begin{frame}{Tổng quan 7 nguyên tắc ISTQB}
  \begin{enumerate}
    \item Kiểm thử cho thấy sự \textbf{hiện diện} của lỗi, không phải sự \textbf{vắng mặt}
    \item Kiểm thử \textbf{toàn diện} là không khả thi
    \item Kiểm thử \textbf{sớm} tiết kiệm chi phí
    \item \textbf{Phân cụm} lỗi
    \item \textbf{Nghịch lý thuốc trừ sâu}
    \item Kiểm thử phụ thuộc vào \textbf{ngữ cảnh}
    \item \textbf{Ngụy biện} không có lỗi
  \end{enumerate}

  \vspace{0.5em}
  \begin{block}{Lưu ý}
    Các nguyên tắc này không phải quy tắc cứng nhắc mà là \textbf{hướng dẫn tư duy}
    cho kiểm thử viên khi ra quyết định.
  \end{block}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 11
%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 1: Kiểm thử cho thấy sự hiện diện của lỗi}
  \begin{block}{Phát biểu}
    Kiểm thử có thể chứng minh phần mềm \textbf{có lỗi}, nhưng \textbf{không thể}
    chứng minh phần mềm hoàn toàn không có lỗi.
  \end{block}

  \vspace{0.5em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Hệ quả thực tế:}
    \begin{itemize}
      \item Khi tất cả test case pass, không có nghĩa phần mềm đúng
      \item Cần liên tục thiết kế test case mới
      \item Mục tiêu: \textit{giảm rủi ro}, không phải \textit{đảm bảo hoàn hảo}
    \end{itemize}

    \column{0.5\textwidth}
    \begin{exampleblock}{Ví dụ}
      \texttt{OrderService} vượt qua 100 test case không có nghĩa không còn
      lỗi trong logic xử lý đơn hàng đồng thời (concurrency).
    \end{exampleblock}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 12
%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 2: Kiểm thử toàn diện là không khả thi}
  \begin{block}{Vấn đề}
    Số lượng tổ hợp đầu vào của bất kỳ hệ thống phi tầm thường nào
    là \textbf{vô hạn hoặc quá lớn} để kiểm thử đầy đủ.
  \end{block}

  \vspace{0.5em}
  \begin{exampleblock}{Ví dụ định lượng}
    API \texttt{POST /api/order} có hai trường:
    \begin{itemize}
      \item \texttt{skuCode}: chuỗi tối đa 50 ký tự
      \item \texttt{quantity}: số nguyên
    \end{itemize}
    Số tổ hợp đầu vào lý thuyết: $> 10^{100}$ --- không thể kiểm thử hết.
  \end{exampleblock}

  \vspace{0.3em}
  \textbf{Giải pháp}: ưu tiên hóa test case dựa trên \textbf{rủi ro} và
  \textbf{tầm quan trọng nghiệp vụ} (sẽ học ở Buổi 2: EP, BVA).
\end{frame}

%---------------------------------------------------------------------
% SLIDE 13
%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 3 \& 4: Kiểm thử sớm và Phân cụm lỗi}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Nguyên tắc 3: Kiểm thử sớm}
      \begin{itemize}
        \item Bắt đầu kiểm thử ngay từ giai đoạn phân tích yêu cầu
        \item Review đặc tả, review thiết kế --- đây là kiểm thử tĩnh
        \item Chi phí sửa lỗi tăng theo cấp bậc khi phát hiện muộn
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Nguyên tắc 4: Phân cụm lỗi}
      Nguyên lý Pareto (80/20): phần lớn lỗi tập trung ở
      \textbf{một số ít module}.

      \vspace{0.3em}
      Hệ quả: tập trung nỗ lực kiểm thử vào các module:
      \begin{itemize}
        \item Phức tạp về logic
        \item Thay đổi thường xuyên
        \item Nghiệp vụ quan trọng
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------

\end{frame}

% SLIDE 15
%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc 6 \& 7}

      \textit{Trong CI/CD}: mỗi lần push code, toàn bộ regression test
      chạy tự động.

\begin{frame}[fragile,shrink=10]{Đọc mã nguồn: OrderService}

\begin{lstlisting}[language=java,basicstyle=\ttfamily\tiny]
@Service
@RequiredArgsConstructor
public class OrderService {
  private final OrderRepository orderRepository;
  private final InventoryClient inventoryClient; // Feign
  private final KafkaTemplate<String, OrderPlacedEvent> kafka;

  @Transactional
  public void placeOrder(OrderRequest request) {
    boolean inStock = inventoryClient.isInStock(
        request.getSkuCode(), request.getQuantity());

    if (!inStock) {
      throw new RuntimeException("Product not in stock");
    }

    Order order = mapToOrder(request);
    orderRepository.save(order);
    kafka.send("notificationTopic",
        new OrderPlacedEvent(order.getOrderNumber()));
  }
}
\end{lstlisting}
  \end{block}

    \begin{enumerate}
      \item \texttt{inStock == true} $\to$ tạo order, publish Kafka
      \item \texttt{inStock == false} $\to$ throw exception
    \end{enumerate}

    \vspace{0.5em}
    \textbf{Xác định dependencies:}
    \begin{itemize}
      \item \texttt{OrderRepository} --- MySQL
      \item \texttt{InventoryClient} --- HTTP call
      \item \texttt{KafkaTemplate} --- Kafka
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Test cases cần viết:}
    \begin{exampleblock}{}
      \begin{itemize}
        \item TC-01: inventory đủ $\to$ order được save
        \item TC-02: inventory đủ $\to$ Kafka event được gửi
        \item TC-03: inventory = 0 $\to$ exception throw
        \item TC-04: inventory = 0 $\to$ repository.save() \textbf{không} được gọi
        \item TC-05: orderNumber được generate tự động
      \end{itemize}
    \end{exampleblock}

    \begin{block}{Cấu trúc \texttt{ProductService}}
      \begin{itemize}
        \item \texttt{createProduct(...)} $\to$ lưu MongoDB
        \item \texttt{getAllProducts()} $\to$ trả về danh sách
        \item Không có dependency nội bộ phức tạp
        \item Phù hợp để học unit test ở \textbf{Buổi 3}
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Cấu trúc \texttt{InventoryService}}
      \begin{itemize}
        \item \texttt{isInStock(List skuCodes)} $\to$ query MySQL
        \item Logic: so sánh số lượng tồn kho với yêu cầu
        \item Trả về \texttt{List<InventoryResponse>}
        \item Phù hợp để học boundary testing ở \textbf{Buổi 2}
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.5em}
  \begin{alertblock}{Bài tập}
    Đọc mã nguồn của product-service và inventory-service.\\
    Lập danh sách 5 method ưu tiên kiểm thử, kèm lý giải.
  \end{alertblock}
\end{frame}

\section{Test Plan và Test Strategy}
%=====================================================================

%---------------------------------------------------------------------
% SLIDE 33
%---------------------------------------------------------------------
\begin{frame}{Test Plan là gì?}
  \begin{block}{Định nghĩa}
    Test Plan là tài liệu mô tả \textbf{phạm vi, cách tiếp cận, tài nguyên và lịch trình}
    của hoạt động kiểm thử.
  \end{block}

  \vspace{0.5em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Nội dung Test Plan (IEEE 829):}
    \begin{itemize}
      \item Test item: phần mềm/module nào
      \item Test features: chức năng cần test
      \item Test approach: phương pháp
      \item Pass/Fail criteria
      \item Test deliverables
      \item Responsibilities
      \item Schedule
    \end{itemize}

    \column{0.5\textwidth}
    \begin{exampleblock}{Ví dụ cho order-service}
      \begin{itemize}
        \item Item: \texttt{placeOrder()}
        \item Features: happy path, insufficient stock, null input
        \item Approach: unit test với Mockito
        \item Pass criteria: 100\% test pass, coverage $\geq$ 80\%
        \item Deadline: Buổi 4
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
% SLIDE 34
%---------------------------------------------------------------------
\begin{frame}[shrink=10]{Test Strategy so với Test Plan}
  \begin{center}
  \begin{tabular}{p{2.5cm}p{4cm}p{4cm}}
    \toprule
    \textbf{Thuộc tính} & \textbf{Test Strategy} & \textbf{Test Plan} \\
    \midrule
    Phạm vi    & Toàn tổ chức / dự án    & Một vòng kiểm thử cụ thể \\
    Người viết & Test Manager / QA Lead  & Test Lead / Engineer \\
    Nội dung   & Nguyên tắc, công cụ, quy trình chung & Chi tiết: scope, resource, schedule \\
    Tính ổn định & Ít thay đổi            & Cập nhật theo từng release \\
    Ví dụ      & ``Mọi API đều phải có Postman test trước khi merge'' & ``Sprint 5: test 3 endpoint mới của order-service'' \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Test Report}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{Test Summary Report}
      Tổng kết sau mỗi vòng kiểm thử:
      \begin{itemize}
        \item Tổng số test case thực thi
        \item Pass / Fail / Blocked / Not Run
        \item Danh sách defect tìm được
        \item Đánh giá rủi ro còn lại
        \item Khuyến nghị (release hay không)
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{exampleblock}{Ví dụ từ Postman}
      Sau khi chạy Postman Collection bằng Runner:
      \begin{itemize}
        \item Requests: 12
        \item Tests: 36
        \item Passed: 34 (94.4\%)
        \item Failed: 2
        \item Duration: 3.2s
      \end{itemize}
      Report: tab \texttt{Run Results} hoặc export kết quả từ Postman.
    \end{exampleblock}
  \end{columns}
\end{frame}
