\section{Khuôn mẫu đặc tả Test Case}
%=====================================================================

%---------------------------------------------------------------------
\begin{frame}{Vì sao cần khuôn mẫu đặc tả chuẩn?}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Không có đặc tả chuẩn}
      \begin{itemize}
        \item Test case ``ở trong đầu'' một người
        \item Người khác chạy $\to$ kết quả khác nhau
        \item Không truy vết được về yêu cầu
        \item Không đo được tiến độ kiểm thử
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Có đặc tả chuẩn}
      \begin{itemize}
        \item Bất kỳ ai cũng thực thi được y hệt
        \item Pass/Fail là khách quan
        \item Truy vết: yêu cầu $\leftrightarrow$ test $\leftrightarrow$ defect
        \item Là đầu vào để \textbf{tự động hóa}
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{Chuẩn tham chiếu}
    ISO/IEC/IEEE 29119-3 (Test documentation) định nghĩa các thành phần
    tối thiểu của một Test Case Specification.
  \end{exampleblock}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}[shrink=6]{Các trường của khuôn mẫu đặc tả}
  \begin{center}
  \small
  \renewcommand{\arraystretch}{0.95}
  \begin{tabular}{>{\raggedright\arraybackslash}p{3.2cm}>{\raggedright\arraybackslash}p{8.6cm}}
    \toprule
    \textbf{Trường} & \textbf{Nội dung} \\
    \midrule
    TC-ID          & Mã duy nhất, vd \texttt{TC-INT-001} (INT = module interception) \\
    Tiêu đề        & Một câu mô tả điều được kiểm thử \\
    Mục tiêu       & Kiểm chứng điều gì, vì sao quan trọng \\
    Tham chiếu     & ID yêu cầu / user story / dòng bảng quyết định \\
    Kỹ thuật       & EP / BVA / Decision Table / State / Use Case \\
    Precondition   & Trạng thái hệ thống + dữ liệu cần có trước khi chạy \\
    Test Data      & Dữ liệu đầu vào \textbf{cụ thể} (không ghi ``giá trị hợp lệ'') \\
    Test Steps     & Các bước đánh số, mỗi bước một hành động \\
    Expected Result& Kết quả mong đợi --- cụ thể, đo được \\
    Actual Result  & Kết quả thực tế (điền khi thực thi) \\
    Status         & Pass / Fail / Blocked / Not Run \\
    Severity       & Mức nghiêm trọng nếu fail (Critical $\to$ Low) \\
    Người thực hiện, ngày & Ai chạy, chạy lúc nào, trên môi trường nào \\
    \bottomrule
  \end{tabular}
  \renewcommand{\arraystretch}{1}
  \end{center}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Nguyên tắc viết Expected Result}
  \begin{block}{Expected Result phải: cụ thể --- đo được --- không mơ hồ}
  \end{block}

  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{alertblock}{Sai --- quá mơ hồ}
      \begin{itemize}
        \item ``Hệ thống chặn giao dịch''
        \item ``Trả về lỗi''
        \item ``API hoạt động đúng''
      \end{itemize}
    \end{alertblock}

    \column{0.5\textwidth}
    \begin{exampleblock}{Đúng --- cụ thể}
      \begin{itemize}
        \item ``HTTP 200; \texttt{decision="PAUSE"}''
        \item ``HTTP 403; không lộ dữ liệu''
        \item ``\texttt{signals} chứa đúng 3 phần tử''
      \end{itemize}
    \end{exampleblock}
  \end{columns}

  \vspace{0.3em}
  \begin{alertblock}{Quy tắc vàng}
    Nếu hai người đọc cùng đặc tả mà kết luận Pass/Fail khác nhau
    thì Expected Result \textbf{chưa đủ cụ thể}.
  \end{alertblock}
\end{frame}

\section{REST API và bản đồ API DigiShield}
%=====================================================================

%---------------------------------------------------------------------
\begin{frame}{API testing = kiểm thử hộp đen ở tầng HTTP}
  \begin{block}{Vì sao thực thi test case qua API?}
    \begin{itemize}
      \item Đúng tinh thần hộp đen: chỉ thấy request vào / response ra.
      \item Nhanh và ổn định hơn test qua giao diện (buổi 10--11).
      \item Kiểm tra được routing, phân quyền, serialization JSON ---
        những thứ unit test (buổi 5--7) không chạm tới.
    \end{itemize}
  \end{block}

  \begin{center}
  \scriptsize
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{p{2.4cm}p{2.2cm}p{3.4cm}p{3.6cm}}
    \toprule
    \textbf{Khía cạnh} & \textbf{Kỹ thuật} & \textbf{Ví dụ DigiShield} & \textbf{Assertion} \\
    \midrule
    Status code & EP / BVA & 200, 401, 403 & \texttt{status === 200} \\
    Body & Cấu trúc, miền giá trị & \texttt{decision}, \texttt{confidence} & \texttt{oneOf}, \texttt{within} \\
    Phân quyền & Ma trận role & LEARNER gọi API ANALYST & \texttt{status === 403} \\
    Thời gian & Hiệu năng & $< 500$ms & \texttt{responseTime < 500} \\

%---------------------------------------------------------------------
\begin{frame}{Pre-request vs Post-response}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Pre-request script}
      Chạy \textbf{trước} khi gửi request:
      \begin{itemize}
        \item Sinh dữ liệu động: UUID, timestamp
        \item Tính giá trị phụ thuộc
        \item Chuẩn bị precondition
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{Post-response (Tests)}
      Chạy \textbf{sau} khi nhận response:
      \begin{itemize}
        \item Assertion Pass/Fail
        \item Lưu biến cho request sau
        \item Ghi log chẩn đoán
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.3em}
  \begin{exampleblock}{Đối chiếu với khuôn mẫu đặc tả}
    Pre-request $\approx$ \emph{Precondition + Test Data};
    Post-response $\approx$ \emph{Expected Result}. Đặc tả tốt thì
    chuyển sang script gần như một--một.
  \end{exampleblock}
\end{frame}

      \begin{itemize}
        \item \texttt{.to.equal / .to.eql}
        \item \texttt{.to.have.property}
        \item \texttt{.to.be.oneOf([...])}
        \item \texttt{.to.be.within(a, b)}
      \end{itemize}

  \end{columns}
\end{frame}

\begin{frame}{Collection Runner --- chạy cả chuỗi}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{block}{Cách chạy}
      \begin{enumerate}
        \item Chọn collection $\to$ \emph{Run collection}
        \item Chọn Environment \texttt{dev-local}
        \item Iterations = 1, giữ nguyên thứ tự
        \item \emph{Run DigiShield API}
      \end{enumerate}

      chúng ghi là \textbf{Blocked}, không phải Fail.

    vì vậy \textbf{tên test phải mô tả hành vi}, không đặt ``test 1'', ``test 2''.
  \end{block}

\begin{frame}[shrink=6]{Yêu cầu tối thiểu của collection}
  \begin{block}{Collection \texttt{DigiShield API} nộp cuối giờ}
    \begin{itemize}
      \item \(\geq\) \textbf{8 request}, mỗi request \(\geq\) 3 assertion có tên mô tả
      \item Phủ \textbf{happy path}: health, login, me, evaluate (PAUSE),
        watchlist, classify, submit report, triage
      \item \(\geq\) \textbf{4 ca negative}: 401 (không auth), 403 (LEARNER
        gọi \texttt{/interventions}), body thiếu \texttt{channel}
        (ghi nhận defect), \texttt{size=101} (clamp)
      \item \(\geq\) \textbf{1 chuỗi data chaining} hoàn chỉnh
        (report\_id đi qua 3 request)
      \item Mỗi request ghi \textbf{TC-ID} tương ứng trong phần mô tả
    \end{itemize}

\begin{frame}{Bài nộp (về nhà --- gắn với BTL, hạn: trước buổi 10)}
  \setbeamerfont{itemize/enumerate body}{size=\small}
  \begin{block}{Sản phẩm nộp (theo nhóm BTL, cho module nhóm phụ trách)}
    \begin{enumerate}
      \item \textbf{Bảng đặc tả test case} (Google Sheets): \(\geq\) 10 TC
        theo khuôn mẫu 13 trường, có tham chiếu kỹ thuật buổi 8
      \item \texttt{module.postman\_collection.json}: \(\geq\) 12 request
        (\(\geq\) 4 negative), mỗi request \(\geq\) 3 assertion
      \item \texttt{dev-local.postman\_environment.json}
        + file CSV data-driven (nếu có)
      \item \texttt{report.html} của Newman (htmlextra), pass rate \(\geq\) 90\%
      \item \textbf{Defect log}: mỗi ca fail một dòng --- TC-ID, mô tả,
        expected/actual, severity
    \end{enumerate}
  \end{block}

  {\small \textbf{Tiêu chí chấm}: đặc tả rõ ràng truy vết được (30\%) ---
  assertion đúng và đủ (30\%) --- negative có hệ thống (20\%) --- Newman chạy
  được một lệnh, report sạch (20\%).}
\end{frame}
